39static int get_file_type(
40 char const * const fname)
41{
42 /* find last . in filename */
43 char const * const suffix = strrchr(fname, '.');
44 if(suffix == NULL) {
45 goto NOT_FOUND;
46 }
47
48 size_t idx = 0;
49 do {
50 if(strcmp(suffix, file_extensions[idx].extension) == 0) {
51 return file_extensions[idx].type;
52 }
53 } while(file_extensions[++idx].extension != NULL);
54
55
56 /* default to text coordinate format */
57 NOT_FOUND:
58 fprintf(stderr, "SPLATT: extension for '%s' not recognized. "
59 "Defaulting to ASCII coordinate form.\n", fname);
60 return 0;
61}

63static int p_tt_read_file(ptiSparseTensor *tsr, ptiIndex start_index, FILE *fp)
64{
65 int iores, retval;
66 ptiIndex mode;
67 iores = fscanf(fp, "%u", &tsr->nmodes);
68 pti_CheckOSError(iores < 0, "SpTns Load");
69 /* Only allocate space for sortorder, initialized to 0s. */
70 tsr->sortorder = malloc(tsr->nmodes * sizeof tsr->sortorder[0]);
71 pti_CheckOSError(!tsr->sortorder, "SpTns Load");
72 memset(tsr->sortorder, 0, tsr->nmodes * sizeof tsr->sortorder[0]);
73 tsr->ndims = malloc(tsr->nmodes * sizeof *tsr->ndims);
74 pti_CheckOSError(!tsr->ndims, "SpTns Load");
75 for(mode = 0; mode < tsr->nmodes; ++mode) {
76 iores = fscanf(fp, "%u", &tsr->ndims[mode]);
77 pti_CheckOSError(iores != 1, "SpTns Load");
78 }
79 tsr->nnz = 0;
80 tsr->inds = malloc(tsr->nmodes * sizeof *tsr->inds);
81 pti_CheckOSError(!tsr->inds, "SpTns Load");
82 for(mode = 0; mode < tsr->nmodes; ++mode) {
83 retval = ptiNewIndexVector(&tsr->inds[mode], 0, 0);
84 pti_CheckError(retval, "SpTns Load", NULL);
85 }
86 retval = ptiNewValueVector(&tsr->values, 0, 0);
87 pti_CheckError(retval, "SpTns Load", NULL);
88 while(retval == 0) {
89 double value;
90 for(mode = 0; mode < tsr->nmodes; ++mode) {
91 ptiIndex index;
92 iores = fscanf(fp, "%u", &index);
93 if(iores != 1) {
94 retval = -1;
95 break;
96 }
97 if(index < start_index) {
98 pti_CheckError(PTIERR_VALUE_ERROR, "SpTns Load", "index < start_index");
99 }
100 ptiAppendIndexVector(&tsr->inds[mode], index-start_index);
101 }
102 if(retval == 0) {
103 iores = fscanf(fp, "%lf", &value);
104 if(iores != 1) {
105 retval = -1;
106 break;
107 }
108 ptiAppendValueVector(&tsr->values, value);
109 ++tsr->nnz;
110 }
111 }
112 for(mode = 0; mode < tsr->nmodes; ++mode) {
113 tsr->inds[mode].len = tsr->nnz;
114 }
115 // ptiSparseTensorCollectZeros(tsr);
116
117 return 0;
118}

121static void read_binary_header(
122 FILE * fin,
123 bin_header * header)
124{
125 fread(&(header->magic), sizeof(header->magic), 1, fin);
126 fread(&(header->idx_width), sizeof(header->idx_width), 1, fin);
127 fread(&(header->val_width), sizeof(header->val_width), 1, fin);
128
129 if(header->idx_width > HIPARTI_INDEX_TYPEWIDTH / 8) {
130 fprintf(stderr, "SPLATT: ERROR input has %lu-bit integers. "
131 "Build with HIPARTI_INDEX_TYPEWIDTH %lu\n",
132 header->idx_width * 8, header->idx_width * 8);
133 exit(-1);
134 }
135
136 if(header->val_width > HIPARTI_VALUE_TYPEWIDTH / 8) {
137 fprintf(stderr, "SPLATT: WARNING input has %lu-bit floating-point values. "
138 "Build with HIPARTI_VALUE_TYPEWIDTH %lu for full precision\n",
139 header->val_width * 8, header->val_width * 8);
140 }
141}

143static void fill_binary_idx(
144 ptiIndex * const buffer,
145 ptiIndex const count,
146 bin_header const * const header,
147 FILE * fin)
148{
149 if(header->idx_width == sizeof(ptiIndex)) {
150 fread(buffer, sizeof(ptiIndex), count, fin);
151 } else {
152 /* read in uint32_t in a buffered fashion */
153 ptiIndex const BUF_LEN = 1024*1024;
154 uint32_t * ubuf = (uint32_t*)malloc(BUF_LEN * sizeof(*ubuf));
155 for(ptiIndex n=0; n < count; n += BUF_LEN) {
156 ptiIndex const read_count = BUF_LEN < count - n ? BUF_LEN : count - n;
157 fread(ubuf, sizeof(*ubuf), read_count, fin);
158 #pragma omp parallel for schedule(static)
159 for(ptiIndex i=0; i < read_count; ++i) {
160 buffer[n + i] = ubuf[i];
161 }
162 }
163 free(ubuf);
164 }
165}

193static void fill_binary_val(
194 ptiValue * const buffer,
195 ptiIndex const count,
196 bin_header const * const header,
197 FILE * fin)
198{
199 if(header->val_width == sizeof(ptiValue)) {
200 fread(buffer, sizeof(ptiValue), count, fin);
201 } else {
202 /* read in float in a buffered fashion */
203 ptiIndex const BUF_LEN = 1024*1024;
204
205 /* select whichever SPLATT *is not* configured with. */
206#if HIPARTI_VALUE_TYPEWIDTH == 64

208#else
209 double * ubuf = (double*)malloc(BUF_LEN * sizeof(*ubuf));
210#endif

212 for(ptiIndex n=0; n < count; n += BUF_LEN) {
213 ptiIndex const read_count = BUF_LEN < count - n ? BUF_LEN : count - n;
214 fread(ubuf, sizeof(*ubuf), read_count, fin);
215 #pragma omp parallel for schedule(static)
216 for(ptiIndex i=0; i < read_count; ++i) {
217 buffer[n + i] = ubuf[i];
218 }
219 }
220 free(ubuf);

232static int p_tt_read_binary_file(ptiSparseTensor *tsr, FILE * fin)
233{
234 int result;
235 bin_header header;
236 read_binary_header(fin, &header);
237 // printf("header.magic: %d\n", header.magic);
238 // printf("header.idx_width: %lu\n", header.idx_width);
239 // printf("header.val_width: %lu\n", header.val_width);
240
241 ptiNnzIndex nnz = 0;
242 ptiIndex nmodes = 0;
243
244 fill_binary_idx(&nmodes, 1, &header, fin);
245
246 ptiIndex * dims = (ptiIndex *) malloc (nmodes * sizeof(*dims));
247 fill_binary_idx(dims, nmodes, &header, fin);
248 fill_binary_nnzidx(&nnz, 1, &header, fin);
249
250 /* allocate structures */
251 ptiNewSparseTensor(tsr, nmodes, dims);
252 tsr->nnz = nnz;
253 for(ptiIndex m=0; m < nmodes; ++m) {
254 result = ptiResizeIndexVector(&tsr->inds[m], nnz);
255 pti_CheckError(result, "SpTns Read", NULL);
256 }
257 result = ptiResizeValueVector(&tsr->values, nnz);
258 pti_CheckError(result, "SpTns Read", NULL);
259
260 /* fill in tensor data */
261 for(ptiIndex m=0; m < nmodes; ++m) {
262 fill_binary_idx(tsr->inds[m].data, nnz, &header, fin);
263 }
264 fill_binary_val(tsr->values.data, nnz, &header, fin);
265
266 return 0;
267}

277int ptiLoadSparseTensor(ptiSparseTensor *tsr, ptiIndex start_index, char const * const fname)
278{
279 FILE * fp = fopen(fname, "r");
280 ptiAssert(fp != NULL);
281
282 /* A TensorSuite .tns shares the extension but not the layout; detect it by banner. */
283 if(pti_TensorSuiteSniff(fp)) {
284 fclose(fp);
285 return ptiLoadSparseTensorTensorSuite(tsr, start_index, fname, PTI_TS_FILL_ONES);
286 }
287
288 int iores;
289 switch(get_file_type(fname)) {
290 case 0:
291 iores = p_tt_read_file(tsr, start_index, fp);
292 pti_CheckOSError(iores != 0, "SpTns Load");
293 break;
294 case 1:
295 iores = p_tt_read_binary_file(tsr, fp);
296 pti_CheckOSError(iores != 0, "SpTns Load");
297 break;
298 }
299
300 fclose(fp);
301
302 return 0;
303}

306void ptiLoadShuffleFile(ptiSparseTensor *tsr, FILE *fs, ptiIndex ** map_inds)
307{
308 ptiNnzIndex line_count = 0;
309 ptiNnzIndex dim_count = 0;
310 int iores;
311 for(ptiIndex mode = 0; mode < tsr->nmodes; ++mode) {
312 dim_count += tsr->ndims[mode];
313 for(ptiIndex i = 0; i < tsr->ndims[mode]; ++i) {
314 iores = fscanf(fs, "%u", &(map_inds[mode][i]));
315 -- map_inds[mode][i];
316 ++ line_count;
317 }
318 }
319 ptiAssert(dim_count == line_count);
320
321 return;
322}
