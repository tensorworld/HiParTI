27int ptiSparseTensorKhatriRaoMul(ptiSparseTensor *Y, const ptiSparseTensor *A, const ptiSparseTensor *B) {
28 ptiIndex nmodes;
29 ptiIndex mode;
30 ptiIndex *inds;
31 ptiNnzIndex i, j;
32 int result;
33 if(A->nmodes != B->nmodes) {
34 pti_CheckError(PTIERR_SHAPE_MISMATCH, "Khatri-Rao", "shape mismatch");
35 }
36 nmodes = A->nmodes;
37 if(nmodes == 0) {
38 pti_CheckError(PTIERR_SHAPE_MISMATCH, "Khatri-Rao", "shape mismatch");
39 }
40 if(A->ndims[nmodes-1] != B->ndims[nmodes-1]) {
41 pti_CheckError(PTIERR_SHAPE_MISMATCH, "Khatri-Rao", "shape mismatch");
42 }
43 inds = malloc(nmodes * sizeof *inds);
44 pti_CheckOSError(!inds, "Khatri-Rao");
45 for(mode = 0; mode < nmodes-1; ++mode) {
46 inds[mode] = A->ndims[mode] * B->ndims[mode];
47 }
48 inds[nmodes-1] = A->ndims[mode];
49 result = ptiNewSparseTensor(Y, nmodes, inds);
50 free(inds);
51 pti_CheckError(result, "Khatri-Rao", NULL);
52 /* For each element in A and B */
53 for(i = 0; i < A->nnz; ++i) {
54 for(j = 0; j < B->nnz; ++j) {
55 if(A->inds[nmodes-1].data[i] == B->inds[nmodes-1].data[j]) {
56 /*
57 Y[f(i0,j0), ..., f(i(N-2), j(N-2))] = a[i10 ..., i(N-2)] * b[j0, ..., j(N-2)]
58 where f(in, jn) = jn + in * Jn
59 */
60 for(mode = 0; mode < nmodes-1; ++mode) {
61 ptiAppendIndexVector(&Y->inds[mode], A->inds[mode].data[i] * B->ndims[mode] + B->inds[mode].data[j]);
62 }
63 ptiAppendIndexVector(&Y->inds[nmodes-1], A->inds[nmodes-1].data[i]);
64 ptiAppendValueVector(&Y->values, A->values.data[i] * B->values.data[j]);
65 ++Y->nnz;
66 }
67 }
68 }
69 ptiSparseTensorSortIndex(Y, 1, 1);
70 return 0;
71}
