41int ptiCudaMTTKRPOneKernel(
42 ptiSparseTensor const * const X,
43 ptiMatrix ** const mats, // mats[nmodes] as temporary space.
44 ptiIndex * const mats_order, // Correspond to the mode order of X.
45 ptiIndex const mode,
46 ptiIndex const impl_num)
47{
48 ptiIndex const nmodes = X->nmodes;
49 ptiNnzIndex const nnz = X->nnz;
50 ptiIndex const * const ndims = X->ndims;
51 ptiIndex const R = mats[mode]->ncols;
52 ptiIndex const stride = mats[mode]->stride;
53 int result;
54
55 double time_h2d, time_exe, time_d2h;
56 double gbw_h2d, gflops_exe, gbytes_exe, gbw_d2h;
57 ptiTimer timer;
58 ptiNewTimer(&timer, 0);
59
60 /* Check the mats. */
61 for(ptiIndex i=0; i<nmodes; ++i) {
62 if(mats[i]->ncols != mats[nmodes]->ncols) {
63 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CUDA SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
64 }
65 if(mats[i]->nrows != ndims[i]) {
66 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CUDA SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
67 }
68 }
69
70
71 /* Transfer tensor and matrices */
72 /* dev_mats_order: 1st gpu. */
73 ptiIndex * dev_mats_order;
74 /* dev_Xndims: 1st gpu. */
75 ptiIndex * dev_Xndims;
76 /* dev_Xvals: 1st gpu. */
77 ptiValue * dev_Xvals;
78 /* Xinds_header: 1st cpu, 2nd cpu (ghost pointers) */
79 ptiIndex ** Xinds_header = new ptiIndex *[nmodes];
80 /* dev_Xinds: 1st gpu, 2nd gpu. */
81 ptiIndex ** dev_Xinds;
82 /* mats_header: 1st cpu, 2nd cpu (ghost pointers) */
83 ptiValue ** mats_header = new ptiValue *[nmodes+1];
84 /* lengths: 1st cpu, store the lengths of mats */
85 ptiNnzIndex * const lengths = new ptiNnzIndex[nmodes+1];
86 /* dev_mats: 1st gpu, 2nd gpu. */
87 ptiValue ** dev_mats;
88 /* dev_scratch: 1st gpu. */
89 ptiValue * dev_scratch;
90 /* the pointer to dev_mats[nmodes] */
91 ptiValue *dev_part_prod;
92 ptiNnzIndex dev_mem_size = 0;
93 uint64_t dev_flops = 2 * nnz * R + (nmodes - 1) * R;
94 uint64_t dev_bytes = ( nmodes * sizeof(ptiIndex) + sizeof(ptiValue) ) * nnz;
95 for (ptiIndex m=0; m<nmodes; ++m) {
96 dev_bytes += ndims[m] * R * sizeof(ptiValue);
97 }
98
99
100 ptiStartTimer(timer);
101
102 /* dev_mats_order */
103 result = ptiCudaDuplicateMemory(&dev_mats_order, mats_order, nmodes * sizeof (ptiIndex), cudaMemcpyHostToDevice);
104 pti_CheckCudaError(result != 0, "CUDA SpTns SpltMTTKRP");
105 dev_mem_size += nmodes * sizeof (ptiIndex);
106
107 /* dev_Xndims */
108 result = ptiCudaDuplicateMemory(&dev_Xndims, ndims, nmodes * sizeof (ptiIndex), cudaMemcpyHostToDevice);
109 pti_CheckCudaError(result != 0, "CUDA SpTns SpltMTTKRP");
110 dev_mem_size += nmodes * sizeof (ptiIndex);
111
112 /* dev_Xvals */
113 result = ptiCudaDuplicateMemory(&dev_Xvals, X->values.data, nnz * sizeof (ptiValue), cudaMemcpyHostToDevice);
114 pti_CheckCudaError(result != 0, "CUDA SpTns SpltMTTKRP");
115 dev_mem_size += nnz * sizeof (ptiValue);
116
117 /* Xinds_header */
118 for(ptiIndex m = 0; m < nmodes; ++m) {
119 Xinds_header[m] = X->inds[m].data;
120 }
121 /* dev_Xinds */
122 result = ptiCudaDuplicateMemoryIndirect(&dev_Xinds, Xinds_header, nmodes, nnz, cudaMemcpyHostToDevice);
123 pti_CheckCudaError(result != 0, "CUDA SpTns SpltMTTKRP");
124 dev_mem_size += nmodes * nnz * sizeof(ptiIndex);
125
126 /* mats_header and lengths */
127 memset(mats[nmodes]->values, 0, mats[mode]->nrows * stride * sizeof(ptiValue));
128 ptiNnzIndex sum_mat_length = 0;
129 for(ptiIndex m = 0; m < nmodes; ++m) {
130 mats_header[m] = mats[m]->values;
131 lengths[m] = mats[m]->nrows * stride;
132 sum_mat_length += mats[m]->nrows * stride;
133 }
134 mats_header[nmodes] = mats[nmodes]->values;
135 lengths[nmodes] = mats[mode]->nrows * stride;
136 sum_mat_length += mats[mode]->nrows * stride;
137 /* dev_mats */
138 result = ptiCudaDuplicateMemoryIndirect(&dev_mats, mats_header, nmodes+1, lengths, cudaMemcpyHostToDevice);
139 pti_CheckCudaError(result != 0, "CUDA SpTns SpltMTTKRP");
140 dev_mem_size += sum_mat_length * sizeof(ptiValue);
141
142 if(nmodes > 4) {
143 /* dev_scratch */
144 result = cudaMalloc((void **) &dev_scratch, nnz * stride * sizeof (ptiValue));
145 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
146 result = cudaMemset(dev_scratch, 0, nnz * stride * sizeof (ptiValue));
147 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
148 dev_mem_size += nnz * stride * sizeof (ptiValue);
149 }
150
151 ptiStopTimer(timer);
152 time_h2d = ptiElapsedTime(timer);
153 gbw_h2d = dev_mem_size / time_h2d /1e9;
154 ptiPrintElapsedTime(timer, "CUDA SpTns MTTKRP H2D");
155 printf("[Bandwidth H2D]: %lf GBytes/sec\n", gbw_h2d);
156
157
158 // ptiNnzIndex max_nthreads_per_block = 512; // old run
159 ptiNnzIndex max_nthreads_per_block = 256;
160 ptiNnzIndex max_nblocks = 32768;
161 ptiNnzIndex max_nthreadsy = 16;
162
163 ptiNnzIndex nthreadsx = 0;
164 ptiNnzIndex nthreadsy = 0;
165 ptiNnzIndex all_nblocks = 0;
166 ptiNnzIndex nblocks = 0;
167 switch(impl_num) {
168 // case 1:
169 case 11: // Naive, 1D
170 if(nnz < max_nthreads_per_block) {
171 nthreadsx = nnz;
172 nblocks = 1;
173 } else {
174 nthreadsx = max_nthreads_per_block;
175 all_nblocks = (nnz + nthreadsx -1) / nthreadsx;
176 if(all_nblocks < max_nblocks) {
177 nblocks = all_nblocks;
178 } else {
179 nblocks = max_nblocks;
180 }
181 }
182 break;
183 // case 2: // 2D
184 // case 3: // 2D, rank split
185 // if(R <= max_nthreadsy)
186 // nthreadsy = R;
187 // else
188 // nthreadsy = max_nthreadsy;
189 // nthreadsx = max_nthreads_per_block / nthreadsy;
190 // all_nblocks = (nnz + nthreadsx -1) / nthreadsx;
191 // break;
192 // case 4: // 2D, exchange x and y
193 // nthreadsx = R;
194 // nthreadsy = max_nthreads_per_block / nthreadsx;
195 // all_nblocks = (nnz + nthreadsy -1) / nthreadsy;
196 // break;
197 // case 5:
198 case 15: // 2D, exchange x and y, rank split. Best performance
199 case 16:
200 if(R <= max_nthreadsy)
201 nthreadsx = R;
202 else
203 nthreadsx = max_nthreadsy;
204 nthreadsy = max_nthreads_per_block / nthreadsx;
205
206 if(nnz < nthreadsy) {
207 nthreadsy = nnz;
208 nblocks = 1;
209 } else {
210 all_nblocks = (nnz + nthreadsy -1) / nthreadsy;
211 if(all_nblocks < max_nblocks) {
212 nblocks = all_nblocks;
213 } else {
214 nblocks = max_nblocks;
215 }
216 }
217 break;
218 default:
219 fprintf(stderr, "[CUDA SpTns MTTKRP OneKernel] Error: unsupported impl_num %d (valid: 11,15,16).\n", (int) impl_num);
220 return -1;
221 }
222 if(nthreadsx == 0) {
223 fprintf(stderr, "[CUDA SpTns MTTKRP OneKernel] Error: unsupported impl_num %d, no kernel geometry selected.\n", (int) impl_num);
224 return -1;
225 }
226 dim3 dimBlock(nthreadsx, nthreadsy);
227 printf("all_nblocks: %lu, nthreadsx: %lu, nthreadsy: %lu\n", all_nblocks, nthreadsx, nthreadsy);
228
229
230 ptiStartTimer(timer);
231
232 if(nmodes != 3) {
233 fprintf(stderr, "[CUDA SpTns MTTKRP OneKernel] Error: unsupported nmodes %u.\n", nmodes);
234 return -1;
235 }
236 switch(nmodes) {
237 case 3:
238 switch(impl_num) {
239 // case 1:
240 case 11: // Naive
241 printf("Execute pti_MTTKRPKernelNnz3DOneKernel (%lu, %lu)\n", nblocks, nthreadsx);
242 pti_MTTKRPKernelNnz3DOneKernel<<<nblocks, nthreadsx>>>(
243 mode,
244 nmodes,
245 nnz,
246 R,
247 stride,
248 dev_Xndims,
249 dev_Xinds,
250 dev_Xvals,
251 dev_mats_order,
252 dev_mats);
253 break;
254 // case 2:
255 // case 5:
256 case 15:
257 printf("Execute pti_MTTKRPKernelRankSplitNnz3DOneKernel (%lu, (%u, %u))\n", nblocks, dimBlock.x, dimBlock.y);
258 pti_MTTKRPKernelRankSplitNnz3DOneKernel<<<nblocks, dimBlock>>>(
259 mode,
260 nmodes,
261 nnz,
262 R,
263 stride,
264 dev_Xndims,
265 dev_Xinds,
266 dev_Xvals,
267 dev_mats_order,
268 dev_mats);
269 break;
270 case 16:
271 printf("Execute pti_MTTKRPKernelRankSplitNnzRB3DOneKernel (%lu, (%u, %u))\n", nblocks, dimBlock.x, dimBlock.y);
272 pti_MTTKRPKernelRankSplitNnzRB3DOneKernel<<<nblocks, dimBlock>>>(
273 mode,
274 nmodes,
275 nnz,
276 R,
277 stride,
278 dev_Xndims,
279 dev_Xinds,
280 dev_Xvals,
281 dev_mats_order,
282 dev_mats);
283 break;
284 default:
285 fprintf(stderr, "[CUDA SpTns MTTKRP OneKernel] Error: unsupported impl_num %d (valid: 11,15,16).\n", (int) impl_num);
286 return -1;
287 } // End switch impl_num
288 break;
289
290 case 4:
291 switch(impl_num) {
292 default:
293 printf("Not support: Execute pti_MTTKRPKernelScratch (%lu, %lu)\n", nblocks, nthreadsx);
294 // pti_MTTKRPKernelScratch<<<nblocks, nthreadsx>>>(
295 // mode,
296 // nmodes,
297 // nnz,
298 // R,
299 // stride,
300 // dev_Xndims,
301 // dev_Xinds,
302 // dev_Xvals,
303 // dev_mats_order,
304 // dev_mats,
305 // dev_scratch,
306 // block_offset);
307 } // End switch impl_num
308 break;
309
310 default:
311 printf("Execute pti_MTTKRPKernelScratch (%lu, %lu)\n", nblocks, nthreadsx);
312 // pti_MTTKRPKernelScratch<<<nblocks, nthreadsx>>>(
313 // mode,
314 // nmodes,
315 // nnz,
316 // R,
317 // stride,
318 // dev_Xndims,
319 // dev_Xinds,
320 // dev_Xvals,
321 // dev_mats_order,
322 // dev_mats,
323 // dev_scratch,
324 // block_offset);
325 } // End switch nmodes
326 result = cudaDeviceSynchronize();
327 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
328
329
330
331 ptiStopTimer(timer);
332 time_exe = ptiElapsedTime(timer);
333 gflops_exe = (double)dev_flops / time_exe / 1e9;
334 gbytes_exe = (double)dev_bytes / time_exe / 1e9;
335 ptiPrintElapsedTime(timer, "CUDA SpTns MTTKRP");
336 printf("[GFLOPS]: %.2lf GFlops, [Bandwidth]: %.2lf GB/s\n", gflops_exe, gbytes_exe);
337
338 ptiStartTimer(timer);
339
340 dev_mem_size = 0;
341 /* Copy back the pointer to dev_mats[nmodes] to the result */
342 result = cudaMemcpy(&dev_part_prod, dev_mats + nmodes, sizeof dev_part_prod, cudaMemcpyDeviceToHost);
343 pti_CheckCudaError(result != 0, "CUDA SpTns SpltMTTKRP");
344 dev_mem_size += sizeof dev_part_prod;
345
346 result = cudaMemcpy(mats[nmodes]->values, dev_part_prod, mats[mode]->nrows * stride * sizeof (ptiValue), cudaMemcpyDeviceToHost);
347 pti_CheckCudaError(result != 0, "CUDA SpTns SpltMTTKRP");
348 dev_mem_size += mats[mode]->nrows * stride * sizeof (ptiValue);
349
350 ptiStopTimer(timer);
351 time_d2h = ptiElapsedTime(timer);
352 gbw_d2h = dev_mem_size / time_d2h /1e9;
353 ptiPrintElapsedTime(timer, "CUDA SpTns MTTKRP D2H");
354 printf("[Bandwidth D2H]: %lf GBytes/sec\n", gbw_d2h);
355 printf("\n");
356 ptiFreeTimer(timer);
357
358 result = cudaFree(dev_mats_order);
359 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
360 result = cudaFree(dev_Xndims);
361 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
362 result = cudaFree(dev_Xvals);
363 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
364 result = cudaFree(dev_Xinds);
365 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
366 result = cudaFree(dev_mats);
367 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
368 if(nmodes > 4) {
369 result = cudaFree(dev_scratch);
370 pti_CheckCudaError(result != 0, "CUDA SpTns MTTKRP");
371 }
372 delete[] Xinds_header;
373 delete[] mats_header;
374 delete[] lengths;
375
376 return 0;
