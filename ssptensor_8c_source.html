32int ptiNewSemiSparseTensor(ptiSemiSparseTensor *tsr, ptiIndex nmodes, ptiIndex mode, const ptiIndex ndims[]) {
33 ptiIndex i;
34 int result;
35 if(nmodes < 2) {
36 pti_CheckError(PTIERR_SHAPE_MISMATCH, "SspTns New", "nmodes < 2");
37 }
38 tsr->nmodes = nmodes;
39 tsr->ndims = malloc(nmodes * sizeof *tsr->ndims);
40 pti_CheckOSError(!tsr->ndims, "SspTns New");
41 memcpy(tsr->ndims, ndims, nmodes * sizeof *tsr->ndims);
42 tsr->mode = mode;
43 tsr->nnz = 0;

52 pti_CheckError(result, "SspTns New", NULL);

132 tsr->strides[i] = ((ndims[dmodes[i]]-1)/8+1)*8;

135 pti_CheckError(result, "SspTns New", NULL);
