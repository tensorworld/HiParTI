27{
28 int result;
29 ptiIndex *ind_buf;
30 ptiNnzIndexVector fiberidx;
31 if(mode >= X->nmodes) {
32 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns * Vec", "shape mismatch");
33 }
34 if(X->ndims[mode] != V->len) {
35 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns * Vec", "shape mismatch");
36 }
37 ptiSparseTensorSortIndexAtMode(X, mode, 0);
38 // jli: try to avoid malloc in all operation functions.
39 ind_buf = malloc(X->nmodes * sizeof *ind_buf);
40 pti_CheckOSError(!ind_buf, "CPU SpTns * Vec");
41 for(ptiIndex m = 0; m < X->nmodes; ++m) {
42 ind_buf[m] = X->ndims[m];
43 }
44 ind_buf[mode] = 1;
45 // jli: use pre-processing to allocate Y size outside this function.
46 result = ptiNewSemiSparseTensor(Y, X->nmodes, mode, ind_buf);
47 free(ind_buf);
48 pti_CheckError(result, "CPU SpTns * Vec", NULL);
49 ptiSemiSparseTensorSetIndices(Y, &fiberidx, X);
50
51 ptiTimer timer;
52 ptiNewTimer(&timer, 0);
53 ptiStartTimer(timer);
54
55 for(ptiNnzIndex i = 0; i < Y->nnz; ++i) {
56 ptiNnzIndex inz_begin = fiberidx.data[i];
57 ptiNnzIndex inz_end = fiberidx.data[i+1];
58 ptiNnzIndex j;
59 // jli: exchange the two loops
60 for(j = inz_begin; j < inz_end; ++j) {
61 ptiIndex r = X->inds[mode].data[j];
62 Y->values.values[i*Y->stride] += X->values.data[j] * V->data[r];
63 }
64 }
65
66 ptiStopTimer(timer);
67 ptiPrintElapsedTime(timer, "CPU SpTns * Vec");
68 ptiFreeTimer(timer);
69
70 ptiFreeNnzIndexVector(&fiberidx);
71 return 0;
