24int ptiNewSparseMatrix(ptiSparseMatrix *mtx, ptiIndex nrows, ptiIndex ncols, ptiIndex nnz)
25{
26 int result;
27 mtx->nrows = nrows;
28 mtx->ncols = ncols;
29 mtx->nnz = nnz;
30 result = ptiNewIndexVector(&mtx->rowind, nnz, nnz);
31 pti_CheckError(result, "SpMtx New", NULL);
32 result = ptiNewIndexVector(&mtx->colind, nnz, nnz);
33 pti_CheckError(result, "SpMtx New", NULL);
34 result = ptiNewValueVector(&mtx->values, nnz, nnz);
35 pti_CheckError(result, "SpMtx New", NULL);
36
37 return 0;
38}

54void ptiSparseMatrixShuffleIndices(ptiSparseMatrix *mtx, ptiIndex ** map_inds) {
55 /* Renumber nonzero elements */
56 ptiIndex tmp_ind;
57 for(ptiNnzIndex z = 0; z < mtx->nnz; ++z) {
58 tmp_ind = mtx->rowind.data[z];
59 mtx->rowind.data[z] = map_inds[0][tmp_ind];
60 tmp_ind = mtx->colind.data[z];
61 mtx->colind.data[z] = map_inds[1][tmp_ind];
62 }
63
64}
