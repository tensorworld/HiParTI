24int ptiNewSparseMatrixCSR(ptiSparseMatrixCSR *csrmtx, ptiIndex nrows, ptiIndex ncols, ptiIndex nnz)
25{
26 int result;
27 csrmtx->nrows = nrows;
28 csrmtx->ncols = ncols;
29 csrmtx->nnz = nnz;
30 result = ptiNewNnzIndexVector(&csrmtx->rowptr, (nrows+1), (nrows+1));
31 pti_CheckError(result, "SpMtx New", NULL);
32 result = ptiNewIndexVector(&csrmtx->colind, nnz, nnz);
33 pti_CheckError(result, "SpMtx New", NULL);
34 result = ptiNewValueVector(&csrmtx->values, nnz, nnz);
35 pti_CheckError(result, "SpMtx New", NULL);
36
37 return 0;
38}
