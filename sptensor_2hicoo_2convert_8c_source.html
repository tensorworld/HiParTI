28static int ptiLargerThanCoordinates(
29 ptiSparseTensor *tsr,
30 const ptiNnzIndex z,
31 const ptiIndex * item)
32{
33 ptiIndex nmodes = tsr->nmodes;
34 ptiIndex i1, i2;
35
36 for(ptiIndex m=0; m<nmodes; ++m) {
37 i1 = tsr->inds[m].data[z];
38 i2 = item[m];
39 if(i1 > i2) {
40 return 1;
41 break;
42 }
43 }
44 return 0;
45}

144static int ptiNextBlockBegin(
145 ptiIndex * out_item,
146 ptiSparseTensor *tsr,
147 const ptiIndex * in_item,
148 const ptiElementIndex sb)
149{
150 ptiIndex nmodes = tsr->nmodes;
151
152 for(int32_t m=nmodes-1; m>=0; --m) {
153 if(in_item[m] < tsr->ndims[m]-1) {
154 out_item[m] = in_item[m]+sb-1 < tsr->ndims[m] ? in_item[m]+sb-1 : tsr->ndims[m] - 1;
155 break;
156 }
157 }
158
159 return 0;

211static int ptiKernelStrides(
212 ptiIndex * strides,
213 ptiSparseTensor *tsr,
214 const ptiIndex sk)
215{
216 ptiIndex nmodes = tsr->nmodes;
217 ptiIndex kernel_size = 0;
218
219 // TODO: efficiently use bitwise operation
220 strides[nmodes-1] = 1;
221 for(ptiIndex m=nmodes-2; m>=1; --m) {
222 kernel_size = (ptiIndex)(tsr->ndims[m+1] + sk - 1) / sk;
223 strides[m] = strides[m+1] * kernel_size;
224 }
225 kernel_size = (ptiIndex)(tsr->ndims[1] + sk - 1) / sk;
226 strides[0] = strides[1] * kernel_size;
227
228 return 0;
229}

264int ptiGetRowBlockPointers(
265 ptiNnzIndexVector *mptr,
266 ptiSparseTensor *tsr,
267 const ptiIndex sk)
268{
269 ptiNnzIndex nnz = tsr->nnz;
270 ptiIndex i = tsr->inds[0].data[0];
271 ptiNnzIndex k = 0; // count blocks
272 ptiNnzIndex knnz = 0; // #Nonzeros per block
273 mptr->data[0] = 0;
274 while(1) {
275 /* check if mode-0 index in block-b */
276 if(i >= sk * k && i < sk * (k+1)) {
277 ++ knnz;
278 break;
279 } else {
280 ++ k;
281 mptr->data[k] = knnz + mptr->data[k-1];
282 knnz = 0;
283 }
284 }
285
286
287 for(ptiNnzIndex z=1; z<nnz; ++z) {
288 i = tsr->inds[0].data[z];
289 /* Compare with the next block row index */
290 while(1) {
291 if(i >= sk * k && i < sk * (k+1)) {
292 ++ knnz;
293 break;
294 } else {
295 ++ k;
296 mptr->data[k] = knnz + mptr->data[k-1];
297 knnz = 0;
298 }
299 }
300 }
301 ptiAssert(k < (tsr->ndims[0] + sk -1 ) / sk);
302 ptiAssert(mptr->data[mptr->len-1] + knnz == nnz);
303
304 return 0;
305}

314int ptiSetKernelPointers(
315 ptiNnzIndexVector *kptr,
316 ptiNnzIndexVector *knnzs,
317 ptiSparseTensor *tsr,
318 const ptiElementIndex sk_bits)
319{
320 ptiIndex nmodes = tsr->nmodes;
321 ptiNnzIndex nnz = tsr->nnz;
322 ptiNnzIndex k = 0; // count kernels
323 ptiNnzIndex knnz = 0; // #Nonzeros per kernel
324 int result = 0;
325 result = ptiAppendNnzIndexVector(kptr, 0);
326 pti_CheckError(result, "HiSpTns Convert", NULL);
327
328 ptiIndex * coord = (ptiIndex *)malloc(nmodes * sizeof(*coord));
329 ptiIndex * kernel_coord = (ptiIndex *)malloc(nmodes * sizeof(*kernel_coord));
330 ptiIndex * kernel_coord_prior = (ptiIndex *)malloc(nmodes * sizeof(*kernel_coord_prior));
331
332 /* Process first nnz to get the first kernel_coord_prior */
333 for(ptiIndex m=0; m<nmodes; ++m)
334 coord[m] = tsr->inds[m].data[0]; // first nonzero indices
335 result = ptiLocateBeginCoord(kernel_coord_prior, tsr, coord, sk_bits);
336 pti_CheckError(result, "HiSpTns Convert", NULL);
337
338 for(ptiNnzIndex z=0; z<nnz; ++z) {
339 for(ptiIndex m=0; m<nmodes; ++m)
340 coord[m] = tsr->inds[m].data[z];
341 result = ptiLocateBeginCoord(kernel_coord, tsr, coord, sk_bits);
342 pti_CheckError(result, "HiSpTns Convert", NULL);
343
344 if(ptiEqualWithTwoCoordinates(kernel_coord, kernel_coord_prior, nmodes) == 1) {
345 ++ knnz;
346 } else {
347 ++ k;
348 result = ptiAppendNnzIndexVector(kptr, knnz + kptr->data[k-1]);
349 pti_CheckError(result, "HiSpTns Convert", NULL);
350 result = ptiAppendNnzIndexVector(knnzs, knnz);
351 pti_CheckError(result, "HiSpTns Convert", NULL);
352 for(ptiIndex m=0; m<nmodes; ++m)
353 kernel_coord_prior[m] = kernel_coord[m];
354 knnz = 1;
355 }
356 }
357 ptiAssert(k < kptr->len);
358 ptiAssert(kptr->data[kptr->len-1] + knnz == nnz);
359
360 /* Set the last element for kptr */
361 ptiAppendNnzIndexVector(kptr, nnz);
362 ptiAppendNnzIndexVector(knnzs, knnz);
363
364 free(coord);
365 free(kernel_coord);
366 free(kernel_coord_prior);
367
368 return 0;
369}

378static int ptiSetKernelScheduler(
379 ptiIndexVector **kschr,
380 ptiIndex *nkiters,
381 ptiNnzIndexVector * const kptr,
382 ptiSparseTensor *tsr,
383 const ptiElementIndex sk_bits)
384{
385 ptiIndex nmodes = tsr->nmodes;
386 ptiIndex * ndims = tsr->ndims;
387 int result = 0;
388
389 ptiIndex * coord = (ptiIndex *)malloc(nmodes * sizeof(*coord));
390 ptiIndex * kernel_coord = (ptiIndex *)malloc(nmodes * sizeof(*kernel_coord));
391
392 for(ptiNnzIndex k=0; k<kptr->len - 1; ++k) {
393 ptiNnzIndex z = kptr->data[k];
394 for(ptiIndex m=0; m<nmodes; ++m)
395 coord[m] = tsr->inds[m].data[z];
396 result = ptiLocateBeginCoord(kernel_coord, tsr, coord, sk_bits);
397 pti_CheckError(result, "HiSpTns Convert", NULL);
398
399 for(ptiIndex m=0; m<nmodes; ++m) {
400 result = ptiAppendIndexVector(&(kschr[m][kernel_coord[m]]), k);
401 pti_CheckError(result, "HiSpTns Convert", NULL);
402 }
403 }
404
405 free(coord);
406 free(kernel_coord);
407
408 ptiIndex sk = (ptiIndex)pow(2, sk_bits);
409 ptiIndex tmp;
410 for(ptiIndex m=0; m<nmodes; ++m) {
411 tmp = 0;
412 ptiIndex kernel_ndim = (ndims[m] + sk - 1) / sk;
413 for(ptiIndex i=0; i<kernel_ndim; ++i) {
414 if(tmp < kschr[m][i].len)
415 tmp = kschr[m][i].len;
416 }
417 nkiters[m] = tmp;
418 }
419
420 return 0;
421}

430int ptiPreprocessSparseTensor(
431 ptiNnzIndexVector * kptr,
432 ptiIndexVector **kschr,
433 ptiIndex *nkiters,
434 ptiIndexVector **kschr_balanced,
435 ptiIndexVector **kschr_balanced_pos,
436 ptiIndex *nkpars,
437 ptiIndexVector * kschr_rest,
438 ptiNnzIndexVector * knnzs,
439 ptiSparseTensor *tsr,
440 const ptiElementIndex sb_bits,
441 const ptiElementIndex sk_bits,
442 int const tk)
443{
444 ptiNnzIndex nnz = tsr->nnz;
445 int result;
446
447 // TODO: possible permute modes to improve parallelism
448
449 /* Sort tsr in a Row-major Block order to get all kernels. Not use Morton-order for kernels: 1. better support for higher-order tensors by limiting kernel size, because Morton key bit <= 128; */
450 ptiTimer rowblock_sort_timer;
451 ptiNewTimer(&rowblock_sort_timer, 0);
452 ptiStartTimer(rowblock_sort_timer);
453
454 ptiSparseTensorSortIndexRowBlock(tsr, 1, 0, nnz, sk_bits, tk); // Parallelized inside
455
456 ptiStopTimer(rowblock_sort_timer);
457 ptiPrintElapsedTime(rowblock_sort_timer, "\t\trowblock sorting");
458 ptiFreeTimer(rowblock_sort_timer);
459#if PARTI_DEBUG == 3
460 printf("Sorted by ptiSparseTensorSortIndexRowBlock.\n");
461 ptiAssert(ptiDumpSparseTensor(tsr, 0, stdout) == 0);
462#endif
463
464 ptiTimer set_kernel_timer;
465 ptiNewTimer(&set_kernel_timer, 0);
466 ptiStartTimer(set_kernel_timer);
467
468 result = ptiSetKernelPointers(kptr, knnzs, tsr, sk_bits);
469 pti_CheckError(result, "HiSpTns Preprocess", NULL);
470 result = ptiSetKernelScheduler(kschr, nkiters, kptr, tsr, sk_bits);
471 pti_CheckError(result, "HiSpTns Preprocess", NULL);
472 // printf("OK\n"); fflush(stdout);
473
474 /* Set balanced data structures: kschr_balanced, kschr_rest */
475 ptiNnzIndex avg_nnzk = tsr->nnz / (kptr->len - 1);
476 ptiNnzIndex max_nnzk = 0;
477 for(ptiIndex k=0; k<kptr->len - 1; ++k) {
478 ptiNnzIndex nnzk = knnzs->data[k];
479 if(max_nnzk < nnzk) max_nnzk = nnzk;
480 }
481 // ptiNnzIndex par_nnzk_th = 20 * avg_nnzk; // threshold for nnzk per thread
482 ptiNnzIndex par_nnzk_th = 5 * max_nnzk; // threshold for nnzk per thread
483 printf("par_nnzk_th: %lu\n", par_nnzk_th);
484 ptiIndex sk = (ptiIndex)pow(2, sk_bits);
485 // printf("OK-2\n"); fflush(stdout);
486
487 for(ptiIndex m=0; m < tsr->nmodes; ++m) { // Loop kschr for each mode
488
489 ptiIndexVector * restrict kschr_mode = kschr[m];
490 ptiIndexVector * restrict kschr_balanced_mode = kschr_balanced[m];
491 ptiIndexVector * restrict kschr_balanced_pos_mode = kschr_balanced_pos[m];
492 ptiIndex kernel_ndim = (tsr->ndims[m] + sk - 1)/sk;
493 for(ptiIndex i=0; i < kernel_ndim; ++i) {
494 ptiAppendIndexVector(&(kschr_balanced_pos_mode[i]), 0);
495 }
496 ptiIndex j_rest = nkiters[m];
497 ptiIndex npars = 0;
498 int tag_rest = 0;
499 ptiIndex count_nk = 0;
500 ptiIndex empty_schr_rows_th = 1.0 * kernel_ndim > 1 ? 1.0 * kernel_ndim : 1;
501 printf("[mode %u] empty_schr_rows_th: %u\n", m, empty_schr_rows_th);
502
503 while(tag_rest == 0 && count_nk < kptr->len - 1) { // Loop for partitions. tag_rest = 1, maybe there is no rest.
504 /* Check two ranges: npars and j or tmp_j !!! */
505 ptiIndex max_nnzk_per_col = 0, par_nnzk = 0;
506 ptiIndex count_empty_schr_rows = 0;
507 for(ptiIndex i=0; i < kernel_ndim; ++i) { // Find the max nnzk
508 if(count_empty_schr_rows > empty_schr_rows_th) {
509 tag_rest = 1;
510 break;
511 }
512 if(npars >= kschr_balanced_pos_mode[i].len) {
513 ++ count_empty_schr_rows;
514 continue;
515 } else {
516 ptiIndex j = kschr_balanced_pos_mode[i].data[npars];
517 if(j >= kschr_mode[i].len) {
518 ++ count_empty_schr_rows;
519 continue;
520 }
521 ptiIndex kernel_num = kschr_mode[i].data[j];
522 ptiNnzIndex kernel_nnz = knnzs->data[kernel_num];
523 if (max_nnzk_per_col < kernel_nnz) {
524 max_nnzk_per_col = kernel_nnz;
525 }
526 }
527 } // End of i
528
529 if(tag_rest == 1) { // an empty superblock met, to kschr_rest
530 for(ptiIndex i=0; i < kernel_ndim; ++i) {
531 if(npars >= kschr_balanced_pos_mode[i].len) continue;
532 ptiIndex j2 = kschr_balanced_pos_mode[i].data[npars];
533 for(; j2 < kschr_mode[i].len; ++j2) {
534 ptiAppendIndexVector(&kschr_rest[m], kschr_mode[i].data[j2]);
535 ++ count_nk;
536 }
537 }
538 } else { // all non-empty superblocks for this column, to kschr_balanced, kschr_balanced_pos
539 /* set par_nnzk */
540 if(max_nnzk_per_col > par_nnzk_th) {
541 par_nnzk = max_nnzk_per_col; // split according to the superblock with the max nnzk
542 } else {
543 par_nnzk = par_nnzk_th;
544 }
545
546 /* Real partition */
547 for(ptiIndex i=0; i < kernel_ndim; ++i) {
548 if(npars >= kschr_balanced_pos_mode[i].len) continue;
549 ptiIndex tmp_j = kschr_balanced_pos_mode[i].data[npars];
550 if(tmp_j >= kschr_mode[i].len) continue;
551 ptiIndex kernel_num = kschr_mode[i].data[tmp_j];
552 ptiNnzIndex sum_nnzk = knnzs->data[kernel_num];
553 while(sum_nnzk <= par_nnzk) {
554 ptiAppendIndexVector(&(kschr_balanced_mode[i]), kernel_num);
555 ++ count_nk;
556 ++ tmp_j;
557 if(tmp_j < kschr_mode[i].len) {
558 kernel_num = kschr_mode[i].data[tmp_j]; // j + 1
559 sum_nnzk += knnzs->data[kernel_num];
560 } else {
561 break;
562 }
563 } // End of while
564 ptiAppendIndexVector(&(kschr_balanced_pos_mode[i]), tmp_j);
565 }
566 ++ npars;
567 }
568 // printf("count_nk: %u\n", count_nk); fflush(stdout);
569 } // End of while
570 nkpars[m] = npars; // kschr_balanced_pos.len is npars + 1.
571 } // End loop of modes
572
573
574 ptiStopTimer(set_kernel_timer);
575 ptiPrintElapsedTime(set_kernel_timer, "\t\tSet Kernel Ptrs");
576 ptiFreeTimer(set_kernel_timer);
577
578 ptiTimer morton_sort_timer;
579 ptiNewTimer(&morton_sort_timer, 0);
580 ptiStartTimer(morton_sort_timer);
581
582 /* Sort blocks in each kernel in Morton-order */
583 ptiNnzIndex k_begin, k_end;
584 /* Loop for all kernels, 0-kptr.len for OMP code */
585 #pragma omp parallel for num_threads(tk)
586 for(ptiNnzIndex k=0; k<kptr->len - 1; ++k) {
587 k_begin = kptr->data[k];
588 k_end = kptr->data[k+1]; // exclusive
589 /* Sort blocks in each kernel in Morton-order */
590 ptiSparseTensorSortIndexMorton(tsr, 1, k_begin, k_end, sb_bits, tk);
591 // ptiSparseTensorSortIndexRowBlock(tsr, 1, k_begin, k_end, sb_bits, tk);
592#if PARTI_DEBUG == 3
593 printf("Kernel %"HIPARTI_PRI_NNZ_INDEX ": Sorted by ptiSparseTensorSortIndexMorton.\n", k);
594 ptiAssert(ptiDumpSparseTensor(tsr, 0, stdout) == 0);
595#endif
596 }
597
598 ptiStopTimer(morton_sort_timer);
599 ptiPrintElapsedTime(morton_sort_timer, "\t\tMorton sorting");
600 // ptiPrintElapsedTime(morton_sort_timer, "\t\t2nd Rowblock sorting");
601 ptiFreeTimer(morton_sort_timer);
602
603 return 0;
604}

646int ptiSparseTensorToHiCOO(
647 ptiSparseTensorHiCOO *hitsr,
648 ptiNnzIndex *max_nnzb,
649 ptiSparseTensor *tsr,
650 const ptiElementIndex sb_bits,
651 const ptiElementIndex sk_bits,
652 const ptiElementIndex sc_bits,
653 int const tk)
654{
655 ptiAssert(sk_bits >= sb_bits);
656 ptiAssert(sc_bits >= sb_bits);
657
658 ptiIndex i;
659 int result;
660 ptiIndex nmodes = tsr->nmodes;
661 ptiNnzIndex nnz = tsr->nnz;
662
663 ptiElementIndex sb = pow(2, sb_bits);
664 ptiIndex sc = pow(2, sc_bits);
665
666 /* Set HiCOO parameters. ndims for type conversion, size_t -> ptiIndex */
667 ptiIndex * ndims = malloc(nmodes * sizeof *ndims);
668 pti_CheckOSError(!ndims, "HiSpTns Convert");
669 for(i = 0; i < nmodes; ++i) {
670 ndims[i] = (ptiIndex)tsr->ndims[i];
671 }
672
673 result = ptiNewSparseTensorHiCOO(hitsr, (ptiIndex)tsr->nmodes, ndims, (ptiNnzIndex)tsr->nnz, sb_bits, sk_bits, sc_bits);
674 pti_CheckError(result, "HiSpTns Convert", NULL);
675
676 /* Pre-process tensor to get hitsr->kptr, values are nonzero locations. */
677 ptiTimer sort_timer;
678 ptiNewTimer(&sort_timer, 0);
679 ptiStartTimer(sort_timer);
680
681 ptiPreprocessSparseTensor(&hitsr->kptr, hitsr->kschr, hitsr->nkiters, hitsr->kschr_balanced, hitsr->kschr_balanced_pos, hitsr->nkpars, hitsr->kschr_rest, &hitsr->knnzs, tsr, sb_bits, sk_bits, tk);
682
683 ptiStopTimer(sort_timer);
684 ptiPrintElapsedTime(sort_timer, "\tHiCOO sorting (rowblock + morton)");
685 ptiFreeTimer(sort_timer);
686#if PARTI_DEBUG >= 2
687 printf("Kernels: Row-major, blocks: Morton-order sorted:\n");
688 ptiAssert(ptiDumpSparseTensor(tsr, 0, stdout) == 0);
689 printf("hitsr->kptr:\n");
690 ptiDumpNnzIndexVector(&hitsr->kptr, stdout);
691#endif
692
693 ptiTimer gen_timer;
694 ptiNewTimer(&gen_timer, 0);
695 ptiStartTimer(gen_timer);
696
697 /* Temporary storage */
698 ptiIndex * block_begin = (ptiIndex *)malloc(nmodes * sizeof(*block_begin));
699 ptiIndex * block_end = (ptiIndex *)malloc(nmodes * sizeof(*block_end));
700 ptiIndex * block_begin_prior = (ptiIndex *)malloc(nmodes * sizeof(*block_begin_prior));
701 ptiIndex * block_coord = (ptiIndex *)malloc(nmodes * sizeof(*block_coord));
702
703 ptiNnzIndex k_begin, k_end; // #Nonzeros locations
704 ptiNnzIndex nk = 0; // #Kernels
705 ptiNnzIndex nc = 0; // #Chunks
706 ptiNnzIndex nb = 1; // #Blocks // counting from the first nnz
707 ptiNnzIndex nb_tmp = 0;
708 ptiNnzIndex ne = 0; // #Nonzeros per block
709 ptiIndex eindex = 0;
710 ptiBlockIndex chunk_size = 0;
711
712 /* different appending methods:
713 * elements: append every nonzero entry
714 * blocks: append when seeing a new block.
715 * chunks: appending when seeting a new chunk. Notice the boundary of kernels and the last chunk of the whole tensor may be larger than the sc.
716 * kernels: append when seeing a new kernel. Not appending a vector, just write data into an allocated array.
717 */
718 /* Process first nnz */
719 for(ptiIndex m=0; m<nmodes; ++m)
720 block_coord[m] = tsr->inds[m].data[0]; // first nonzero indices
721 result = ptiLocateBeginCoord(block_begin_prior, tsr, block_coord, sb_bits);
722 pti_CheckError(result, "HiSpTns Convert", NULL);
723 for(ptiIndex m=0; m<nmodes; ++m)
724 ptiAppendBlockIndexVector(&hitsr->binds[m], (ptiBlockIndex)block_begin_prior[m]);
725 ptiAppendNnzIndexVector(&hitsr->bptr, 0);
726
727
728 /* Loop for all kernels, 0 - hitsr->kptr.len - 1 for OMP code */
729 for(ptiNnzIndex k=0; k<hitsr->kptr.len - 1; ++k) {
730 k_begin = hitsr->kptr.data[k];
731 k_end = hitsr->kptr.data[k+1]; // exclusive
732 nb_tmp = k == 0 ? 0: nb;
733 /* Modify kptr pointing to block locations */
734 hitsr->kptr.data[k] = nb_tmp;
735 ++ nk;
736
737 /* Only append a chunk for the new kernel, the last chunk in the old kernel may be larger than sc */
738 ptiAppendNnzIndexVector(&hitsr->cptr, nb_tmp);
739 // printf("cptr 1:\n");
740 // ptiDumpNnzIndexVector(&hitsr->cptr, stdout);
741 ++ nc;
742 chunk_size = 0;
743
744 /* Loop nonzeros in each kernel */
745 for(ptiNnzIndex z = k_begin; z < k_end; ++z) {
746 #if PARTI_DEBUG == 5
747 printf("z: %"HIPARTI_PRI_NNZ_INDEX "\n", z);
748 #endif
749
750 for(ptiIndex m=0; m<nmodes; ++m)
751 block_coord[m] = tsr->inds[m].data[z]; // first nonzero indices
752 #if PARTI_DEBUG == 5
753 printf("block_coord:\n");
754 ptiAssert(ptiDumpIndexArray(block_coord, nmodes, stdout) == 0);
755 #endif
756
757 result = ptiLocateBeginCoord(block_begin, tsr, block_coord, sb_bits);
758 // pti_CheckError(result, "HiSpTns Convert", NULL);
759 #if PARTI_DEBUG == 5
760 printf("block_begin_prior:\n");
761 ptiAssert(ptiDumpIndexArray(block_begin_prior, nmodes, stdout) == 0);
762 printf("block_begin:\n");
763 ptiAssert(ptiDumpIndexArray(block_begin, nmodes, stdout) == 0);
764 #endif
765
766 result = ptiBlockEnd(block_end, tsr, block_begin, sb); // exclusive
767 // pti_CheckError(result, "HiSpTns Convert", NULL);
768
769 /* Append einds and values */
770 for(ptiIndex m=0; m<nmodes; ++m) {
771 eindex = tsr->inds[m].data[z] < (block_begin[m] << sb_bits) ? tsr->inds[m].data[z] : tsr->inds[m].data[z] - (block_begin[m] << sb_bits);
772 ptiAssert(eindex < sb);
773 ptiAppendElementIndexVector(&hitsr->einds[m], (ptiElementIndex)eindex);
774 }
775 ptiAppendValueVector(&hitsr->values, tsr->values.data[z]);
776
777
778 /* z in the same block with last z */
779 if (ptiEqualWithTwoCoordinates(block_begin, block_begin_prior, nmodes) == 1)
780 {
781 /* ne: #Elements in current block */
782 ++ ne;
783 } else { /* New block */
784 /* ne: #Elements in the last block */
785 /* Append block bptr and bidx */
786 ptiAppendNnzIndexVector(&hitsr->bptr, (ptiBlockIndex)z);
787 for(ptiIndex m=0; m<nmodes; ++m)
788 ptiAppendBlockIndexVector(&hitsr->binds[m], (ptiBlockIndex)block_begin[m]);
789 for(ptiIndex m=0; m<nmodes; ++m)
790 block_begin_prior[m] = block_begin[m];
791
792 /* ne: old block's number of nonzeros */
793 // if(chunk_size + ne > sc || ne >= sc) {
794 // if(chunk_size + ne >= sc && chunk_size > 0) { // calculate the prior block
795 // /* Append a chunk ending by the old block */
796 // ptiAppendNnzIndexVector(&hitsr->cptr, nb-1);
797 // // printf("cptr 2:\n");
798 // // ptiDumpNnzIndexVector(&hitsr->cptr, stdout);
799 // ++ nc;
800 // chunk_size = ne;
801 // } else {
802 // chunk_size += ne;
803 // }
804
805 if(chunk_size + ne >= sc) { // calculate the prior block
806 /* Append a chunk ending by the old block */
807 ptiAppendNnzIndexVector(&hitsr->cptr, nb);
808 // printf("cptr 2:\n");
809 // ptiDumpNnzIndexVector(&hitsr->cptr, stdout);
810 // printf("nb: %u, chunk_size: %u, ne: %u\n", nb, chunk_size, ne);
811 ++ nc;
812 chunk_size = 0;
813 } else {
814 chunk_size += ne;
815 }
816
817 ++ nb;
818 ne = 1;
819 } // End new block
820 #if PARTI_DEBUG == 5
821 printf("nk: %u, nc: %u, nb: %u, ne: %u, chunk_size: %lu\n\n", nk, nc, nb, ne, chunk_size);
822 #endif
823
824 } // End z loop
825
826 } // End k loop
827 ptiAssert(nb <= nnz);
828 ptiAssert(nb == hitsr->binds[0].len);
829 // ptiAssert(nc <= nb);
830 ptiAssert(nk == hitsr->kptr.len - 1);
831
832 /* Last element for kptr, cptr, bptr */
833 hitsr->kptr.data[hitsr->kptr.len - 1] = hitsr->bptr.len;
834 ptiAppendNnzIndexVector(&hitsr->cptr, hitsr->bptr.len);
835 ptiAppendNnzIndexVector(&hitsr->bptr, nnz);
836
837
838 *max_nnzb = hitsr->bptr.data[1] - hitsr->bptr.data[0];
839 ptiNnzIndex sum_nnzb = 0;
840 for(ptiIndex i=0; i < hitsr->bptr.len - 1; ++i) {
841 ptiNnzIndex nnzb = hitsr->bptr.data[i+1] - hitsr->bptr.data[i];
842 sum_nnzb += nnzb;
843 if(*max_nnzb < nnzb) {
844 *max_nnzb = nnzb;
845 }
846 }
847 ptiAssert(sum_nnzb == hitsr->nnz);
848
849 ptiStopTimer(gen_timer);
850 ptiPrintElapsedTime(gen_timer, "\tGenerate HiCOO");
851 ptiFreeTimer(gen_timer);
852
853
854 free(block_begin);
855 free(block_end);
856 free(block_begin_prior);
857 free(block_coord);
858
859 return 0;
860}
