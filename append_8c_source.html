43int pti_SemiSparseTensorAppend(ptiSemiSparseTensor *tsr, const ptiIndex indices[], ptiValue value) {
44 int result;
45 int need_resize = 0;
46 if(tsr->nnz == 0) {
47 need_resize = 1;
48 } else if(pti_CompareIndices(tsr, tsr->nnz-1, indices) != 0) {
49 need_resize = 1;
50 }
51 if(need_resize) {
52 ptiIndex i;
53 for(i = 0; i < tsr->nmodes; ++i) {
54 if(i != tsr->mode) {
55 result = ptiAppendIndexVector(&tsr->inds[i], indices[i]);
56 pti_CheckError(result, "SspTns Append", NULL);
57 }
58 }
59 result = ptiAppendMatrix(&tsr->values, NULL);
60 pti_CheckError(result, "SspTns Append", NULL);
61 memset(&tsr->values.values[tsr->nnz * tsr->stride], 0, tsr->nmodes * sizeof (ptiValue));
62 ++tsr->nnz;
63 }
64 tsr->values.values[(tsr->nnz-1) * tsr->stride + indices[tsr->mode]] = value;
65 return 0;
