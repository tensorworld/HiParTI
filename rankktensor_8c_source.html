23int ptiNewRankKruskalTensor(ptiRankKruskalTensor *ktsr, ptiIndex nmodes, const ptiIndex ndims[], ptiElementIndex rank)
24{
25 ktsr->nmodes = nmodes;
26 ktsr->rank = rank;
27 ktsr->ndims = (ptiIndex*)malloc(nmodes*sizeof(ptiIndex));
28 for(ptiIndex i=0; i<nmodes; ++i)
29 ktsr->ndims[i] = ndims[i];
30 ktsr->lambda = (ptiValue*)malloc(rank*sizeof(ptiValue));
31 ktsr->fit = 0.0;
32
33 return 0;
34}

43void ptiRankKruskalTensorInverseShuffleIndices(ptiRankKruskalTensor * ktsr, ptiIndex ** map_inds) {
44 /* Renumber factor matrices rows */
45 ptiIndex new_i;
46 for(ptiIndex m=0; m < ktsr->nmodes; ++m) {
47 ptiRankMatrix * mtx = ktsr->factors[m];
48 ptiIndex * mode_map_inds = map_inds[m];
49 ptiValue * tmp_values = malloc(mtx->cap * mtx->stride * sizeof (ptiValue));
50
51 for(ptiIndex i=0; i<mtx->nrows; ++i) {
52 new_i = mode_map_inds[i];
53 for(ptiElementIndex j=0; j<mtx->ncols; ++j) {
54 tmp_values[i * mtx->stride + j] = mtx->values[new_i * mtx->stride + j];
55 }
56 }
57 free(mtx->values);
58 mtx->values = tmp_values;
59 }
60}

76double KruskalTensorFitHiCOO(
77 ptiSparseTensorHiCOO const * const hitsr,
78 ptiValue const * const __restrict lambda,
79 ptiRankMatrix ** mats,
80 ptiRankMatrix ** ata)
81{
82 ptiIndex const nmodes = hitsr->nmodes;
83
84 double ptien_normsq = SparseTensorFrobeniusNormSquaredHiCOO(hitsr);
85 // printf("ptien_normsq: %lf\n", ptien_normsq);
86 double const norm_mats = KruskalTensorFrobeniusNormSquaredRank(nmodes, lambda, ata);
87 // printf("norm_mats: %lf\n", norm_mats);
88 double const inner = SparseKruskalTensorInnerProductRank(nmodes, lambda, mats);
89 // printf("inner: %lf\n", inner);
90 double residual = ptien_normsq + norm_mats - 2 * inner;
91 if (residual > 0.0) {
92 residual = sqrt(residual);
93 } else {
94 residual = 0.0;
95 }
96 // printf("residual: %lf\n", residual);
97 double fit = 1 - (residual / sqrt(ptien_normsq));
98
99 return fit;
100}

103// Column-major.
104/* Compute a Kruskal tensor's norm is compute on "ata"s. Check Tammy's sparse */
105double KruskalTensorFrobeniusNormSquaredRank(
106 ptiIndex const nmodes,
107 ptiValue const * const __restrict lambda,
108 ptiRankMatrix ** ata) // ata: column-major
109{
110 ptiElementIndex const rank = ata[0]->ncols;
111 ptiElementIndex const stride = ata[0]->stride;
112 ptiValue * const __restrict tmp_atavals = ata[nmodes]->values; // Column-major
113 double norm_mats = 0;
114
115#ifdef HIPARTI_USE_OPENMP
116 #pragma omp parallel for schedule(static)
117#endif
118 for(ptiIndex x=0; x < rank*stride; ++x) {
119 tmp_atavals[x] = 1.;
120 }
121 // printf("KruskalTensorFrobeniusNormSquaredRank: \n");
122 // ptiDumpRankMatrix(ata[nmodes], stdout);
123
124 /* Compute Hadamard product for all "ata"s */
125 for(ptiIndex m=0; m < nmodes; ++m) {
126 ptiValue const * const __restrict atavals = ata[m]->values;
127#ifdef HIPARTI_USE_OPENMP
128 #pragma omp parallel for schedule(static)
129#endif
130 for(ptiElementIndex i=0; i < rank; ++i) {
131 for(ptiElementIndex j=i; j < rank; ++j) {
132 tmp_atavals[i * stride + j] *= atavals[i * stride + j];
133 }
134 }
135 }
136 // printf("KruskalTensorFrobeniusNormSquaredRank: \n");
137 // ptiDumpRankMatrix(ata[nmodes], stdout);
138
139 /* compute lambda^T * aTa[MAX_NMODES] * lambda, only compute a half of them because of its symmetric */
140// #ifdef HIPARTI_USE_OPENMP
141// #pragma omp parallel for schedule(static) reduction(+:norm_mats)
142// #endif
143 for(ptiElementIndex i=0; i < rank; ++i) {
144 norm_mats += tmp_atavals[i+(i*stride)] * lambda[i] * lambda[i];
145 for(ptiElementIndex j=i+1; j < rank; ++j) {
146 norm_mats += tmp_atavals[j+(i*stride)] * lambda[i] * lambda[j] * 2;
147 }
148 // printf("inter norm_mats: %lf\n", norm_mats);
149 }
150
151 return fabs(norm_mats);
152}

156// Row-major, compute via MTTKRP result (mats[nmodes]) and mats[nmodes-1].
157double SparseKruskalTensorInnerProductRank(
158 ptiIndex const nmodes,
159 ptiValue const * const __restrict lambda,
160 ptiRankMatrix ** mats)
161{
162 ptiElementIndex const rank = mats[0]->ncols;
163 ptiElementIndex const stride = mats[0]->stride;
164 ptiIndex const last_mode = nmodes - 1;
165 ptiIndex const I = mats[last_mode]->nrows;
166
167 // printf("mats[nmodes-1]:\n");
168 // ptiDumpMatrix(mats[nmodes-1], stdout);
169 // printf("mats[nmodes]:\n");
170 // ptiDumpMatrix(mats[nmodes], stdout);
171
172 ptiValue const * const last_vals = mats[last_mode]->values;
173 ptiValue const * const tmp_vals = mats[nmodes]->values;
174 ptiValue * buffer_accum;
175
176 double inner = 0;
177
178 double * const __restrict accum = (double *) malloc(rank*sizeof(*accum));
179
180#ifdef HIPARTI_USE_OPENMP
181 #pragma omp parallel for schedule(static)
182#endif
183 for(ptiElementIndex r=0; r < rank; ++r) {
184 accum[r] = 0.0;
185 }
186
187#ifdef HIPARTI_USE_OPENMP
188 #pragma omp parallel
189 {
190 int const nthreads = omp_get_num_threads();
191 #pragma omp master
192 {
193 buffer_accum = (ptiValue *)malloc(nthreads * rank * sizeof(ptiValue));
194 for(ptiIndex j=0; j < (ptiIndex)nthreads * rank; ++j)
195 buffer_accum[j] = 0.0;
196 }
197 }
198#endif
199
200#ifdef HIPARTI_USE_OPENMP
201 #pragma omp parallel
202 {
203 int const tid = omp_get_thread_num();
204 int const nthreads = omp_get_num_threads();
205 ptiValue * loc_accum = buffer_accum + tid * rank;
206
207 #pragma omp for
208 for(ptiIndex i=0; i < I; ++i) {
209 for(ptiElementIndex r=0; r < rank; ++r) {
210 loc_accum[r] += last_vals[r+(i*stride)] * tmp_vals[r+(i*stride)];
211 }
212 }
213
214 #pragma omp for schedule(static)
215 for(ptiElementIndex j=0; j < rank; ++j) {
216 for(ptiIndex i=0; i < (ptiIndex)nthreads; ++i) {
217 accum[j] += buffer_accum[i*rank + j];
218 }
219 }
220
221 }
222
223#else
224
225 for(ptiIndex i=0; i < I; ++i) {
226 for(ptiElementIndex r=0; r < rank; ++r) {
227 accum[r] += last_vals[r+(i*stride)] * tmp_vals[r+(i*stride)];
228 }
229 }
230
231#endif
232
233#ifdef HIPARTI_USE_OPENMP
234 #pragma omp parallel for schedule(static) reduction(+:inner)
235#endif
236 for(ptiElementIndex r=0; r < rank; ++r) {
237 inner += accum[r] * lambda[r];
238 }
239
240#ifdef HIPARTI_USE_OPENMP
241 free(buffer_accum);
242#endif
243
244 return inner;
245}
