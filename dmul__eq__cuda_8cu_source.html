41int ptiCudaSparseTensorDotMulEq(ptiSparseTensor *Z, const ptiSparseTensor *X, const ptiSparseTensor *Y) {
42 int result;
43 /* Ensure X and Y are in same shape */
44 if(Y->nmodes != X->nmodes) {
45 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CUDA SpTns DotMul", "shape mismatch");
46 }
47 for(ptiIndex i = 0; i < X->nmodes; ++i) {
48 if(Y->ndims[i] != X->ndims[i]) {
49 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CUDA SpTns DotMul", "shape mismatch");
50 }
51 }
52 /* Ensure X and Y have exactly the same nonzero distribution */
53 if(Y->nnz != X->nnz) {
54 pti_CheckError(PTIERR_SHAPE_MISMATCH, "SpTns DotMul", "nonzero distribution mismatch");
55 }
56 ptiNnzIndex nnz = X->nnz;
57
58 ptiCopySparseTensor(Z, X, 1);
59
60 ptiValue *X_val = NULL;
61 result = cudaMalloc((void **) &X_val, X->nnz * sizeof (ptiValue));
62 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
63 result = cudaMemcpy(X_val, X->values.data, X->nnz * sizeof (ptiValue), cudaMemcpyHostToDevice);
64 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
65
66 ptiValue *Y_val = NULL;
67 result = cudaMalloc((void **) &Y_val, Y->nnz * sizeof (ptiValue));
68 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
69 result = cudaMemcpy(Y_val, Y->values.data, Y->nnz * sizeof (ptiValue), cudaMemcpyHostToDevice);
70 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
71
72 ptiValue *Z_val = NULL;
73 result = cudaMalloc((void **) &Z_val, X->nnz * sizeof (ptiValue));
74 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
75 result = cudaMemset(Z_val, 0, X->nnz * sizeof (ptiValue));
76 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
77
78 ptiNnzIndex nthreads = 128;
79 ptiNnzIndex nblocks = (nnz + nthreads -1)/ nthreads;
80
81 ptiTimer timer;
82 ptiNewTimer(&timer, 0);
83 ptiStartTimer(timer);
84
85 pti_DotMulKernel<<<nblocks, nthreads>>>(nnz, Z_val, X_val, Y_val);
86 result = cudaDeviceSynchronize();
87
88 ptiStopTimer(timer);
89 ptiPrintElapsedTime(timer, "CUDA SpTns DotMul");
90 ptiFreeTimer(timer);
91
92 cudaMemcpy(Z->values.data, Z_val, Z->nnz * sizeof (ptiValue), cudaMemcpyDeviceToHost);
93
94 result = cudaFree(X_val);
95 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
96 result = cudaFree(Y_val);
97 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
98 result = cudaFree(Z_val);
99 pti_CheckCudaError(result != 0, "CUDA SpTns DotMul");
100
101 /* Check whether elements become zero after adding.
102 If so, fill the gap with the [nnz-1]'th element.
103 */
104 pti_SparseTensorCollectZeros(Z);
105 /* Sort the indices */
106 ptiSparseTensorSortIndex(Z, 1, 1);
107
108 return 0;
109}
