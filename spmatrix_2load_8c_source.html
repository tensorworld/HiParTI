25int ptiLoadSparseMatrix(ptiSparseMatrix *mtx, ptiIndex start_index, FILE *fid)
26{
27 MM_typecode matcode;
28 int iores, retval;
29
30 if (mm_read_banner(fid, &matcode) != 0){
31 printf("Could not process Matrix Market banner.\n");
32 exit(1);
33 }
34
35 if (!mm_is_valid(matcode)){
36 printf("Invalid Matrix Market file.\n");
37 exit(1);
38 }
39
40 if (!((mm_is_real(matcode) || mm_is_integer(matcode) || mm_is_pattern(matcode)) && mm_is_coordinate(matcode) && mm_is_sparse(matcode) ) ){
41 printf("Sorry, this application does not support ");
42 printf("Market Market type: [%s]\n", mm_typecode_to_str(matcode));
43 printf("Only sparse real-valued or pattern coordinate matrices are supported\n");
44 exit(1);
45 }
46
47 int num_rows, num_cols, num_nonzeros;
48 if ( mm_read_mtx_crd_size(fid,&num_rows,&num_cols,&num_nonzeros) !=0)
49 exit(1);
50
51 mtx->nrows = (ptiIndex) num_rows;
52 mtx->ncols = (ptiIndex) num_cols;
53 mtx->nnz = (ptiNnzIndex) num_nonzeros;
54
55 retval = ptiNewIndexVector(&mtx->rowind, mtx->nnz, mtx->nnz);
56 pti_CheckOSError(iores < 0, "SpMtx Load");
57 retval = ptiNewIndexVector(&mtx->colind, mtx->nnz, mtx->nnz);
58 pti_CheckOSError(iores < 0, "SpMtx Load");
59 retval = ptiNewValueVector(&mtx->values, mtx->nnz, mtx->nnz);
60 pti_CheckError(retval, "SpMtx Load", NULL);
61
62 char strI[64], strJ[64], strV[64];
63 if (mm_is_pattern(matcode)){
64 // pattern matrix defines sparsity pattern, but not values
65 for( ptiNnzIndex i = 0; i < mtx->nnz; i++ ) {
66 fscanf(fid, "%s %s\n", strI, strJ);
67 mtx->rowind.data[i] = (ptiIndex) (atoi(strI)) - 1;
68 mtx->colind.data[i] = (ptiIndex) (atoi(strJ)) - 1;
69 mtx->values.data[i] = 1.0; //use value 1.0 for all nonzero entries
70 }
71 } else if (mm_is_real(matcode) || mm_is_integer(matcode)){
72 for( ptiNnzIndex i = 0; i < mtx->nnz; i++ ) {
73 fscanf(fid, "%s %s %s\n", strI, strJ, strV);
74 mtx->rowind.data[i] = (ptiIndex) (atoi(strI)) - 1;
75 mtx->colind.data[i] = (ptiIndex) (atoi(strJ)) - 1;
76 mtx->values.data[i] = (ptiValue) (atof(strV));
77 }
78 } else {
79 printf("Unrecognized data type\n");
80 exit(1);
81 }
82
83 if( mm_is_symmetric(matcode) ){ //duplicate off diagonal entries
84 ptiIndex off_diagonals = 0;
85 for( ptiNnzIndex i = 0; i < mtx->nnz; i++ ){
86 if( mtx->rowind.data[i] != mtx->colind.data[i] )
87 off_diagonals++;
88 }
89
90 ptiNnzIndex true_nonzeros = 2*off_diagonals + (mtx->nnz - off_diagonals);
91
92 ptiIndex* new_I = (ptiIndex*)malloc(true_nonzeros * sizeof(ptiIndex));
93 ptiIndex* new_J = (ptiIndex*)malloc(true_nonzeros * sizeof(ptiIndex));
94 ptiValue * new_V = (ptiValue*)malloc(true_nonzeros * sizeof(ptiValue));
95
96 ptiNnzIndex ptr = 0;
97 for( ptiNnzIndex i = 0; i < mtx->nnz; i++ ){
98 if( mtx->rowind.data[i] != mtx->colind.data[i] ){
99 new_I[ptr] = mtx->rowind.data[i]; new_J[ptr] = mtx->colind.data[i]; new_V[ptr] = mtx->values.data[i];
100 ptr++;
101 new_J[ptr] = mtx->rowind.data[i]; new_I[ptr] = mtx->colind.data[i]; new_V[ptr] = mtx->values.data[i];
102 ptr++;
103 } else {
104 new_I[ptr] = mtx->rowind.data[i]; new_J[ptr] = mtx->colind.data[i]; new_V[ptr] = mtx->values.data[i];
105 ptr++;
106 }
107 }
108 free(mtx->rowind.data); free(mtx->colind.data); free(mtx->values.data);
109 mtx->rowind.data = new_I; mtx->colind.data = new_J; mtx->values.data = new_V;
110 mtx->nnz = true_nonzeros;
111 mtx->rowind.len = true_nonzeros;
112 mtx->rowind.cap = true_nonzeros;
113 mtx->colind.len = true_nonzeros;
114 mtx->colind.cap = true_nonzeros;
115 mtx->values.len = true_nonzeros;
116 mtx->values.cap = true_nonzeros;
117 } //end symmetric case
118
119 return 0;
120
121}
