39int ptiMTTKRP(ptiSparseTensor const * const X,
40 ptiMatrix * mats[], // mats[nmodes] as temporary space.
41 ptiIndex const mats_order[], // Correspond to the mode order of X.
42 ptiIndex const mode) {
43
44 ptiIndex const nmodes = X->nmodes;
45
46 if(nmodes == 3) {
47 ptiAssert(ptiMTTKRP_3D(X, mats, mats_order, mode) == 0);
48 return 0;
49 }
50
51 ptiNnzIndex const nnz = X->nnz;
52 ptiIndex const * const ndims = X->ndims;
53 ptiValue const * const restrict vals = X->values.data;
54 ptiIndex const stride = mats[0]->stride;
55 ptiValueVector scratch; // Temporary array
56
57 /* Check the mats. */
58 for(ptiIndex i=0; i<nmodes; ++i) {
59 if(mats[i]->ncols != mats[nmodes]->ncols) {
60 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
61 }
62 if(mats[i]->nrows != ndims[i]) {
63 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
64 }
65 }
66
67 ptiIndex const tmpI = mats[mode]->nrows;
68 ptiIndex const R = mats[mode]->ncols;
69 ptiIndex const * const restrict mode_ind = X->inds[mode].data;
70 ptiMatrix * const restrict M = mats[nmodes];
71 ptiValue * const restrict mvals = M->values;
72 memset(mvals, 0, tmpI*stride*sizeof(ptiValue));
73 ptiNewValueVector(&scratch, R, R);
74 ptiConstantValueVector(&scratch, 0);
75
76
77 for(ptiNnzIndex x=0; x<nnz; ++x) {
78
79 ptiIndex times_mat_index = mats_order[1];
80 ptiMatrix * times_mat = mats[times_mat_index];
81 ptiIndex * times_inds = X->inds[times_mat_index].data;
82 ptiIndex tmp_i = times_inds[x];
83 ptiValue const entry = vals[x];
84 for(ptiIndex r=0; r<R; ++r) {
85 scratch.data[r] = entry * times_mat->values[tmp_i * stride + r];
86 }
87
88 for(ptiIndex i=2; i<nmodes; ++i) {
89 times_mat_index = mats_order[i];
90 times_mat = mats[times_mat_index];
91 times_inds = X->inds[times_mat_index].data;
92 tmp_i = times_inds[x];
93
94 for(ptiIndex r=0; r<R; ++r) {
95 scratch.data[r] *= times_mat->values[tmp_i * stride + r];
96 }
97 }
98
99 ptiIndex const mode_i = mode_ind[x];
100 for(ptiIndex r=0; r<R; ++r) {
101 mvals[mode_i * stride + r] += scratch.data[r];
102 }
103 }
104
105 ptiFreeValueVector(&scratch);
106
107 return 0;
108}

111int ptiMTTKRP_3D(ptiSparseTensor const * const X,
112 ptiMatrix * mats[], // mats[nmodes] as temporary space.
113 ptiIndex const mats_order[], // Correspond to the mode order of X.
114 ptiIndex const mode)
115{
116 ptiIndex const nmodes = X->nmodes;
117 ptiNnzIndex const nnz = X->nnz;
118 ptiIndex const * const ndims = X->ndims;
119 ptiValue const * const restrict vals = X->values.data;
120 ptiIndex const stride = mats[0]->stride;
121
122 /* Check the mats. */
123 ptiAssert(nmodes ==3);
124 for(ptiIndex i=0; i<nmodes; ++i) {
125 if(mats[i]->ncols != mats[nmodes]->ncols) {
126 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
127 }
128 if(mats[i]->nrows != ndims[i]) {
129 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
130 }
131 }
132
133
134 ptiIndex const tmpI = mats[mode]->nrows;
135 ptiIndex const R = mats[mode]->ncols;
136 ptiIndex const * const restrict mode_ind = X->inds[mode].data;
137 ptiMatrix * const restrict M = mats[nmodes];
138 ptiValue * const restrict mvals = M->values;
139 memset(mvals, 0, tmpI*stride*sizeof(ptiValue));
140
141 ptiIndex times_mat_index_1 = mats_order[1];
142 ptiMatrix * restrict times_mat_1 = mats[times_mat_index_1];
143 ptiIndex * restrict times_inds_1 = X->inds[times_mat_index_1].data;
144 ptiIndex times_mat_index_2 = mats_order[2];
145 ptiMatrix * restrict times_mat_2 = mats[times_mat_index_2];
146 ptiIndex * restrict times_inds_2 = X->inds[times_mat_index_2].data;
147
148 ptiIndex mode_i;
149 ptiIndex tmp_i_1, tmp_i_2;
150 ptiValue entry;
151 for(ptiNnzIndex x=0; x<nnz; ++x) {
152 mode_i = mode_ind[x];
153 tmp_i_1 = times_inds_1[x];
154 tmp_i_2 = times_inds_2[x];
155 entry = vals[x];
156
157 for(ptiIndex r=0; r<R; ++r) {
158 mvals[mode_i * stride + r] += entry * times_mat_1->values[tmp_i_1 * stride + r] * times_mat_2->values[tmp_i_2 * stride + r];
159 }
160 }
161
162 return 0;
163}
