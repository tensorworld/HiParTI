39void pti_ComplainError(const char *module, int errcode, const char *file, unsigned line, const char *reason) {
40 g_last_error.errcode = errcode;
41 g_last_error.module = module;
42 g_last_error.file = file;
43 g_last_error.line = line;
44 free(g_last_error.reason);
45 if(reason) {
46 size_t len = strlen(reason);
47 g_last_error.reason = malloc(len+1);
48 if(!g_last_error.reason) {
49 abort();
50 }
51 memcpy(g_last_error.reason, reason, len+1);
52 }
53 if(g_last_error.reason && g_last_error.reason[0] != '\0') {
54 fprintf(stderr, "[%s] error 0x%08x at %s:%u, %s\n",
55 g_last_error.module,
56 g_last_error.errcode,
57 g_last_error.file,
58 g_last_error.line,
59 g_last_error.reason
60 );
61 } else {
62 fprintf(stderr, "[%s] error 0x%08x at %s:%u\n",
63 g_last_error.module,
64 g_last_error.errcode,
65 g_last_error.file,
66 g_last_error.line
67 );
68 }
69}
