124 newIndices is of size [nmodes][ndims[modes]] and assumed to be allocted.
125 It will be overwritten. No need to initialize.
126
127 We will need to reshuffle nonzeros. In order to not to touch tsr, we copy the indices of nonzeros

129 */

141 for (z = 0; z < nnz; z++)
142 {
143 coords[z] = (ptiIndex *) malloc(sizeof(ptiIndex) * nm);
144 for (m = 0; m < nm; m++) {
145 coords[z][m] = tsr->inds[m].data[z];
146 }
147 }
148
149 ptiStopTimer(copy_coord_timer);
150 ptiPrintElapsedTime(copy_coord_timer, "Copy coordinate time");
151 ptiFreeTimer(copy_coord_timer);
152
153 /* checkEmptySlices(coords, nnz, nm, tsr->ndims); */
154
155 if (renumber == 1) { /* Lexi-order renumbering */
156
157 ptiIndex ** orgIds = (ptiIndex **) malloc(sizeof(ptiIndex*) * nm);
158

172 orderDim(coords, nnz, nm, tsr->ndims, m, orgIds);

261{
262 /*
263 to sort the nonzeros first on i_1+i_2+...+i_4, if ties then on
264 i_1+i_2+...+3, if ties then on i_1+i_2, if ties then on i_1 only.
265 We do not include dim in the comparisons.
266
267 */
268 ptiIndex m;
269 ptiIndex v1 = 0, v2 = 0;
270
271 for (m = 0; m < nm; m++)
272 {
273 if(m != dim)
274 {
275 v1 += z1[m];
276 v2 += z2[m];
277 }
278 }
279 if(v1 < v2) return -1;
280 else if(v1 > v2) return 1;
281 else{
282 for (m = 0; m < nm; m++)
283 {
284 if(m != dim)
285 {
286 v1 -= z1[m];
287 v2 -= z2[m];
288 if (v1 < v2) return -1;
289 else if (v1 > v2) return 1;
290 }
291 }
292 }
293 return 0; /*are equal*/

352static inline ptiNnzIndex buPartition(ptiIndex **coords, ptiNnzIndex lo, ptiNnzIndex hi, ptiIndex nm, ptiIndex *ndims, ptiIndex dim, ptiIndex *tmpNnz, ptiIndex *wspace)
353{
354 /* copied from the web http://ndevilla.free.fr/median/median/src/quickselect.c */
355 ptiNnzIndex low, high, median, middle, ll, hh;
356
357 low = lo; high = hi; median = (low+high)/2;
358 for(;;)
359 {
360 if (high<=low) return median;
361 if(high == low + 1)
362 {
363 if(isLessThanOrEqualTo(coords[low], coords[high], nm, ndims, dim)== 1)
364 buSwap (coords[high], coords[low], nm, wspace);
365 return median;
366 }
367 middle = (low+high)/2;
368 if(isLessThanOrEqualTo(coords[middle], coords[high], nm, ndims, dim) == 1)
369 buSwap (coords[middle], coords[high], nm, wspace);
370
371 if(isLessThanOrEqualTo(coords[low], coords[high], nm, ndims, dim) == 1)
372 buSwap (coords[low], coords[high], nm, wspace);
373
374 if(isLessThanOrEqualTo(coords[middle], coords[low], nm, ndims, dim) == 1)
375 buSwap (coords[low], coords[middle], nm, wspace);
376
377 buSwap (coords[middle], coords[low+1], nm, wspace);
378
379 ll = low + 1;
380 hh = high;
381 for (;;){
382 do ll++; while (isLessThanOrEqualTo(coords[low], coords[ll], nm, ndims, dim) == 1);
383 do hh--; while (isLessThanOrEqualTo(coords[hh], coords[low], nm, ndims, dim) == 1);
384
385 if (hh < ll) break;
386
387 buSwap (coords[ll], coords[hh], nm, wspace);
388 }
389 buSwap (coords[low], coords[hh], nm,wspace);
390 if (hh <= median) low = ll;
391 if (hh >= median) high = hh - 1;
392 }
393
394}

441static void mySort(ptiIndex ** coords, ptiNnzIndex last, ptiIndex nm, ptiIndex * ndims, ptiIndex dim)
442{
443 /* sorts coords accourding to all dims except dim, where items are refereed with newIndices*/
444 /* an iterative quicksort */
445 ptiNnzIndex *stack, top, lo, hi, pv;
446 ptiIndex *tmpNnz, *wspace;
447
448 tmpNnz = (ptiIndex*) malloc(sizeof(ptiIndex) * nm);
449 wspace = (ptiIndex*) malloc(sizeof(ptiIndex) * nm);
450 stack = (ptiNnzIndex *) malloc(sizeof(ptiNnzIndex) * 2 * (last+2));
451
452 if(stack == NULL) {
453 printf("could not allocated stack. returning\n");
454 exit(14);
455 }
456 top = 0;
457 stack[top++] = 0;
458 stack[top++] = last;
459 while (top>=2)
460 {
461 hi = stack[--top];
462 lo = stack[--top];
463 pv = buPartition(coords, lo, hi, nm, ndims, dim, tmpNnz, wspace);
464
465 if(pv > lo+1)
466 {
467 if(pv - lo > 128)
468 {
469 stack[top++] = lo;
470 stack[top++] = pv-1 ;
471 }
472 else
473 insertionSort(coords, lo, pv-1, nm, ndims, dim, tmpNnz, wspace);
474 }
475 if(top >= 2 * (last+2)){
476 printf("\thow come this tight?\n");
477 exit(13);
478 }
479 if(pv + 1 < hi)
480 {
481 if(hi - pv > 128)
482 {
483 stack[top++] = pv + 1 ;
484 stack[top++] = hi;
485 }
486 else
487 insertionSort(coords, pv+1, hi, nm, ndims, dim, tmpNnz, wspace);
488 }
489 if( top >= 2 * (last+2)) {
490 printf("\thow come this tight?\n");
491 exit(13);
492 }
493 }
494 free(stack);
495 free(wspace);
496 free(tmpNnz);
497}

969{
970 ptiIndex largest, j, l,r, tmp;
971
972 largest = j = i;
973 while(j<=sz/2)
974 {
975 l = 2*j;
976 r = 2*j + 1;
977

979 (key[heapIds[l]] == key[heapIds[j]] && SIZEV(heapIds[l]) < SIZEV(heapIds[j]) )
980 )
981 largest = l;
982 else
983 largest = j;
984
985 if (r<=sz && (key[heapIds[r]]>key[heapIds[largest]] ||
986 (key[heapIds[r]]==key[heapIds[largest]] && SIZEV(heapIds[r]) < SIZEV(heapIds[largest])))

993 heapIds[largest] = heapIds[j];
