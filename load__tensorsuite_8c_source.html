1/*
2 This file is part of HiParTI!.
3
4 Reader for the TensorSuite tensor format (https://tensorworld.github.io/TensorSuite/).
5
6 A TensorSuite text tensor differs from HiParTI's native .tns in three ways:
7
8 native .tns TensorSuite .tns
9 ----------- ----------------
10 <nmodes> %%TensorSuite-TNS <- banner
11 <dim1> .. <dimN> % version: 0.1 <- '%' comment lines
12 i1 .. iN val % name: <tensor-name>
13 <order> <dim1> .. <dimN> <nnz> <- ONE size line, includes nnz
14 i1 .. iN [val] <- the value column is OPTIONAL
15
16 Two further properties are carried in the sidecar "<stem>_metadata.json":
17 * "index_base" : 0 or 1 (both occur in the collection)
18 * "values_provided" : false for many tensors, which store only the pattern
19 * "sparsity_type" : "element" or "block"
20 The sidecar is read when present; otherwise both are inferred from the data.
21
22 When no values are stored, they are synthesised according to `fill`.
23*/
24

35/* splitmix64, identical to ptiRandomizeMatrix, so synthesised values are reproducible */
36static ptiValue pti_TsNextRandom(uint64_t * state)
37{
38 uint64_t z = (*state += 0x9E3779B97F4A7C15ULL);
39 z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
40 z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
41 z = z ^ (z >> 31);
42 return (ptiValue)((double)(z >> 11) / 9007199254740992.0);
43}

49int pti_TensorSuiteSniff(FILE * fp)
50{
51 long const pos = ftell(fp);
52 char head[32];
53 size_t const n = fread(head, 1, sizeof(head) - 1, fp);
54 head[n] = '\0';
55 if(fseek(fp, pos, SEEK_SET) != 0) return 0;
56 return strncmp(head, PTI_TS_BANNER, strlen(PTI_TS_BANNER)) == 0;
57}

59/* Pull one scalar field out of the sidecar JSON. Returns 0 when found. */
60static int pti_TsMetaField(char const * fname, char const * key, char * out, size_t outlen)
61{
62 char path[4096];
63 char const * dot = strrchr(fname, '.');
64 size_t const stem = dot ? (size_t)(dot - fname) : strlen(fname);
65 if(stem + 32 > sizeof(path)) return -1;
66 memcpy(path, fname, stem);
67 snprintf(path + stem, sizeof(path) - stem, "_metadata.json");
68
69 FILE * mf = fopen(path, "r");
70 if(!mf) return -1;
71
72 int found = -1;
73 char line[1024];
74 char needle[128];
75 snprintf(needle, sizeof(needle), "\"%s\"", key);
76 while(fgets(line, sizeof(line), mf)) {
77 char * p = strstr(line, needle);
78 if(!p) continue;
79 p = strchr(p + strlen(needle), ':');
80 if(!p) continue;
81 ++p;
82 while(*p && (isspace((unsigned char)*p) || *p == '"')) ++p;
83 size_t k = 0;
84 while(*p && *p != ',' && *p != '"' && *p != '\n' && k + 1 < outlen) out[k++] = *p++;
85 while(k > 0 && isspace((unsigned char)out[k-1])) --k;
86 out[k] = '\0';
87 found = 0;
88 break;
89 }
90 fclose(mf);
91 return found;
92}

94/* Next line that is neither blank nor a '%' comment. Returns 0 on success. */
95static int pti_TsNextDataLine(FILE * fp, char * buf, size_t buflen)
96{
97 while(fgets(buf, (int)buflen, fp)) {
98 char * p = buf;
99 while(*p && isspace((unsigned char)*p)) ++p;
100 if(*p == '\0' || *p == '%' || *p == '#') continue;
101 return 0;
102 }
103 return -1;
104}

106static size_t pti_TsCountTokens(char const * s)
107{
108 size_t n = 0;
109 while(*s) {
110 while(*s && isspace((unsigned char)*s)) ++s;
111 if(!*s) break;
112 ++n;
113 while(*s && !isspace((unsigned char)*s)) ++s;
114 }
115 return n;
116}

127int ptiLoadSparseTensorTensorSuite(
128 ptiSparseTensor *tsr,
129 ptiIndex start_index,
130 char const * const fname,
131 ptiTensorSuiteFill fill)
132{
133 FILE * fp = fopen(fname, "r");
134 if(!fp) {
135 fprintf(stderr, "[TensorSuite] Error: cannot open %s\n", fname);
136 return -1;
137 }
138 if(!pti_TensorSuiteSniff(fp)) {
139 fprintf(stderr, "[TensorSuite] Error: %s does not start with \"%s\"\n", fname, PTI_TS_BANNER);
140 fclose(fp);
141 return -1;
142 }
143
144 char meta[128];
145 /* Block-sparse files carry a block-grid line the COO reader cannot interpret. */
146 if(pti_TsMetaField(fname, "sparsity_type", meta, sizeof(meta)) == 0 &&
147 strcmp(meta, "block") == 0) {
148 fprintf(stderr, "[TensorSuite] Error: %s is a block-sparse tensor; "
149 "only element-wise tensors can be read into COO.\n", fname);
150 fclose(fp);
151 return -1;
152 }
153
154 char line[8192];
155 if(pti_TsNextDataLine(fp, line, sizeof(line)) != 0) {
156 fprintf(stderr, "[TensorSuite] Error: %s has no size line\n", fname);
157 fclose(fp);
158 return -1;
159 }
160
161 /* size line: <order> <dim1> ... <dimN> <nnz> */
162 size_t const ntok = pti_TsCountTokens(line);
163 char * save = NULL;
164 char * tok = strtok_r(line, " \t\n\r", &save);
165 ptiIndex const nmodes = (ptiIndex) strtoul(tok, NULL, 10);
166 if(nmodes == 0 || ntok != (size_t) nmodes + 2) {
167 fprintf(stderr, "[TensorSuite] Error: malformed size line in %s "
168 "(order=%u but %zu fields; expected %u)\n",
169 fname, nmodes, ntok, nmodes + 2);
170 fclose(fp);
171 return -1;
172 }
173
174 ptiIndex * ndims = (ptiIndex *) malloc(nmodes * sizeof *ndims);
175 if(!ndims) { fclose(fp); return -1; }
176 for(ptiIndex m = 0; m < nmodes; ++m) {
177 tok = strtok_r(NULL, " \t\n\r", &save);
178 ndims[m] = (ptiIndex) strtoul(tok, NULL, 10);
179 }
180 tok = strtok_r(NULL, " \t\n\r", &save);
181 ptiNnzIndex const nnz_declared = (ptiNnzIndex) strtoull(tok, NULL, 10);
182
183 /* index base: metadata wins, else inferred from the data below */
184 int file_base = -1;
185 if(pti_TsMetaField(fname, "index_base", meta, sizeof(meta)) == 0) {
186 file_base = atoi(meta);
187 }
188
189 /* Does the first data row carry a value column? */
190 long const first_row = ftell(fp);
191 if(pti_TsNextDataLine(fp, line, sizeof(line)) != 0) {
192 fprintf(stderr, "[TensorSuite] Error: %s declares %" HIPARTI_PRI_NNZ_INDEX
193 " nonzeros but has no data rows\n", fname, nnz_declared);
194 free(ndims); fclose(fp);
195 return -1;
196 }
197 size_t const row_tok = pti_TsCountTokens(line);
198 int has_values;
199 if(row_tok == (size_t) nmodes) has_values = 0;
200 else if(row_tok == (size_t) nmodes + 1) has_values = 1;
201 else {
202 fprintf(stderr, "[TensorSuite] Error: %s data row has %zu fields, expected %u or %u\n",
203 fname, row_tok, nmodes, nmodes + 1);
204 free(ndims); fclose(fp);
205 return -1;
206 }
207 fseek(fp, first_row, SEEK_SET);
208
209 int result = ptiNewSparseTensor(tsr, nmodes, ndims);
210 free(ndims);
211 if(result != 0) { fclose(fp); return result; }
212
213 uint64_t rng = HIPARTI_RANDOM_SEED;
214 ptiNnzIndex nread = 0;
215 ptiIndex * idx = (ptiIndex *) malloc(nmodes * sizeof *idx);
216 if(!idx) { fclose(fp); return -1; }
217 int saw_zero_index = 0;
218
219 while(pti_TsNextDataLine(fp, line, sizeof(line)) == 0) {
220 save = NULL;
221 tok = strtok_r(line, " \t\n\r", &save);
222 int ok = 1;
223 for(ptiIndex m = 0; m < nmodes; ++m) {
224 if(!tok) { ok = 0; break; }
225 unsigned long long const v = strtoull(tok, NULL, 10);
226 if(v == 0) saw_zero_index = 1;
227 idx[m] = (ptiIndex) v;
228 tok = strtok_r(NULL, " \t\n\r", &save);
229 }
230 if(!ok) continue;
231
232 ptiValue val;
233 if(has_values) {
234 val = tok ? (ptiValue) atof(tok) : (ptiValue) 0;
235 } else {
236 val = (fill == PTI_TS_FILL_RANDOM) ? pti_TsNextRandom(&rng) : (ptiValue) 1;
237 }
238
239 for(ptiIndex m = 0; m < nmodes; ++m) {
240 result = ptiAppendIndexVector(&tsr->inds[m], idx[m]);
241 if(result != 0) { free(idx); fclose(fp); return result; }
242 }
243 result = ptiAppendValueVector(&tsr->values, val);
244 if(result != 0) { free(idx); fclose(fp); return result; }
245 ++nread;
246 }
247 free(idx);
248 fclose(fp);
249 tsr->nnz = nread;
250
251 if(file_base < 0) file_base = saw_zero_index ? 0 : 1;
252
253 /* HiParTI stores indices 0-based internally (the native reader does
254 `index - start_index`). Unlike the native path we know the file's real
255 base from the metadata, so use that rather than the caller's guess. */
256 (void) start_index;
257 if(file_base != 0) {
258 for(ptiIndex m = 0; m < nmodes; ++m) {
259 for(ptiNnzIndex z = 0; z < nread; ++z) {
260 tsr->inds[m].data[z] -= (ptiIndex) file_base;
261 }
262 }
263 }
264
265 if(nnz_declared != nread) {
266 fprintf(stderr, "[TensorSuite] Warning: %s declares %" HIPARTI_PRI_NNZ_INDEX
267 " nonzeros but %" HIPARTI_PRI_NNZ_INDEX " were read\n",
268 fname, nnz_declared, nread);
269 }
270 return 0;
271}
