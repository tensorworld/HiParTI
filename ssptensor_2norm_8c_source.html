23double pti_SemiSparseTensorNorm(const ptiSemiSparseTensor *X) {
24 double sqnorm = 0;
25 ptiNnzIndex i;
26 for(i = 0; i < X->nnz; ++i) {
27 ptiIndex j;
28 for(j = 0; j < X->values.ncols; ++j) {
29 double cell_value = X->values.values[i * X->stride + j];
30 sqnorm += cell_value * cell_value;
31 }
32 }
33 return sqrt(sqnorm);
34}
