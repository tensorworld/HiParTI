150void pti_SwapValues(ptiSparseTensor *tsr, ptiNnzIndex ind1, ptiNnzIndex ind2) {
151
152 for(ptiIndex i = 0; i < tsr->nmodes; ++i) {
153 ptiIndex eleind1 = tsr->inds[i].data[ind1];
154 tsr->inds[i].data[ind1] = tsr->inds[i].data[ind2];
155 tsr->inds[i].data[ind2] = eleind1;
156 }
157 ptiValue val1 = tsr->values.data[ind1];
158 tsr->values.data[ind1] = tsr->values.data[ind2];
159 tsr->values.data[ind2] = val1;
160}

175void ptiGetBestModeOrder(
176 ptiIndex * mode_order,
177 ptiIndex const mode,
178 ptiIndex const * ndims,
179 ptiIndex const nmodes)
180{
181 ptiKeyValuePair * sorted_ndims = (ptiKeyValuePair*)malloc(nmodes * sizeof(*sorted_ndims));
182 for(ptiIndex m=0; m<nmodes; ++m) {
183 sorted_ndims[m].key = m;
184 sorted_ndims[m].value = ndims[m];
185 }
186
187 /* Increasingly sort */
188 ptiPairArraySort(sorted_ndims, nmodes);
189
190 for(ptiIndex m=0; m<nmodes; ++m) {
191 mode_order[m] = sorted_ndims[m].key;
192 }
193 /* Find the location of mode */
194 ptiIndex mode_loc = 0;
195 for(ptiIndex m=0; m<nmodes; ++m) {
196 if(mode_order[m] == mode) {
197 mode_loc = m;
198 }
199 }
200 /* Shift mode to moder_order[0] */
201 if(mode_loc != 0) {
202 for(ptiIndex m=mode_loc; m>=1; --m) {
203 mode_order[m] = mode_order[m-1];
204 }
205 mode_order[0] = mode;
206 }
207
208 free(sorted_ndims);
209}

221void ptiGetWorstModeOrder(
222 ptiIndex * mode_order,
223 ptiIndex const mode,
224 ptiIndex const * ndims,
225 ptiIndex const nmodes)
226{
227 ptiKeyValuePair * sorted_ndims = (ptiKeyValuePair*)malloc(nmodes * sizeof(*sorted_ndims));
228 for(ptiIndex m=0; m<nmodes; ++m) {
229 sorted_ndims[m].key = m;
230 sorted_ndims[m].value = ndims[m];
231 }
232
233 /* Increasingly sort */
234 ptiPairArraySort(sorted_ndims, nmodes);
235
236 for(ptiIndex m=0; m<nmodes; ++m) {
237 mode_order[m] = sorted_ndims[nmodes - 1 - m].key;
238 }
239
240 /* Find the location of mode */
241 ptiIndex mode_loc = 0;
242 for(ptiIndex m=0; m<nmodes; ++m) {
243 if(mode_order[m] == mode) {
244 mode_loc = m;
245 }
246 }
247 /* Shift mode to moder_order[0] */
248 if(mode_loc != nmodes - 1) {
249 for(ptiIndex m=mode_loc; m<nmodes; ++m) {
250 mode_order[m] = mode_order[m+1];
251 }
252 mode_order[nmodes - 1] = mode;
253 }
254
255 free(sorted_ndims);
256}

265int ptiSparseTensorMixedOrder(
266 ptiSparseTensor *tsr,
267 const ptiElementIndex sb_bits,
268 const ptiElementIndex sk_bits,
269 int const tk)
270{
271 ptiNnzIndex nnz = tsr->nnz;
272 int result;
273
274 /* Sort tsr in a Row-major Block order to get all kernels. Not use Morton-order for kernels: 1. better support for higher-order tensors by limiting kernel size, because Morton key bit <= 128; */
275 ptiSparseTensorSortIndexRowBlock(tsr, 1, 0, nnz, sk_bits, tk);
276
277 ptiNnzIndexVector kptr, knnzs;
278 result = ptiNewNnzIndexVector(&kptr, 0, 0);
279 pti_CheckError(result, "HiSpTns New", NULL);
280 result = ptiSetKernelPointers(&kptr, &knnzs, tsr, sk_bits);
281 pti_CheckError(result, "HiSpTns Preprocess", NULL);
282
283 /* Sort blocks in each kernel in Morton-order */
284 ptiNnzIndex k_begin, k_end;
285 /* Loop for all kernels, 0-kptr.len for OMP code */
286 for(ptiNnzIndex k=0; k<kptr.len - 1; ++k) {
287 k_begin = kptr.data[k];
288 k_end = kptr.data[k+1]; // exclusive
289 /* Sort blocks in each kernel in Morton-order */
290 ptiSparseTensorSortIndexMorton(tsr, 1, k_begin, k_end, sb_bits, tk);
291
292 }
293
294 return 0;
295}

305 ptiSparseTensor *tsr,
306 ptiIndex const * mode_order,
307 const ptiElementIndex sb_bits,
308 int const tk)
309{
310 ptiNnzIndex nnz = tsr->nnz;
311 ptiIndex * ndims = tsr->ndims;
312 ptiIndex const mode = mode_order[0];
313 int result;
314
315 ptiSparseTensorSortIndexCustomOrder(tsr, mode_order, 1, tk);
316
317 ptiNnzIndexVector ptir;
318 result = ptiNewNnzIndexVector(&ptir, 0, 0);
319 pti_CheckError(result, "HiSpTns New", NULL);
320 ptiNnzIndex slice_nnz = 0;
321 ptiIndex pre_idx = tsr->inds[mode].data[0];
322 result = ptiAppendNnzIndexVector(&ptir, 0);
323 for (ptiNnzIndex z = 0; z < nnz; ++z ) {
324 ++ slice_nnz;
325 if (tsr->inds[mode].data[z] > pre_idx ) {
326 result = ptiAppendNnzIndexVector(&ptir, slice_nnz-1);
327 pre_idx = tsr->inds[mode].data[z];
328 }
329 }
330 result = ptiAppendNnzIndexVector(&ptir, nnz);
331 ptiDumpNnzIndexVector(&ptir, stdout);
332
333 ptiNnzIndex s_begin, s_end;
334 // Loop for slices
335 for(ptiNnzIndex s = 0; s < ndims[mode]; ++ s) {
336 s_begin = ptir.data[s];
337 s_end = ptir.data[s+1]; // exclusive
338 /* Sort blocks in each kernel in plain row-order */
339 ptiSparseTensorSortIndexRowBlock(tsr, 1, s_begin, s_end, sb_bits, tk);
340 }
341
342 return 0;
343}

373{
374 /* Get randomly renumbering indices */

963int pti_SparseTensorCompareIndicesMorton2D(
964 ptiSparseTensor * const tsr1,
965 uint64_t loc1,
966 ptiSparseTensor * const tsr2,
967 uint64_t loc2,
968 ptiIndex * mode_order,
969 ptiElementIndex sb_bits)
970{
971 assert(tsr1->nmodes == tsr2->nmodes);
972 uint64_t mkey1 = 0, mkey2 = 0;
973
974 /* Only support 3-D tensors, with 32-bit indices. */
975 uint32_t x1 = tsr1->inds[mode_order[0]].data[loc1];
976 uint32_t y1 = tsr1->inds[mode_order[1]].data[loc1];
977 uint32_t x2 = tsr2->inds[mode_order[0]].data[loc2];
978 uint32_t y2 = tsr2->inds[mode_order[1]].data[loc2];
979
980 /* Compare block indices */
981 ptiIndex blk_x1 = x1 >> sb_bits;
982 ptiIndex blk_y1 = y1 >> sb_bits;
983 ptiIndex blk_x2 = x2 >> sb_bits;
984 ptiIndex blk_y2 = y2 >> sb_bits;
985
986 if(blk_x1 < blk_x2) {
987 return -1;
988 } else if(blk_x1 > blk_x2) {
989 return 1;
990 } else if(blk_y1 < blk_y2) { // if blk_x1 == blk_x2
991 return -1;
992 } else if(blk_y1 > blk_y2) { // if blk_x1 == blk_x2
993 return 1;
994 }
995
996 /* blk_x1 == blk_x2, blk_y1 == blk_y2, sort inside a block in Z-Morton order */
997 uint64_t x = x1 - (blk_x1 << sb_bits);
998 uint64_t y = y1 - (blk_y1 << sb_bits);
999 x = (x | (x << SHIFTS[3])) & MASKS[3];
1000 x = (x | (x << SHIFTS[2])) & MASKS[2];
1001 x = (x | (x << SHIFTS[1])) & MASKS[1];
1002 x = (x | (x << SHIFTS[0])) & MASKS[0];
1003 y = (y | (y << SHIFTS[3])) & MASKS[3];
1004 y = (y | (y << SHIFTS[2])) & MASKS[2];
1005 y = (y | (y << SHIFTS[1])) & MASKS[1];
1006 y = (y | (y << SHIFTS[0])) & MASKS[0];
1007 mkey1 = y | (x << 1);
1008
1009 x = x2 - (blk_x2 << sb_bits);
1010 y = y2 - (blk_y2 << sb_bits);
1011 x = (x | (x << SHIFTS[3])) & MASKS[3];
1012 x = (x | (x << SHIFTS[2])) & MASKS[2];
1013 x = (x | (x << SHIFTS[1])) & MASKS[1];
1014 x = (x | (x << SHIFTS[0])) & MASKS[0];
1015 y = (y | (y << SHIFTS[3])) & MASKS[3];
1016 y = (y | (y << SHIFTS[2])) & MASKS[2];
1017 y = (y | (y << SHIFTS[1])) & MASKS[1];
1018 y = (y | (y << SHIFTS[0])) & MASKS[0];
1019 mkey2 = y | (x << 1);
1020
1021 if(mkey1 < mkey2) {
1022 return -1;
1023 } else if(mkey1 > mkey2) {
1024 return 1;
1025 } else {
1026 return 0;
1027 }
