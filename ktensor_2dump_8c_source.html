23int ptiDumpKruskalTensor(ptiKruskalTensor *ktsr, FILE *fp)
24{
25 int iores;
26 ptiIndex mode;
27
28 iores = fprintf(fp, "nmodes: %"HIPARTI_PRI_INDEX ", rank: %"HIPARTI_PRI_INDEX "\n", ktsr->nmodes, ktsr->rank);
29 pti_CheckOSError(iores < 0, "KruskalTns Dump");
30 for(mode = 0; mode < ktsr->nmodes; ++mode) {
31 if(mode != 0) {
32 iores = fputs(" ", fp);
33 pti_CheckOSError(iores < 0, "KruskalTns Dump");
34 }
35 iores = fprintf(fp, "%"HIPARTI_PRI_INDEX, ktsr->ndims[mode]);
36 pti_CheckOSError(iores < 0, "KruskalTns Dump");
37 }
38 fputs("\n", fp);
39
40 iores = fprintf(fp, "fit: %lf\n", ktsr->fit);
41 fprintf(fp, "lambda:\n");
42 for(ptiIndex r = 0; r < ktsr->rank; ++r) {

47

51 pti_CheckOSError(iores != 0, "KruskalTns Dump");
