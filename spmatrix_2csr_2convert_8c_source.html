23int ptiSparseMatrixToCSR (ptiSparseMatrixCSR * csrmtx, ptiSparseMatrix * mtx)
24{
25 ptiNnzIndex nnz = mtx->nnz;
26 ptiIndex nrows = mtx->nrows;
27
28 ptiNewSparseMatrixCSR(csrmtx, nrows, mtx->ncols, nnz);
29
30 for (ptiIndex i = 0; i < nrows + 1; i++)
31 csrmtx->rowptr.data[i] = 0;
32
33 for (ptiNnzIndex i = 0; i < nnz; i++)
34 csrmtx->rowptr.data[ mtx->rowind.data[i] ] ++;
35
36 // cumsum the nnz per row to get rowptr
37 for(ptiIndex i = 0, cumsum = 0; i < nrows; i++) {
38 ptiNnzIndex temp = csrmtx->rowptr.data[i];
39 csrmtx->rowptr.data[i] = cumsum;
40 cumsum += temp;
41 }
42 csrmtx->rowptr.data[nrows] = nnz;
43
44 for(ptiNnzIndex i = 0; i < nnz; i++) {
45 ptiIndex row = mtx->rowind.data[i];
46 ptiNnzIndex dest = csrmtx->rowptr.data[row];
47
48 csrmtx->colind.data[dest] = mtx->colind.data[i];
49 csrmtx->values.data[dest] = mtx->values.data[i];
50
51 csrmtx->rowptr.data[row] ++;
52 }
53
54 // Recover rowptr
55 for(ptiIndex i = 0, last = 0; i <= nrows; i++) {
56 ptiNnzIndex temp = csrmtx->rowptr.data[i];
57 csrmtx->rowptr.data[i] = last;
58 last = temp;
59 }
60
61 return 0;
62}
