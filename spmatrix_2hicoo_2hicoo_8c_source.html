21int ptiNewSparseMatrixHiCOO(
22 ptiSparseMatrixHiCOO *himtx,
23 const ptiIndex nrows,
24 const ptiIndex ncols,
25 const ptiNnzIndex nnz,
26 const ptiElementIndex sb_bits,
27 const ptiElementIndex sk_bits)
28{
29 int result;
30
31 himtx->nrows = nrows;
32 himtx->ncols = ncols;
33 himtx->nnz = nnz;
34
35 /* Parameters */
36 himtx->sb_bits = sb_bits; // block size by nnz
37 himtx->sk_bits = sk_bits; // superblock size by nnz
38
39 result = ptiNewNnzIndexVector(&himtx->bptr, 0, 0);
40 pti_CheckError(result, "HiSpMtx New", NULL);
41 result = ptiNewBlockIndexVector(&himtx->bindI, 0, 0);
42 pti_CheckError(result, "HiSpMtx New", NULL);
43 result = ptiNewBlockIndexVector(&himtx->bindJ, 0, 0);
44 pti_CheckError(result, "HiSpMtx New", NULL);
45
46 result = ptiNewElementIndexVector(&himtx->eindI, 0, 0);
47 pti_CheckError(result, "HiSpMtx New", NULL);
48 result = ptiNewElementIndexVector(&himtx->eindJ, 0, 0);
49 pti_CheckError(result, "HiSpMtx New", NULL);
50
51 result = ptiNewValueVector(&himtx->values, 0, 0);
52 pti_CheckError(result, "HiSpMtx New", NULL);
53
54 /* Allocate superblock scheduler */
55 ptiIndex sk = (ptiIndex)pow(2, sk_bits);
56 ptiIndex kernel_ndim = (nrows + sk - 1)/sk;
57 himtx->kschr = (ptiIndexVector*)malloc(kernel_ndim * sizeof(*(himtx->kschr)));
58 pti_CheckOSError(!himtx->kschr, "HiSpTns New");
59 for(ptiIndex i = 0; i < kernel_ndim; ++i) {
60 result = ptiNewIndexVector(&(himtx->kschr[i]), 0, 0);
61 pti_CheckError(result, "HiSpTns New", NULL);
62 }
63 himtx->nkiters = 0;
64
65 result = ptiNewNnzIndexVector(&himtx->kptr, 0, 0);
66 pti_CheckError(result, "HiSpTns New", NULL);
67
68 return 0;
69}

72void ptiFreeSparseMatrixHiCOO(ptiSparseMatrixHiCOO *himtx)
73{
74 ptiFreeNnzIndexVector(&himtx->bptr);
75 ptiFreeBlockIndexVector(&himtx->bindI);
76 ptiFreeBlockIndexVector(&himtx->bindJ);
77 ptiFreeElementIndexVector(&himtx->eindI);
78 ptiFreeElementIndexVector(&himtx->eindJ);
79 ptiFreeValueVector(&himtx->values);
80
81 ptiFreeNnzIndexVector(&himtx->kptr);
82 ptiIndex sk = (ptiIndex)pow(2, himtx->sk_bits);
83 ptiIndex kernel_ndim = (himtx->nrows + sk - 1)/sk;
84 for(ptiIndex i = 0; i < kernel_ndim; ++i) {
85 ptiFreeIndexVector(&(himtx->kschr[i]));
86 }
87 free(himtx->kschr);
88
89 himtx->nnz = 0;
90 himtx->nrows = 0;
91 himtx->ncols = 0;
92 himtx->sb_bits = 0;
93 himtx->sk_bits = 0;
94 himtx->nkiters = 0;
95}
