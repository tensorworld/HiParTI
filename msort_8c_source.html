33void ptiSparseTensorSortIndexAtMode(ptiSparseTensor *tsr, ptiIndex const mode, int force) {
34 ptiIndex m;
35 int needsort = 0;
36
37 for(m = 0; m < mode; ++m) {
38 if(tsr->sortorder[m] != m) {
39 tsr->sortorder[m] = m;
40 needsort = 1;
41 }
42 }
43 for(m = mode+1; m < tsr->nmodes; ++m) {
44 if(tsr->sortorder[m-1] != m) {

46 needsort = 1;

49 if(tsr->sortorder[tsr->nmodes-1] != mode) {
50 tsr->sortorder[tsr->nmodes-1] = mode;
51 needsort = 1;
52 }
53
54 if(needsort || force) {
55 pti_QuickSortAtMode(tsr, 0, tsr->nnz, mode);

59static void pti_QuickSortAtMode(ptiSparseTensor *tsr, ptiNnzIndex const l, ptiNnzIndex const r, ptiIndex const mode) {
60 ptiNnzIndex i, j, p;
61 if(r-l < 2) {
62 return;
63 }
64 p = (l+r) / 2;
65 for(i = l, j = r-1; ; ++i, --j) {
66 while(pti_SparseTensorCompareAtMode(tsr, i, tsr, p, mode) < 0) {
67 ++i;
68 }
69 while(pti_SparseTensorCompareAtMode(tsr, p, tsr, j, mode) < 0) {
70 --j;
71 }
72 if(i >= j) {
73 break;
74 }
75 pti_SwapValues(tsr, i, j);
76 if(i == p) {
77 p = j;
78 } else if(j == p) {
79 p = i;
80 }
81 }
82 pti_QuickSortAtMode(tsr, l, i, mode);
83 pti_QuickSortAtMode(tsr, i, r, mode);
84}

87 ptiIndex i;

93 eleind2 = tsr2->inds[i].data[ind2];
