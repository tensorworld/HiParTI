38int ptiNewRankMatrix(ptiRankMatrix *mtx, ptiIndex const nrows, ptiElementIndex const ncols) {
39 mtx->nrows = nrows;
40 mtx->ncols = ncols;
41 mtx->cap = nrows != 0 ? nrows : 1;
42 mtx->stride = ((ncols-1)/8+1)*8;
43#ifdef _ISOC11_SOURCE
44 mtx->values = aligned_alloc(8 * sizeof (ptiValue), mtx->cap * mtx->stride * sizeof (ptiValue));
45#elif _POSIX_C_SOURCE >= 200112L
46 {
47 int result = posix_memalign((void **) &mtx->values, 8 * sizeof (ptiValue), mtx->cap * mtx->stride * sizeof (ptiValue));
48 if(result != 0) {
49 mtx->values = NULL;
50 }
51 }
52#else
53 mtx->values = malloc(mtx->cap * mtx->stride * sizeof (ptiValue));
54#endif
55 pti_CheckOSError(!mtx->values, "RankMtx New");
56 return 0;
57}

70int ptiRandomizeRankMatrix(ptiRankMatrix *mtx, ptiIndex const nrows, ptiElementIndex const ncols)
71{
72 /* splitmix64, same scheme as ptiRandomizeMatrix: deterministic and
73 reproducible, uniform in [0,1). */
74 uint64_t state = HIPARTI_RANDOM_SEED;
75 for(ptiIndex i=0; i<nrows; ++i)
76 for(ptiElementIndex j=0; j<ncols; ++j) {
77 uint64_t z = (state += 0x9E3779B97F4A7C15ULL);
78 z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
79 z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
80 z = z ^ (z >> 31);
81 mtx->values[i * mtx->stride + j] = (ptiValue)((double)(z >> 11) / 9007199254740992.0);
82 }
83 return 0;
84}

108void ptiRankMatrixInverseShuffleIndices(ptiRankMatrix *mtx, ptiIndex * mode_map_inds) {
109 /* Renumber matrix rows */
110 ptiIndex new_i;
111 ptiValue * tmp_values = malloc(mtx->cap * mtx->stride * sizeof (ptiValue));
112
113 for(ptiIndex i=0; i<mtx->nrows; ++i) {
114 new_i = mode_map_inds[i];
115 for(ptiElementIndex j=0; j<mtx->ncols; ++j) {
116 tmp_values[i * mtx->stride + j] = mtx->values[new_i * mtx->stride + j];
117 }
118 }
119
120 free(mtx->values);
121 mtx->values = tmp_values;
122}

142/* mats (aTa) only stores upper triangle elements. */
143int ptiRankMatrixDotMulSeqTriangle(ptiIndex const mode, ptiIndex const nmodes, ptiRankMatrix ** mats)
144{
145 ptiIndex const nrows = mats[0]->nrows;
146 ptiElementIndex const ncols = mats[0]->ncols;
147 ptiElementIndex const stride = mats[0]->stride;
148 for(ptiIndex m=1; m<nmodes+1; ++m) {
149 assert(mats[m]->ncols == ncols);
150 assert(mats[m]->nrows == nrows);
151 }
152
153 ptiValue * ovals = mats[nmodes]->values;
154#ifdef HIPARTI_USE_OPENMP
155 #pragma omp parallel for schedule(static)
156#endif
157 for(ptiIndex i=0; i < nrows; ++i) {
158 for(ptiElementIndex j=0; j < ncols; ++j) {
159 ovals[j * stride + i] = 1.0;
160 }
161 }
162
163
164 for(ptiIndex m=1; m < nmodes; ++m) {
165 ptiIndex const pm = (mode + m) % nmodes;
166 ptiValue const * vals = mats[pm]->values;
167#ifdef HIPARTI_USE_OPENMP
168 #pragma omp parallel for schedule(static)
169#endif
170 for(ptiIndex i=0; i < nrows; ++i) {
171 for(ptiElementIndex j=i; j < ncols; ++j) {
172 ovals[i * stride + j] *= vals[i * stride + j];
173 }
174 }
175 }
176
177 /* Copy upper triangle to lower part */
178#ifdef HIPARTI_USE_OPENMP
179 #pragma omp parallel for schedule(static)
180#endif
181 for(ptiIndex i=0; i < nrows; ++i) {
182 for(ptiElementIndex j=0; j < i; ++j) {
183 ovals[i * stride + j] = ovals[j * stride + i];
184 }
185 }
186
187 return 0;
188}

191// Row-major
192int ptiRankMatrix2Norm(ptiRankMatrix * const A, ptiValue * const lambda)
193{
194 ptiIndex const nrows = A->nrows;
195 ptiElementIndex const ncols = A->ncols;
196 ptiElementIndex const stride = A->stride;
197 ptiValue * const vals = A->values;
198 ptiValue * buffer_lambda;
199 int nthreads = 1;
200
201#ifdef HIPARTI_USE_OPENMP
202 #pragma omp parallel for schedule(static)
203#endif
204 for(ptiElementIndex j=0; j < ncols; ++j) {
205 lambda[j] = 0.0;
206 }
207
208#ifdef HIPARTI_USE_OPENMP
209 #pragma omp parallel
210 {
211 nthreads = omp_get_num_threads();
212 }
213 buffer_lambda = (ptiValue *)malloc(nthreads * ncols * sizeof(ptiValue));
214#endif
215
216
217#ifdef HIPARTI_USE_OPENMP
218 #pragma omp parallel
219 {
220 #pragma omp for schedule(static)
221 for(ptiNnzIndex j=0; j < (ptiNnzIndex)nthreads * ncols; ++j)
222 buffer_lambda[j] = 0.0;
223
224 int const tid = omp_get_thread_num();
225 int const nthreads = omp_get_num_threads();
226 ptiValue * loc_lambda = buffer_lambda + tid * ncols;
227
228 #pragma omp for
229 for(ptiIndex i=0; i < nrows; ++i) {
230 for(ptiElementIndex j=0; j < ncols; ++j) {
231 loc_lambda[j] += vals[i*stride + j] * vals[i*stride + j];
232 }
233 }
234
235 #pragma omp for schedule(static)
236 for(ptiElementIndex j=0; j < ncols; ++j) {
237 for(ptiIndex i=0; i < (ptiIndex)nthreads; ++i) {
238 lambda[j] += buffer_lambda[i*ncols + j];
239 }
240 }
241 } /* end parallel pragma */
242
243#else
244
245 for(ptiIndex i=0; i < nrows; ++i) {
246 for(ptiElementIndex j=0; j < ncols; ++j) {
247 lambda[j] += vals[i*stride + j] * vals[i*stride + j];
248 }
249 }
250
251#endif
252
253#ifdef HIPARTI_USE_OPENMP
254 #pragma omp parallel for schedule(static)
255#endif
256 for(ptiElementIndex j=0; j < ncols; ++j) {
257 lambda[j] = sqrt(lambda[j]);
258 }
259
260#ifdef HIPARTI_USE_OPENMP
261 #pragma omp parallel for
262#endif
263 for(ptiIndex i=0; i < nrows; ++i) {
264 for(ptiElementIndex j=0; j < ncols; ++j) {
265 vals[i*stride + j] /= lambda[j];
266 }
267 }
268
269
270#ifdef HIPARTI_USE_OPENMP
271 free(buffer_lambda);
272#endif
273
274 return 0;
275}

366void GetRankFinalLambda(
367 ptiElementIndex const rank,
368 ptiIndex const nmodes,
369 ptiRankMatrix ** mats,
370 ptiValue * const lambda)
371{
372 ptiValue * tmp_lambda = (ptiValue *) malloc(rank * sizeof(*tmp_lambda));
373
374 for(ptiIndex m=0; m < nmodes; ++m) {
375 ptiRankMatrix2Norm(mats[m], tmp_lambda);
376 for(ptiElementIndex r=0; r < rank; ++r) {
377 lambda[r] *= tmp_lambda[r];
378 }
379 }
380
381 free(tmp_lambda);
382}
