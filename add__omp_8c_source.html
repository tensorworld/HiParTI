24int ptiSparseTensorAddOMP(ptiSparseTensor *Y, ptiSparseTensor *X, int const nthreads) {
25 /* Ensure X and Y are in same shape */
26 if(Y->nmodes != X->nmodes) {
27 pti_CheckError(PTIERR_SHAPE_MISMATCH, "OMP SpTns Add", "shape mismatch");
28 }
29 for(ptiIndex i = 0; i < X->nmodes; ++i) {
30 if(Y->ndims[i] != X->ndims[i]) {
31 pti_CheckError(PTIERR_SHAPE_MISMATCH, "OMP SpTns Add", "shape mismatch");
32 }
33 }
34
35 /* Determine partationing strategy. */
36 ptiNnzIndex * dist_nnzs_X = (ptiNnzIndex*)malloc(nthreads*sizeof(ptiNnzIndex));
37 ptiNnzIndex * dist_nnzs_Y = (ptiNnzIndex*)malloc(nthreads*sizeof(ptiNnzIndex));
38 ptiIndex * dist_nrows_Y = (ptiIndex*)malloc(nthreads*sizeof(ptiIndex));
39
40 pti_DistSparseTensor(Y, nthreads, dist_nnzs_Y, dist_nrows_Y);
41 pti_DistSparseTensorFixed(X, nthreads, dist_nnzs_X, dist_nnzs_Y);
42 free(dist_nrows_Y);
43
44 printf("dist_nnzs_Y:\n");
45 for(int i=0; i<nthreads; ++i) {
46 printf("%" HIPARTI_PRI_NNZ_INDEX " ", dist_nnzs_Y[i]);
47 }
48 printf("\n");
49 printf("dist_nnzs_X:\n");
50 for(int i=0; i<nthreads; ++i) {
51 printf("%" HIPARTI_PRI_NNZ_INDEX " ", dist_nnzs_X[i]);
52 }
53 printf("\n");
54 fflush(stdout);
55
56
57 /* Build a private arrays to append values. */
58 ptiNnzIndex nnz_gap = llabs((long long) Y->nnz - (long long) X->nnz);
59 ptiNnzIndex increase_size = 0;
60 if(nnz_gap == 0) increase_size = 10;
61 else increase_size = nnz_gap;
62
63 ptiIndexVector **local_inds = (ptiIndexVector**)malloc(nthreads* sizeof *local_inds);
64 for(int k=0; k<nthreads; ++k) {
65 local_inds[k] = (ptiIndexVector*)malloc(Y->nmodes* sizeof *(local_inds[k]));
66 for(ptiIndex m=0; m<Y->nmodes; ++m) {
67 ptiNewIndexVector(&(local_inds[k][m]), 0, increase_size);
68 }
69 }
70
71 ptiValueVector *local_vals = (ptiValueVector*)malloc(nthreads* sizeof *local_vals);
72 for(int k=0; k<nthreads; ++k) {
73 ptiNewValueVector(&(local_vals[k]), 0, increase_size);
74 }
75
76 /* Add elements one by one, assume indices are ordered */
77 ptiNnzIndex Ynnz = 0;
78 omp_set_dynamic(0);
79 omp_set_num_threads(nthreads);
80 #pragma omp parallel reduction(+:Ynnz)
81 {
82 int tid = omp_get_thread_num();
83 ptiNnzIndex i=0, j=0;
84 Ynnz = dist_nnzs_Y[tid];
85
86 while(i < dist_nnzs_X[tid] && j < dist_nnzs_Y[tid]) {
87 int compare = pti_SparseTensorCompareIndices(X, i, Y, j);
88 if(compare > 0) { // X(i) > Y(j)
89 ++j;
90 } else if(compare < 0) { // X(i) < Y(j)
91 ptiIndex mode;
92 int result;
93 for(mode = 0; mode < X->nmodes; ++mode) {
94 result = ptiAppendIndexVector(&(local_inds[tid][mode]), X->inds[mode].data[i]);
95 pti_CheckOmpError(result, "OMP SpTns Add", NULL);
96 }
97 result = ptiAppendValueVector(&(local_vals[tid]), X->values.data[i]);
98 pti_CheckOmpError(result, "OMP SpTns Add", NULL);
99 ++Ynnz;
100 ++i;
101 } else { // X(i) = Y(j)
102 Y->values.data[j] += X->values.data[i];
103 ++i;
104 ++j;
105 }
106 }
107 /* Append remaining elements of X to Y */
108 while(i < dist_nnzs_X[tid]) {
109 ptiIndex mode;
110 int result;
111 for(mode = 0; mode < X->nmodes; ++mode) {
112 result = ptiAppendIndexVector(&(local_inds[tid][mode]), X->inds[mode].data[i]);
113 pti_CheckOmpError(result, "OMP SpTns Add", NULL);
114 }
115 result = ptiAppendValueVector(&(local_vals[tid]), X->values.data[i]);
116 pti_CheckOmpError(result, "OMP SpTns Add", NULL);
117 ++Ynnz;
118 ++i;
119 }
120
121 }
122 Y->nnz = Ynnz;
123
124 /* Append all the local arrays to Y. */
125 for(int k=0; k<nthreads; ++k) {
126 for(ptiIndex m=0; m<Y->nmodes; ++m) {
127 ptiAppendIndexVectorWithVector(&(Y->inds[m]), &(local_inds[k][m]));
128 }
129 ptiAppendValueVectorWithVector(&(Y->values), &(local_vals[k]));
130 }
131
132
133 for(int k=0; k<nthreads; ++k) {
134 for(ptiIndex m=0; m<Y->nmodes; ++m) {
135 ptiFreeIndexVector(&(local_inds[k][m]));
136 }
137 free(local_inds[k]);
138 ptiFreeValueVector(&(local_vals[k]));
139 }
140 free(local_inds);
141 free(local_vals);
142 free(dist_nnzs_X);
143 free(dist_nnzs_Y);
144
145
146 /* Check whether elements become zero after adding.
147 If so, fill the gap with the [nnz-1]'th element.
148 */
149 pti_SparseTensorCollectZeros(Y);
150 /* Sort the indices */
151 ptiSparseTensorSortIndex(Y, 1, nthreads);
152
153 return 0;
154}
