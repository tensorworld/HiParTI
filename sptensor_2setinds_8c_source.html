40 ptiNnzIndex lastidx = ref->nnz;
41 for(ptiNnzIndex i = 0; i < ref->nnz; ++i) {
42 if(lastidx == ref->nnz || pti_SparseTensorCompareIndicesCustomize(ref, lastidx, mode_order, ref, i, mode_order, num_cmodes) != 0) {

44 if(fiberidx != NULL) {
45 result = ptiAppendNnzIndexVector(fiberidx, i);

51 result = ptiAppendNnzIndexVector(fiberidx, ref->nnz);
52 pti_CheckError(result, "SspTns SetIndices", NULL);
