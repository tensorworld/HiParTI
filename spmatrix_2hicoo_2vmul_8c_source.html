22int ptiSparseMatrixMulVectorHiCOO(ptiValueVector * y, const ptiSparseMatrixHiCOO *himtx, ptiValueVector * x)
23{
24 ptiNnzIndex nb = himtx->bptr.len - 1;
25 ptiElementIndex sb_bits = himtx->sb_bits;
26
27 for(ptiNnzIndex b = 0; b < nb; ++b) { // Loop blocks
28 ptiNnzIndex bptr_begin = himtx->bptr.data[b];
29 ptiNnzIndex bptr_end = himtx->bptr.data[b+1];
30 ptiValue * restrict blocked_yvals = y->data + (himtx->bindI.data[b] << sb_bits);
31 ptiValue * restrict blocked_xvals = x->data + (himtx->bindJ.data[b] << sb_bits);
32
33 for(ptiNnzIndex z=bptr_begin; z<bptr_end; ++z) { // Loop entries
34 ptiElementIndex ei = himtx->eindI.data[z];
35 ptiElementIndex ej = himtx->eindJ.data[z];
36 blocked_yvals[ei] += himtx->values.data[z] * blocked_xvals[ej];
37 }
38 }
39
40 return 0;
41}

69{

71 ptiIndex sk = (ptiIndex)pow(2, himtx->sk_bits);
72 ptiIndex num_kernel_dim = (himtx->nrows + sk - 1) / sk;
73
74 /* Loop parallel iterations */
75 for(ptiIndex i=0; i<himtx->nkiters; ++i) { // Loop schedule columns
76
77 #pragma omp parallel for schedule(dynamic, 1)

124int ptiOmpSparseMatrixMulVectorHiCOO_Schedule_Reduce(ptiValueVector * y, ptiValueVector * ybufs, const ptiSparseMatrixHiCOO *himtx, ptiValueVector * x)
125{
126 ptiElementIndex sb_bits = himtx->sb_bits;
127 ptiIndex sk = (ptiIndex)pow(2, himtx->sk_bits);
128 ptiIndex num_kernel_dim = (himtx->nrows + sk - 1) / sk;
129 int nthreads;
130 #pragma omp parallel
131 nthreads = omp_get_num_threads();
132
133 /* Loop parallel iterations */
134 #pragma omp parallel for schedule(dynamic, 1)
135 for(ptiIndex i=0; i<himtx->nkiters; ++i) { // Loop schedule columns
136 // int tid = omp_get_thread_num();

138 for(ptiIndex k=0; k<num_kernel_dim; ++k) { // Loop schedule rows
139 if(i >= himtx->kschr[k].len) {

141 }
142
143 ptiIndex kptr_loc = himtx->kschr[k].data[i];
144 ptiNnzIndex kptr_begin = himtx->kptr.data[kptr_loc];
145 ptiNnzIndex kptr_end = himtx->kptr.data[kptr_loc+1];
146
147 /* Loop blocks in a kernel */
148 for(ptiNnzIndex b = kptr_begin; b < kptr_end; ++b) { // Loop blocks
149 ptiNnzIndex bptr_begin = himtx->bptr.data[b];
150 ptiNnzIndex bptr_end = himtx->bptr.data[b+1];
151 ptiValue * restrict blocked_yvals = y->data + (himtx->bindI.data[b] << sb_bits);
152 ptiValue * restrict blocked_xvals = x->data + (himtx->bindJ.data[b] << sb_bits);

155 ptiElementIndex ei = himtx->eindI.data[z];
156 ptiElementIndex ej = himtx->eindJ.data[z];
157 blocked_yvals[ei] += himtx->values.data[z] * blocked_xvals[ej];
158 }

166 for(int t=0; t<nthreads; ++t) {
