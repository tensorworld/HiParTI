29int ptiNewSparseTensorHiCOO(
30 ptiSparseTensorHiCOO *hitsr,
31 const ptiIndex nmodes,
32 const ptiIndex ndims[],
33 const ptiNnzIndex nnz,
34 const ptiElementIndex sb_bits,
35 const ptiElementIndex sk_bits,
36 const ptiElementIndex sc_bits)
37{
38 ptiIndex i;
39 int result;
40
41 hitsr->nmodes = nmodes;
42 hitsr->sortorder = malloc(nmodes * sizeof hitsr->sortorder[0]);
43 for(i = 0; i < nmodes; ++i) {
44 hitsr->sortorder[i] = i;
45 }
46 hitsr->ndims = malloc(nmodes * sizeof *hitsr->ndims);
47 pti_CheckOSError(!hitsr->ndims, "HiSpTns New");
48 memcpy(hitsr->ndims, ndims, nmodes * sizeof *hitsr->ndims);
49 hitsr->nnz = nnz;
50
51 /* Parameters */
52 hitsr->sb_bits = sb_bits; // block size by nnz
53 hitsr->sk_bits = sk_bits; // kernel size by nnz
54 hitsr->sc_bits = sc_bits; // chunk size by blocks
55 ptiIndex sk = (ptiIndex)pow(2, sk_bits);
56
57 hitsr->kschr = (ptiIndexVector**)malloc(nmodes * sizeof *hitsr->kschr);
58 pti_CheckOSError(!hitsr->kschr, "HiSpTns New");
59 for(ptiIndex m = 0; m < nmodes; ++m) {
60 ptiIndex kernel_ndim = (ndims[m] + sk - 1)/sk;
61 hitsr->kschr[m] = (ptiIndexVector*)malloc(kernel_ndim * sizeof(*(hitsr->kschr[m])));
62 pti_CheckOSError(!hitsr->kschr[m], "HiSpTns New");
63 for(ptiIndex i = 0; i < kernel_ndim; ++i) {
64 result = ptiNewIndexVector(&(hitsr->kschr[m][i]), 0, 0);
65 pti_CheckError(result, "HiSpTns New", NULL);
66 }
67 }
68 hitsr->nkiters = (ptiIndex*)malloc(nmodes * sizeof *hitsr->nkiters);
69
70 result = ptiNewNnzIndexVector(&hitsr->kptr, 0, 0);
71 pti_CheckError(result, "HiSpTns New", NULL);
72 result = ptiNewNnzIndexVector(&hitsr->cptr, 0, 0);
73 pti_CheckError(result, "HiSpTns New", NULL);

75 /* Balanced structures */

77 pti_CheckOSError(!hitsr->kschr_balanced, "HiSpTns New");

81 pti_CheckOSError(!hitsr->kschr_balanced[m], "HiSpTns New");
82 for(ptiIndex i = 0; i < kernel_ndim; ++i) {

84 pti_CheckError(result, "HiSpTns New", NULL);
85 }

88 pti_CheckOSError(!hitsr->kschr_balanced_pos, "HiSpTns New");
89 for(ptiIndex m = 0; m < nmodes; ++m) {
90 ptiIndex kernel_ndim = (ndims[m] + sk - 1)/sk;
91 hitsr->kschr_balanced_pos[m] = (ptiIndexVector*)malloc(kernel_ndim * sizeof(*(hitsr->kschr_balanced_pos[m])));
92 pti_CheckOSError(!hitsr->kschr_balanced_pos[m], "HiSpTns New");
93 for(ptiIndex i = 0; i < kernel_ndim; ++i) {
94 result = ptiNewIndexVector(&(hitsr->kschr_balanced_pos[m][i]), 0, 0);
95 pti_CheckError(result, "HiSpTns New", NULL);
96 }
97 }
98 hitsr->nkpars = (ptiIndex*)malloc(nmodes * sizeof(ptiIndex));
99 pti_CheckOSError(!hitsr->nkpars, "HiSpTns New");
100 hitsr->kschr_rest = (ptiIndexVector*)malloc(nmodes * sizeof *hitsr->kschr_rest);
101 pti_CheckOSError(!hitsr->kschr_rest, "HiSpTns New");
102 for(ptiIndex m = 0; m < nmodes; ++m) {
103 result = ptiNewIndexVector(&(hitsr->kschr_rest[m]), 0, 0);
104 pti_CheckError(result, "HiSpTns New", NULL);
105 }
106 result = ptiNewNnzIndexVector(&hitsr->knnzs, 0, 0);
107 pti_CheckError(result, "HiSpTns New", NULL);
108
109 result = ptiNewNnzIndexVector(&hitsr->bptr, 0, 0);
110 pti_CheckError(result, "HiSpTns New", NULL);
111 hitsr->binds = malloc(nmodes * sizeof *hitsr->binds);
112 pti_CheckOSError(!hitsr->binds, "HiSpTns New");
113 for(i = 0; i < nmodes; ++i) {
114 result = ptiNewBlockIndexVector(&hitsr->binds[i], 0, 0);
115 pti_CheckError(result, "HiSpTns New", NULL);
116 }
117
118 hitsr->einds = malloc(nmodes * sizeof *hitsr->einds);
119 pti_CheckOSError(!hitsr->einds, "HiSpTns New");
120 for(i = 0; i < nmodes; ++i) {
121 result = ptiNewElementIndexVector(&hitsr->einds[i], 0, 0);
122 pti_CheckError(result, "HiSpTns New", NULL);
123 }
124 result = ptiNewValueVector(&hitsr->values, 0, 0);
125 pti_CheckError(result, "HiSpTns New", NULL);
126
127
128 return 0;
129}

209void ptiFreeSparseTensorHiCOO(ptiSparseTensorHiCOO *hitsr)
210{
211 ptiIndex i;
212 ptiIndex nmodes = hitsr->nmodes;
213 ptiIndex sk = (ptiIndex)pow(2, hitsr->sk_bits);
214
215 for(ptiIndex m = 0; m < nmodes; ++m) {
216 ptiIndex kernel_ndim = (hitsr->ndims[m] + sk - 1)/sk;
217 for(i = 0; i < kernel_ndim; ++i) {
218 ptiFreeIndexVector(&(hitsr->kschr[m][i]));
219 }
220 free(hitsr->kschr[m]);
221 }
222 free(hitsr->kschr);
223 free(hitsr->nkiters);
224
225 ptiFreeNnzIndexVector(&hitsr->kptr);
226 ptiFreeNnzIndexVector(&hitsr->cptr);
227
228 ptiFreeNnzIndexVector(&hitsr->bptr);
229 for(i = 0; i < nmodes; ++i) {
230 ptiFreeBlockIndexVector(&hitsr->binds[i]);
231 ptiFreeElementIndexVector(&hitsr->einds[i]);
232 }
233 free(hitsr->binds);
234 free(hitsr->einds);
235 ptiFreeValueVector(&hitsr->values);
236
237 hitsr->nmodes = 0;
238 hitsr->nnz = 0;
239 hitsr->sb_bits = 0;
240 hitsr->sk_bits = 0;
241 hitsr->sc_bits = 0;
242
243 free(hitsr->sortorder);
244 free(hitsr->ndims);
245}

248double SparseTensorFrobeniusNormSquaredHiCOO(ptiSparseTensorHiCOO const * const hitsr)
249{
250 double norm = 0;
251 ptiValue const * const restrict vals = hitsr->values.data;
252
253#ifdef HIPARTI_USE_OPENMP
254 #pragma omp parallel for reduction(+:norm)
255#endif
256 for(size_t n=0; n < hitsr->nnz; ++n) {
257 norm += vals[n] * vals[n];
258 }
259 return norm;
260}
