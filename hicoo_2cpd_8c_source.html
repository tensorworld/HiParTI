31double CpdAlsStepHiCOO(
32 ptiSparseTensorHiCOO const * const hitsr,
33 ptiIndex const rank,
34 ptiIndex const niters,
35 double const tol,
36 ptiRankMatrix ** mats,
37 ptiValue * const lambda)
38{
39 ptiIndex const nmodes = hitsr->nmodes;
40 ptiIndex const stride = mats[0]->stride;
41 double fit = 0;
42
43 for(ptiIndex m=0; m < nmodes; ++m) {
44 ptiAssert(hitsr->ndims[m] == mats[m]->nrows);
45 ptiAssert(mats[m]->ncols == rank);
46 }
47
48 ptiValue alpha = 1.0, beta = 0.0;
49 char notrans = 'N';
50 char trans = 'T';
51 char uplo = 'L';
52 int blas_rank = (int) rank;
53 int blas_stride = (int) stride;
54
55 ptiRankMatrix * tmp_mat = mats[nmodes];
56 ptiRankMatrix ** ata = (ptiRankMatrix **)malloc((nmodes+1) * sizeof(*ata));
57 for(ptiIndex m=0; m < nmodes+1; ++m) {
58 ata[m] = (ptiRankMatrix *)malloc(sizeof(ptiRankMatrix));
59 ptiAssert(ptiNewRankMatrix(ata[m], rank, rank) == 0);
60 ptiAssert(mats[m]->stride == ata[m]->stride);
61 }
62
63 /* Compute all "ata"s */
64 for(ptiIndex m=0; m < nmodes; ++m) {
65 /* ata[m] = mats[m]^T * mats[m]), actually do A * A' due to row-major mats, and output an upper triangular matrix. */
66 int blas_nrows = (int)(mats[m]->nrows);
67 ssyrk_(&uplo, &notrans, &blas_rank, &blas_nrows, &alpha,
68 mats[m]->values, &blas_stride, &beta, ata[m]->values, &blas_stride);
69 }
70 // printf("Initial mats:\n");
71 // for(size_t m=0; m < nmodes+1; ++m)
72 // ptiDumpRankMatrix(mats[m], stdout);
73 // printf("Initial ata:\n");
74 // for(ptiIndex m=0; m < nmodes+1; ++m)
75 // ptiDumpRankMatrix(ata[m], stdout);
76
77 double oldfit = 0;
78 ptiIndex * mats_order = (ptiIndex*)malloc(nmodes * sizeof(*mats_order));
79
80 for(ptiIndex it=0; it < niters; ++it) {
81 // printf(" its = %3lu\n", it+1);
82 ptiTimer timer;
83 ptiNewTimer(&timer, 0);
84 ptiStartTimer(timer);
85
86 for(ptiIndex m=0; m < nmodes; ++m) {
87 printf("\nmode %u \n", m);
88 tmp_mat->nrows = mats[m]->nrows;
89
90 /* Factor Matrices order */
91 mats_order[0] = m;
92 for(ptiIndex i=1; i<nmodes; ++i)
93 mats_order[i] = (m+i) % nmodes;
94
95 ptiAssert (ptiMTTKRPHiCOO_MatrixTiling(hitsr, mats, mats_order, m) == 0);
96 // printf("ptiMTTKRPHiCOO_MatrixTiling mats[nmodes]:\n");
97 // ptiDumpRankMatrix(mats[nmodes], stdout);
98
99 memcpy(mats[m]->values, tmp_mat->values, mats[m]->nrows * stride * sizeof(ptiValue));
100 /* Solve ? * ata[nmodes] = mats[nmodes] (tmp_mat) */
101 ptiAssert ( ptiRankMatrixSolveNormals(m, nmodes, ata, mats[m]) == 0 );
102 // printf("Inverse mats[m]:\n");
103 // ptiDumpRankMatrix(mats[m], stdout);
104
105 /* Normalized mats[m], store the norms in lambda. Use different norms to avoid precision explosion. */
106 if (it == 0 ) {
107 ptiRankMatrix2Norm(mats[m], lambda);
108 } else {
109 ptiRankMatrixMaxNorm(mats[m], lambda);
110 }
111 // printf("Normalize mats[m]:\n");
112 // ptiDumpRankMatrix(mats[m], stdout);
113 // printf("lambda:\n");
114 // for(size_t i=0; i<rank; ++i)
115 // printf("%lf ", lambda[i]);
116 // printf("\n\n");
117
118 /* ata[m] = mats[m]^T * mats[m]) */
119 int blas_nrows = (int)(mats[m]->nrows);
120 ssyrk_(&uplo, &notrans, &blas_rank, &blas_nrows, &alpha,
121 mats[m]->values, &blas_stride, &beta, ata[m]->values, &blas_stride);
122 // printf("Update ata[m]:\n");
123 // ptiDumpRankMatrix(ata[m], stdout);
124
125 } // Loop nmodes
126
127 // PrintDenseValueVector(lambda, rank, "lambda", "debug.txt");
128 fit = KruskalTensorFitHiCOO(hitsr, lambda, mats, ata);
129
130 ptiStopTimer(timer);
131 double its_time = ptiElapsedTime(timer);
132 ptiFreeTimer(timer);
133
134 printf(" its = %3u ( %.3lf s ) fit = %0.5f delta = %+0.4e\n",
135 it+1, its_time, fit, fit - oldfit);
136 if(it > 0 && fabs(fit - oldfit) < tol) {
137 break;
138 }
139 oldfit = fit;
140
141 } // Loop niters
142
143 GetRankFinalLambda(rank, nmodes, mats, lambda);
144
145 for(ptiIndex m=0; m < nmodes+1; ++m) {
146 ptiFreeRankMatrix(ata[m]);
147 }
148 free(ata);
149 free(mats_order);
150
151
152 return fit;
153}

156int ptiCpdAlsHiCOO(
157 ptiSparseTensorHiCOO const * const hitsr,
158 ptiIndex const rank,
159 ptiIndex const niters,
160 double const tol,
161 ptiRankKruskalTensor * ktensor)
162{
163 ptiIndex nmodes = hitsr->nmodes;
164#ifdef HIPARTI_USE_MAGMA
165 magma_init();
166#endif
167
168 /* Initialize factor matrices */
169 ptiIndex max_dim = 0;
170 for(ptiIndex m=0; m < nmodes; ++m) {
171 max_dim = (hitsr->ndims[m] > max_dim) ? hitsr->ndims[m] : max_dim;
172 }
173 ptiRankMatrix ** mats = (ptiRankMatrix **)malloc((nmodes+1) * sizeof(*mats));
174 for(ptiIndex m=0; m < nmodes+1; ++m) {
175 mats[m] = (ptiRankMatrix *)malloc(sizeof(ptiRankMatrix));
176 }
177 for(ptiIndex m=0; m < nmodes; ++m) {
178 ptiAssert(ptiNewRankMatrix(mats[m], hitsr->ndims[m], rank) == 0);
179 // ptiAssert(ptiConstantRankMatrix(mats[m], 1) == 0);
180 ptiAssert(ptiRandomizeRankMatrix(mats[m], hitsr->ndims[m], rank) == 0);
181 }
182 ptiAssert(ptiNewRankMatrix(mats[nmodes], max_dim, rank) == 0);
183 ptiAssert(ptiConstantRankMatrix(mats[nmodes], 0) == 0);
184 // printf("max_dim: %u\n", max_dim);
185
186 ptiTimer timer;
187 ptiNewTimer(&timer, 0);
188 ptiStartTimer(timer);
189
190 ktensor->fit = CpdAlsStepHiCOO(hitsr, rank, niters, tol, mats, ktensor->lambda);
191
192 ptiStopTimer(timer);
193 ptiPrintElapsedTime(timer, "CPU HiCOO SpTns CPD-ALS");
194 ptiFreeTimer(timer);
195
196 ktensor->factors = mats;
197
198#ifdef HIPARTI_USE_MAGMA
199 magma_finalize();
200#endif
201 ptiFreeRankMatrix(mats[nmodes]);
202
203 return 0;
204}
