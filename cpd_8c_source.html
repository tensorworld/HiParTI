32double CpdAlsStep(
33 ptiSparseTensor const * const ptien,
34 ptiIndex const rank,
35 ptiIndex const niters,
36 double const tol,
37 ptiMatrix ** mats, // Row-major
38 ptiValue * const lambda)
39{
40 ptiIndex const nmodes = ptien->nmodes;
41 ptiIndex const stride = mats[0]->stride;
42 double fit = 0;
43
44 for(ptiIndex m=0; m < nmodes; ++m) {
45 ptiAssert(ptien->ndims[m] == mats[m]->nrows);
46 ptiAssert(mats[m]->ncols == rank);
47 }
48
49 ptiValue alpha = 1.0, beta = 0.0;
50 char notrans = 'N';
51 char trans = 'T';
52 char uplo = 'L';
53 int blas_rank = (int) rank;
54 int blas_stride = (int) stride;
55
56 ptiMatrix * tmp_mat = mats[nmodes];
57 ptiMatrix ** ata = (ptiMatrix **)malloc((nmodes+1) * sizeof(*ata)); // symmetric matrices, but in column-major
58 for(ptiIndex m=0; m < nmodes+1; ++m) {
59 ata[m] = (ptiMatrix *)malloc(sizeof(ptiMatrix));
60 ptiAssert(ptiNewMatrix(ata[m], rank, rank) == 0);
61 ptiAssert(mats[m]->stride == ata[m]->stride);
62 }
63
64 /* Compute all "ata"s */
65 for(ptiIndex m=0; m < nmodes; ++m) {
66 /* ata[m] = mats[m]^T * mats[m]), actually do A * A' due to row-major mats, and output an upper triangular matrix. */
67 int blas_nrows = (int)(mats[m]->nrows);
68 ssyrk_(&uplo, &notrans, &blas_rank, &blas_nrows, &alpha,
69 mats[m]->values, &blas_stride, &beta, ata[m]->values, &blas_stride);
70 // blasf77_sgemm("N", "T", (magma_int_t*)&rank, (magma_int_t*)&rank, (magma_int_t*)&(mats[m]->nrows), &alpha,
71 // mats[m]->values, (magma_int_t*)&(mats[m]->stride), mats[m]->values, (magma_int_t*)&(mats[m]->stride), &beta, ata[m]->values, (magma_int_t*)&(ata[m]->stride));
72 }
73 // printf("Initial mats:\n");
74 // for(size_t m=0; m < nmodes+1; ++m)
75 // ptiDumpMatrix(mats[m], stdout);
76 // printf("Initial ata:\n");
77 // for(size_t m=0; m < nmodes+1; ++m)
78 // ptiDumpMatrix(ata[m], stdout);
79
80
81 double oldfit = 0;
82 ptiIndex * mats_order = (ptiIndex*)malloc(nmodes * sizeof(*mats_order));
83
84
85 for(ptiIndex it=0; it < niters; ++it) {
86 // printf(" its = %3lu\n", it+1);
87 ptiTimer timer;
88 ptiNewTimer(&timer, 0);
89 ptiStartTimer(timer);
90
91 for(ptiIndex m=0; m < nmodes; ++m) {
92 // printf("\nmode %u \n", m);
93 tmp_mat->nrows = mats[m]->nrows;
94
95 /* Factor Matrices order */
96 mats_order[0] = m;
97 for(ptiIndex i=1; i<nmodes; ++i)
98 mats_order[i] = (m+i) % nmodes;
99
100 // mats[nmodes]: row-major
101 ptiAssert (ptiMTTKRP(ptien, mats, mats_order, m) == 0);
102 // printf("ptiMTTKRP mats[nmodes]:\n");
103 // ptiDumpMatrix(mats[nmodes], stdout);
104
105 // Column-major calculation
106 memcpy(mats[m]->values, tmp_mat->values, mats[m]->nrows * stride * sizeof(ptiValue));
107
108 /* Solve ? * ata[nmodes] = mats[nmodes] (tmp_mat) */
109 ptiAssert ( ptiMatrixSolveNormals(m, nmodes, ata, mats[m]) == 0 );
110 // printf("Inverse mats[m]:\n");
111 // ptiDumpMatrix(mats[m], stdout);
112
113 /* Normalized mats[m], store the norms in lambda. Use different norms to avoid precision explosion. */
114 if (it == 0 ) {
115 ptiMatrix2Norm(mats[m], lambda);
116 } else {
117 ptiMatrixMaxNorm(mats[m], lambda);
118 }
119 // printf("Normalize mats[m]:\n");
120 // ptiDumpMatrix(mats[m], stdout);
121 // printf("lambda:\n");
122 // for(size_t i=0; i<rank; ++i)
123 // printf("%lf ", lambda[i]);
124 // printf("\n\n");
125
126 /* ata[m] = mats[m]^T * mats[m]) */
127 int blas_nrows = (int)(mats[m]->nrows);
128 ssyrk_(&uplo, &notrans, &blas_rank, &blas_nrows, &alpha,
129 mats[m]->values, &blas_stride, &beta, ata[m]->values, &blas_stride);
130 // printf("Update ata[m]:\n");
131 // ptiDumpMatrix(ata[m], stdout);
132
133 } // Loop nmodes
134
135 // PrintDenseValueVector(lambda, rank, "lambda", "debug.txt");
136 fit = KruskalTensorFit(ptien, lambda, mats, ata);
137
138 ptiStopTimer(timer);
139 double its_time = ptiElapsedTime(timer);
140 ptiFreeTimer(timer);
141
142 printf(" its = %"HIPARTI_PRI_INDEX " ( %.3lf s ) fit = %0.5f delta = %+0.4e\n",
143 it+1, its_time, fit, fit - oldfit);
144 if(it > 0 && fabs(fit - oldfit) < tol) {
145 break;
146 }
147 oldfit = fit;
148
149 } // Loop niters
150
151 GetFinalLambda(rank, nmodes, mats, lambda);
152
153 for(ptiIndex m=0; m < nmodes+1; ++m) {
154 ptiFreeMatrix(ata[m]);
155 }
156 free(ata);
157 free(mats_order);
158
159 return fit;
160}

163int ptiCpdAls(
164 ptiSparseTensor const * const ptien,
165 ptiIndex const rank,
166 ptiIndex const niters,
167 double const tol,
168 ptiKruskalTensor * ktensor)
169{
170 ptiIndex nmodes = ptien->nmodes;
171#ifdef HIPARTI_USE_MAGMA
172 magma_init();
173#endif
174
175 /* Initialize factor matrices */
176 ptiIndex max_dim = ptiMaxIndexArray(ptien->ndims, nmodes);
177 ptiMatrix ** mats = (ptiMatrix **)malloc((nmodes+1) * sizeof(*mats));
178 for(ptiIndex m=0; m < nmodes+1; ++m) {
179 mats[m] = (ptiMatrix *)malloc(sizeof(ptiMatrix));
180 }
181 for(ptiIndex m=0; m < nmodes; ++m) {
182 ptiAssert(ptiNewMatrix(mats[m], ptien->ndims[m], rank) == 0);
183 // ptiAssert(ptiConstantMatrix(mats[m], 1) == 0);
184 ptiAssert(ptiRandomizeMatrix(mats[m]) == 0);
185 }
186 ptiAssert(ptiNewMatrix(mats[nmodes], max_dim, rank) == 0);
187 ptiAssert(ptiConstantMatrix(mats[nmodes], 0) == 0);
188
189
190 ptiTimer timer;
191 ptiNewTimer(&timer, 0);
192 ptiStartTimer(timer);
193
194 ktensor->fit = CpdAlsStep(ptien, rank, niters, tol, mats, ktensor->lambda);
195
196 ptiStopTimer(timer);
197 ptiPrintElapsedTime(timer, "CPU SpTns CPD-ALS");
198 ptiFreeTimer(timer);
199
200 ktensor->factors = mats;
201
202#ifdef HIPARTI_USE_MAGMA
203 magma_finalize();
204#endif
205 ptiFreeMatrix(mats[nmodes]);
206
207 return 0;
208}
