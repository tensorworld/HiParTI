37int ptiNewMatrix(ptiMatrix *mtx, ptiIndex const nrows, ptiIndex const ncols) {
38 mtx->nrows = nrows;
39 mtx->ncols = ncols;
40 mtx->cap = nrows != 0 ? nrows : 1;
41 mtx->stride = ((ncols-1)/8+1)*8;
42#ifdef _ISOC11_SOURCE
43 mtx->values = aligned_alloc(8 * sizeof (ptiValue), mtx->cap * mtx->stride * sizeof (ptiValue));
44#elif _POSIX_C_SOURCE >= 200112L
45 {
46 int result = posix_memalign((void **) &mtx->values, 8 * sizeof (ptiValue), mtx->cap * mtx->stride * sizeof (ptiValue));
47 if(result != 0) {
48 mtx->values = NULL;
49 }
50 }
51#else
52 mtx->values = malloc(mtx->cap * mtx->stride * sizeof (ptiValue));
53#endif
54 pti_CheckOSError(!mtx->values, "Mtx New");
55 return 0;
56}

68int ptiRandomizeMatrix(ptiMatrix *mtx) {
69 /* splitmix64: deterministic and reproducible across runs and platforms, so a
70 test can regenerate the exact same matrix and check the result independently.
71 Row-major over [0, ncols), one draw per element. */
72 uint64_t state = HIPARTI_RANDOM_SEED;
73 for(ptiIndex i=0; i<mtx->nrows; ++i)
74 for(ptiIndex j=0; j<mtx->ncols; ++j) {
75 uint64_t z = (state += 0x9E3779B97F4A7C15ULL);
76 z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9ULL;
77 z = (z ^ (z >> 27)) * 0x94D049BB133111EBULL;
78 z = z ^ (z >> 31);
79 /* top 53 bits -> uniform in [0,1) */
80 mtx->values[i * mtx->stride + j] = (ptiValue)((double)(z >> 11) / 9007199254740992.0);
81 }
82 return 0;
83}

94int ptiIdentityMatrix(ptiMatrix *mtx) {
95 ptiIndex const nrows = mtx->nrows;
96 ptiIndex const ncols = mtx->ncols;
97 assert(nrows == ncols);
98 for(ptiIndex i=0; i<nrows; ++i)
99 for(ptiIndex j=0; j<ncols; ++j)
100 mtx->values[i * mtx->stride + j] = 0;
101 for(ptiIndex i=0; i<nrows; ++i)
102 mtx->values[i * mtx->stride + i] = 1;
103
104 return 0;

130void ptiMatrixInverseShuffleIndices(ptiMatrix *mtx, ptiIndex * mode_map_inds) {
131 /* Renumber matrix rows */
132 ptiIndex new_i;
133 ptiValue * tmp_values = malloc(mtx->cap * mtx->stride * sizeof (ptiValue));
134
135 for(ptiIndex i=0; i<mtx->nrows; ++i) {
136 new_i = mode_map_inds[i];
137 for(ptiIndex j=0; j<mtx->ncols; ++j) {
138 tmp_values[i * mtx->stride + j] = mtx->values[new_i * mtx->stride + j];
139 }
140 }
141

169int ptiAppendMatrix(ptiMatrix *mtx, const ptiValue values[]) {
170 if(mtx->cap <= mtx->nrows) {
171#ifndef MEMCHECK_MODE
172 ptiIndex newcap = mtx->cap + mtx->cap/2;
173#else
174 ptiIndex newcap = mtx->nrows+1;
175#endif
176 ptiValue *newdata;
177#ifdef _ISOC11_SOURCE
178 newdata = aligned_alloc(8 * sizeof (ptiValue), newcap * mtx->stride * sizeof (ptiValue));
179#elif _POSIX_C_SOURCE >= 200112L
180 {
181 int result = posix_memalign((void **) &newdata, 8 * sizeof (ptiValue), newcap * mtx->stride * sizeof (ptiValue));
182 if(result != 0) {
183 newdata = NULL;
184 }
185 }
186#else
187 newdata = malloc(newcap * mtx->stride * sizeof (ptiValue));
188#endif
189 pti_CheckOSError(!newdata, "Mtx Append");
190 memcpy(newdata, mtx->values, mtx->nrows * mtx->stride * sizeof (ptiValue));
191 free(mtx->values);
192 mtx->cap = newcap;
193 mtx->values = newdata;
194 }
195 if(values != NULL) {
196 memcpy(&mtx->values[mtx->nrows * mtx->stride], values, mtx->ncols * sizeof (ptiValue));
197 }
198 ++ mtx->nrows;
199 return 0;
200}

250int ptiMatrixDotMul(ptiMatrix const * A, ptiMatrix const * B, ptiMatrix const * C)
251{
252 ptiIndex nrows = A->nrows;
253 ptiIndex ncols = A->ncols;
254 ptiIndex stride = A->stride;
255 assert(nrows == B->nrows && nrows == C->nrows);
256 assert(ncols == B->ncols && ncols == C->ncols);
257 assert(stride == B->stride && stride == C->stride);
258
259 for(ptiIndex i=0; i < nrows; ++i) {
260 for(ptiIndex j=0; j < ncols; ++j) {
261 C->values[i*stride+j] = A->values[i*stride+j] * B->values[i*stride+j];
262 }
263 }
264
265 return 0;
266}

269int ptiMatrixDotMulSeq(ptiIndex const mode, ptiIndex const nmodes, ptiMatrix ** mats)
270{
271 ptiIndex const nrows = mats[0]->nrows;
272 ptiIndex const ncols = mats[0]->ncols;
273 ptiIndex const stride = mats[0]->stride;
274
275 for(ptiIndex m=1; m<nmodes+1; ++m) {
276 assert(mats[m]->ncols == ncols);
277 assert(mats[m]->nrows == nrows);
278 assert(mats[m]->stride == stride);
279 }
280
281 ptiValue * ovals = mats[nmodes]->values;
282#ifdef HIPARTI_USE_OPENMP
283 #pragma omp parallel for
284#endif
285 for(ptiIndex i=0; i < nrows; ++i) {
286 for(ptiIndex j=0; j < ncols; ++j) {
287 ovals[i * stride + j] = 1;
288 }
289 }
290
291 for(ptiIndex m=1; m < nmodes; ++m) {
292 ptiIndex const pm = (mode + m) % nmodes;
293 ptiValue const * vals = mats[pm]->values;
294#ifdef HIPARTI_USE_OPENMP
295 #pragma omp parallel for
296#endif
297 for(ptiIndex i=0; i < nrows; ++i) {
298 for(ptiIndex j=0; j < ncols; ++j) {
299 ovals[i * stride + j] *= vals[i * stride + j];
300 }
301 }
302 }
303
304 return 0;
305}

398int ptiMatrix2Norm(ptiMatrix * const A, ptiValue * const lambda)
399{
400 ptiIndex const nrows = A->nrows;
401 ptiIndex const ncols = A->ncols;
402 ptiIndex const stride = A->stride;
403 ptiValue * const vals = A->values;
404 ptiValue * buffer_lambda;
405
406#ifdef HIPARTI_USE_OPENMP
407 #pragma omp parallel for
408#endif
409 for(ptiIndex j=0; j < ncols; ++j) {
410 lambda[j] = 0.0;
411 }
412
413#ifdef HIPARTI_USE_OPENMP
414 #pragma omp parallel
415 {
416 int const nthreads = omp_get_num_threads();
417 #pragma omp master
418 {
419 buffer_lambda = (ptiValue *)malloc(nthreads * ncols * sizeof(ptiValue));
420 for(ptiNnzIndex j=0; j < nthreads * ncols; ++j)
421 buffer_lambda[j] = 0.0;
422 }
423 }
424
425 #pragma omp parallel
426 {
427 int const tid = omp_get_thread_num();
428 int const nthreads = omp_get_num_threads();
429 ptiValue * loc_lambda = buffer_lambda + tid * ncols;
430
431 #pragma omp for
432 for(ptiIndex i=0; i < nrows; ++i) {
433 for(ptiIndex j=0; j < ncols; ++j) {
434 loc_lambda[j] += vals[i*stride + j] * vals[i*stride + j];
435 }
436 }
437
438 #pragma omp for
439 for(ptiIndex j=0; j < ncols; ++j) {
440 for(int i=0; i < nthreads; ++i) {
441 lambda[j] += buffer_lambda[i*ncols + j];
442 }
443 }
444 } /* end parallel pragma */
445
446#else
447
448 for(ptiIndex i=0; i < nrows; ++i) {
449 for(ptiIndex j=0; j < ncols; ++j) {
450 lambda[j] += vals[i*stride + j] * vals[i*stride + j];
451 }
452 }
453
454#endif
455
456#ifdef HIPARTI_USE_OPENMP
457 #pragma omp parallel for
458#endif
459 for(ptiIndex j=0; j < ncols; ++j) {
460 lambda[j] = sqrt(lambda[j]);
461 }
462
463#ifdef HIPARTI_USE_OPENMP
464 #pragma omp parallel for
465#endif
466 for(ptiIndex i=0; i < nrows; ++i) {
467 for(ptiIndex j=0; j < ncols; ++j) {
468 vals[i*stride + j] /= lambda[j];
469 }
470 }
471
472
473#ifdef HIPARTI_USE_OPENMP
474 free(buffer_lambda);
475#endif
476
477 return 0;
478}
