25int ptiSparseMatrixMulVector(ptiValueVector * y, const ptiSparseMatrix *mtx, ptiValueVector * x)
26{
27#ifdef HIPARTI_USE_OPENMP
28 #pragma omp parallel for
29#endif
30 for(ptiNnzIndex z = 0; z < mtx->nnz; ++z) {
31 ptiIndex row = mtx->rowind.data[z];
32 ptiIndex col = mtx->colind.data[z];
33
34#ifdef HIPARTI_USE_OPENMP
35 #pragma omp atomic update
36#endif
37 y->data[row] += mtx->values.data[z] * x->data[col];
38 }
39 return 0;
40}

92int ptiOmpSparseMatrixMulVector_Reduce(ptiValueVector * y, ptiValueVector * ybufs, const ptiSparseMatrix *mtx, ptiValueVector * x)
93{
94 int nthreads;
95 #pragma omp parallel
96 nthreads = omp_get_num_threads();
97
98 #pragma omp parallel for // schedule(static)
99 for(ptiNnzIndex z = 0; z < mtx->nnz; ++z) {
100 int tid = omp_get_thread_num();
101 ptiIndex row = mtx->rowind.data[z];
102 ptiIndex col = mtx->colind.data[z];
103 ybufs[tid].data[row] += mtx->values.data[z] * x->data[col];
104 }
105
106 /* Reduction */
107 #pragma omp parallel for schedule(static)
108 for(ptiIndex r=0; r<y->len; ++r) {
109 for(int t=0; t<nthreads; ++t) {
110 y->data[r] += ybufs[t].data[r];
111 }
112 }
113
114 return 0;
115}
