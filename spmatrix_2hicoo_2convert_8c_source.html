22static int ptiEqualWithTwoCoordinates(
23 const ptiIndex * item1,
24 const ptiIndex * item2,
25 const ptiIndex nmodes)
26{
27 ptiIndex i1, i2;
28 for(ptiIndex m=0; m<nmodes; ++m) {
29 i1 = item1[m];
30 i2 = item2[m];
31 if(i1 != i2) {
32 return 0;
33 break;
34 }
35 }
36 return 1;
37}

40static int ptiBlockEnd(
41 ptiIndex * out_item,
42 ptiIndex nmodes,
43 ptiIndex nrows,
44 ptiIndex ncols,
45 const ptiIndex * in_item,
46 const ptiElementIndex sb)
47{
48 ptiAssert(in_item[0] < nrows);
49 out_item[0] = in_item[0]+sb < nrows ? in_item[0]+sb : nrows; // exclusive
50 ptiAssert(in_item[1] < ncols);
51 out_item[1] = in_item[1]+sb < ncols ? in_item[1]+sb : ncols; // exclusive
52
53 return 0;
54}

77int ptiSetKernelPointersMat(
78 ptiNnzIndexVector *kptr,
79 ptiSparseMatrix *mtx,
80 const ptiElementIndex sk_bits)
81{
82 ptiIndex nmodes = 2;
83 ptiNnzIndex nnz = mtx->nnz;
84 ptiNnzIndex k = 0; // count kernels
85 ptiNnzIndex knnz = 0; // #Nonzeros per kernel
86 int result = 0;
87 result = ptiAppendNnzIndexVector(kptr, 0);
88 pti_CheckError(result, "HiSpMtx Convert", NULL);
89
90 ptiIndex * coord = (ptiIndex *)malloc(nmodes * sizeof(*coord));
91 ptiIndex * kernel_coord = (ptiIndex *)malloc(nmodes * sizeof(*kernel_coord));
92 ptiIndex * kernel_coord_prior = (ptiIndex *)malloc(nmodes * sizeof(*kernel_coord_prior));
93
94 /* Process first nnz to get the first kernel_coord_prior */
95 coord[0] = mtx->rowind.data[0]; // first nonzero indices
96 coord[1] = mtx->colind.data[0]; // first nonzero indices
97 result = ptiLocateBeginCoord(kernel_coord_prior, nmodes, coord, sk_bits);
98 pti_CheckError(result, "HiSpMtx Convert", NULL);
99
100 for(ptiNnzIndex z=0; z<nnz; ++z) {
101 coord[0] = mtx->rowind.data[z];
102 coord[1] = mtx->colind.data[z];
103 result = ptiLocateBeginCoord(kernel_coord, nmodes, coord, sk_bits);
104 pti_CheckError(result, "HiSpMtx Convert", NULL);
105
106 if(ptiEqualWithTwoCoordinates(kernel_coord, kernel_coord_prior, nmodes) == 1) {
107 ++ knnz;
108 } else {
109 ++ k;
110 result = ptiAppendNnzIndexVector(kptr, knnz + kptr->data[k-1]);
111 pti_CheckError(result, "HiSpMtx Convert", NULL);
112 for(ptiIndex m=0; m<nmodes; ++m)
113 kernel_coord_prior[m] = kernel_coord[m];
114 knnz = 1;
115 }
116 } // End loop z
117 ptiAssert(k < kptr->len);
118 ptiAssert(kptr->data[kptr->len-1] + knnz == nnz);
119
120 /* Set the last element for kptr */
121 ptiAppendNnzIndexVector(kptr, nnz);
122
123 free(coord);
124 free(kernel_coord);
125 free(kernel_coord_prior);
126
127 return 0;
128}

137static int ptiSetKernelScheduler(
138 ptiIndexVector *kschr,
139 ptiIndex *nkiters,
140 ptiNnzIndexVector * const kptr,
141 ptiSparseMatrix *mtx,
142 const ptiElementIndex sk_bits)
143{
144 ptiIndex nmodes = 2;
145 int result = 0;
146
147 ptiIndex * coord = (ptiIndex *)malloc(nmodes * sizeof(*coord));
148 ptiIndex * kernel_coord = (ptiIndex *)malloc(nmodes * sizeof(*kernel_coord));
149
150 for(ptiNnzIndex k=0; k<kptr->len - 1; ++k) { // superblocks
151 ptiNnzIndex z = kptr->data[k];
152 coord[0] = mtx->rowind.data[z];
153 coord[1] = mtx->colind.data[z];
154 result = ptiLocateBeginCoord(kernel_coord, nmodes, coord, sk_bits);
155 pti_CheckError(result, "HiSpMtx Convert", NULL);
156
157 result = ptiAppendIndexVector(&(kschr[kernel_coord[0]]), k); // only according to column index for this scheduler
158 pti_CheckError(result, "HiSpMtx Convert", NULL);
159 } // End loop k
160
161 free(coord);
162 free(kernel_coord);
163
164 ptiIndex sk = (ptiIndex)pow(2, sk_bits);
165 ptiIndex tmp = 0;
166 ptiIndex kernel_ndim = (mtx->ncols + sk - 1) / sk;
167 for(ptiIndex i=0; i<kernel_ndim; ++i) { // find the maximum length of each row of kschr
168 if(tmp < kschr[i].len)
169 tmp = kschr[i].len;
170 }
171 *nkiters = tmp;
172
173 return 0;
174}

181int ptiPreprocessSparseMatrix(
182 ptiNnzIndexVector * kptr,
183 ptiIndexVector *kschr,
184 ptiIndex *nkiters,
185 ptiSparseMatrix *mtx,
186 const ptiElementIndex sb_bits,
187 const ptiElementIndex sk_bits)
188{
189 ptiNnzIndex nnz = mtx->nnz;
190 int result;
191 ptiTimer timer;
192 ptiNewTimer(&timer, 0);
193
194 /* Sort tsr in a Row-major Block order to get all kernels. Not use Morton-order for kernels: 1. better support for higher-order tensors by limiting kernel size, because Morton key bit <= 128; */
195 ptiStartTimer(timer);
196 ptiSparseMatrixSortIndexRowBlock(mtx, 1, 0, nnz, sk_bits); // Parallelized inside
197 ptiStopTimer(timer);
198 ptiPrintElapsedTime(timer, "\t\trowblock sorting");
199#if PARTI_DEBUG == 3
200 printf("Sorted by ptiSparseTensorSortIndexRowBlock.\n");
201 ptiAssert(ptiDumpSparseMatrix(mtx, 0, stdout) == 0);
202#endif
203
204 ptiStartTimer(timer);
205
206 result = ptiSetKernelPointersMat(kptr, mtx, sk_bits);
207 pti_CheckError(result, "HiSpMtx Preprocess", NULL);
208 result = ptiSetKernelScheduler(kschr, nkiters, kptr, mtx, sk_bits);
209 pti_CheckError(result, "HiSpTns Preprocess", NULL);
210
211 ptiStopTimer(timer);
212 ptiPrintElapsedTime(timer, "\t\tSet Kernel Ptrs");
213
214 ptiStartTimer(timer);
215 /* Sort blocks in each kernel in Morton-order */
216 /* Loop for all kernels, 0-kptr.len for OMP code */
217 #pragma omp parallel for
218 for(ptiNnzIndex k=0; k<kptr->len - 1; ++k) {
219 ptiNnzIndex k_begin = kptr->data[k];
220 ptiNnzIndex k_end = kptr->data[k+1]; // exclusive
221 /* Sort blocks in each kernel in Morton-order */
222 ptiSparseMatrixSortIndexMorton(mtx, 1, k_begin, k_end, sb_bits);
223#if PARTI_DEBUG == 3
224 printf("Kernel %"HIPARTI_PRI_NNZ_INDEX ": Sorted by ptiSparseTensorSortIndexMorton.\n", k);
225 ptiAssert(ptiDumpSparseTensor(tsr, 0, stdout) == 0);
226#endif
227 }
228 ptiStopTimer(timer);
229 ptiPrintElapsedTime(timer, "\t\tMorton sorting");
230
231 ptiFreeTimer(timer);
232
233 return 0;
234}

237int ptiSparseMatrixPartition(
238 ptiSparseMatrixHiCOO *himtx,
239 ptiNnzIndex *max_nnzb,
240 ptiSparseMatrix *mtx,
241 const ptiElementIndex sb_bits)
242{
243 int result;
244 ptiNnzIndex nnz = mtx->nnz;
245 ptiIndex nrows = mtx->nrows;
246 ptiIndex ncols = mtx->ncols;
247 ptiIndex nmodes = 2;
248 ptiElementIndex sb = pow(2, sb_bits);
249
250 /* Temporary storage */
251 ptiIndex * block_begin = (ptiIndex *)malloc(nmodes * sizeof(*block_begin));
252 ptiIndex * block_end = (ptiIndex *)malloc(nmodes * sizeof(*block_end));
253 ptiIndex * block_begin_prior = (ptiIndex *)malloc(nmodes * sizeof(*block_begin_prior));
254 ptiIndex * block_coord = (ptiIndex *)malloc(nmodes * sizeof(*block_coord));
255
256 ptiNnzIndex k_begin, k_end; // #Nonzeros locations
257 ptiNnzIndex nk = 0; // #Kernels
258 ptiNnzIndex nb = 1; // #Blocks // counting from the first nnz
259 ptiNnzIndex nb_tmp = 0;
260 ptiNnzIndex ne = 0; // #Nonzeros per block
261 ptiIndex eindex = 0;
262
263 /* different appending methods:
264 * elements: append every nonzero entry
265 * blocks: append when seeing a new block.
266 * chunks: appending when seeting a new chunk. Notice the boundary of kernels and the last chunk of the whole tensor may be larger than the sc.
267 * kernels: append when seeing a new kernel. Not appending a vector, just write data into an allocated array.
268 */
269 /* Process first nnz */
270 block_coord[0] = mtx->rowind.data[0]; // first nonzero indices
271 block_coord[1] = mtx->colind.data[0]; // first nonzero indices
272 result = ptiLocateBeginCoord(block_begin_prior, nmodes, block_coord, sb_bits);
273 pti_CheckError(result, "HiSpMtx Convert", NULL);
274 ptiAppendBlockIndexVector(&himtx->bindI, (ptiBlockIndex)block_begin_prior[0]);
275 ptiAppendBlockIndexVector(&himtx->bindJ, (ptiBlockIndex)block_begin_prior[1]);
276 ptiAppendNnzIndexVector(&himtx->bptr, 0);
277
278 /* Loop for all kernels, 0 - himtx->kptr.len - 1 */
279 for(ptiNnzIndex k=0; k<himtx->kptr.len - 1; ++k) {
280 k_begin = himtx->kptr.data[k];
281 k_end = himtx->kptr.data[k+1]; // exclusive
282 nb_tmp = k == 0 ? 0: nb;
283 /* Modify kptr pointing to block locations */
284 himtx->kptr.data[k] = nb_tmp;
285 ++ nk;
286
287 /* Loop nonzeros in each kernel */
288 for(ptiNnzIndex z = k_begin; z < k_end; ++z) {
289 // printf("z: %"HIPARTI_PRI_NNZ_INDEX "\n", z);
290
291 block_coord[0] = mtx->rowind.data[z]; // first nonzero indices
292 block_coord[1] = mtx->colind.data[z]; // first nonzero indices
293 // printf("block_coord:\n");
294 // ptiAssert(ptiDumpIndexArray(block_coord, nmodes, stdout) == 0);
295
296 result = ptiLocateBeginCoord(block_begin, nmodes, block_coord, sb_bits);
297 pti_CheckError(result, "HiSpMtx Convert", NULL);
298 // printf("block_begin_prior:\n");
299 // ptiAssert(ptiDumpIndexArray(block_begin_prior, nmodes, stdout) == 0);
300 // printf("block_begin:\n");
301 // ptiAssert(ptiDumpIndexArray(block_begin, nmodes, stdout) == 0);
302
303 result = ptiBlockEnd(block_end, nmodes, nrows, ncols, block_begin, sb); // exclusive
304 pti_CheckError(result, "HiSpMtx Convert", NULL);
305
306 /* Append einds and values */
307 eindex = mtx->rowind.data[z] < (block_begin[0] << sb_bits) ? mtx->rowind.data[z] : mtx->rowind.data[z] - (block_begin[0] << sb_bits);
308 ptiAssert(eindex < sb);
309 ptiAppendElementIndexVector(&himtx->eindI, (ptiElementIndex)eindex);
310 eindex = mtx->colind.data[z] < (block_begin[1] << sb_bits) ? mtx->colind.data[z] : mtx->colind.data[z] - (block_begin[1] << sb_bits);
311 ptiAssert(eindex < sb);
312 ptiAppendElementIndexVector(&himtx->eindJ, (ptiElementIndex)eindex);
313 ptiAppendValueVector(&himtx->values, mtx->values.data[z]);
314
315
316 /* z in the same block with last z */
317 if (ptiEqualWithTwoCoordinates(block_begin, block_begin_prior, nmodes) == 1)
318 {
319 /* ne: #Elements in current block */
320 ++ ne;
321 } else { /* New block */
322 /* ne: #Elements in the last block */
323 /* Append block bptr and bidx */
324 ptiAppendNnzIndexVector(&himtx->bptr, (ptiBlockIndex)z);
325 ptiAppendBlockIndexVector(&himtx->bindI, (ptiBlockIndex)block_begin[0]);
326 ptiAppendBlockIndexVector(&himtx->bindJ, (ptiBlockIndex)block_begin[1]);
327 for(ptiIndex m=0; m<nmodes; ++m)
328 block_begin_prior[m] = block_begin[m];
329
330 ++ nb;
331 ne = 1;
332 } // End new block
333 // printf("nb: %"HIPARTI_PRI_NNZ_INDEX ", ne: %"HIPARTI_PRI_NNZ_INDEX "\n\n", nb, ne);
334
335 } // End z loop
336 } // End k loop
337
338 ptiAssert(nb <= nnz);
339 ptiAssert(nb == himtx->bindI.len);
340 ptiAssert(nk == himtx->kptr.len - 1);
341
342 /* Last element for kptr, bptr */
343 himtx->kptr.data[himtx->kptr.len - 1] = himtx->bptr.len; // change kptr pointing to blocks
344 ptiAppendNnzIndexVector(&himtx->bptr, nnz);
345
346 *max_nnzb = himtx->bptr.data[1] - himtx->bptr.data[0];
347 ptiNnzIndex sum_nnzb = 0;
348 for(ptiIndex i=0; i < himtx->bptr.len - 1; ++i) {
349 ptiNnzIndex nnzb = himtx->bptr.data[i+1] - himtx->bptr.data[i];
350 sum_nnzb += nnzb;
351 if(*max_nnzb < nnzb) {
352 *max_nnzb = nnzb;
353 }
354 }
355 ptiAssert(sum_nnzb == himtx->nnz);
356
357 free(block_begin);
358 free(block_end);
359 free(block_begin_prior);
360 free(block_coord);
361 return 0;
362}

364int ptiSparseMatrixToHiCOO(
365 ptiSparseMatrixHiCOO *himtx,
366 ptiNnzIndex *max_nnzb,
367 ptiSparseMatrix *mtx,
368 const ptiElementIndex sb_bits,
369 const ptiElementIndex sk_bits)
370{
371 ptiAssert(sk_bits >= sb_bits);
372
373 int result;
374 ptiNnzIndex nnz = mtx->nnz;
375 ptiIndex nrows = mtx->nrows;
376 ptiIndex ncols = mtx->ncols;
377 ptiTimer timer;
378 ptiNewTimer(&timer, 0);
379
380 result = ptiNewSparseMatrixHiCOO(himtx, nrows, ncols, nnz, sb_bits, sk_bits);
381 pti_CheckError(result, "HiSpMtx Convert", NULL);
382
383 /* Sort blocks in each kernel in Morton-order */
384 ptiStartTimer(timer);
385 ptiPreprocessSparseMatrix(&himtx->kptr, himtx->kschr, &himtx->nkiters, mtx, sb_bits, sk_bits);
386 // ptiSparseMatrixSortIndexMorton(mtx, 1, 0, nnz, sb_bits);
387 ptiStopTimer(timer);
388 ptiPrintElapsedTime(timer, "\tHiCOO sorting (Morton)");
389
390
391 ptiStartTimer(timer);
392 ptiSparseMatrixPartition(himtx, max_nnzb, mtx, sb_bits);
393 ptiStopTimer(timer);
394 ptiPrintElapsedTime(timer, "Generate HiCOO");
395
396 ptiFreeTimer(timer);
397 return 0;
398}
