32int ptiRankMatrixSolveNormals(
33 ptiIndex const mode,
34 ptiIndex const nmodes,
35 ptiRankMatrix ** aTa,
36 ptiRankMatrix * rhs)
37{
38 int rank = (int)(aTa[0]->ncols);
39 int stride = (int)(aTa[0]->stride);
40
41 ptiRankMatrixDotMulSeqTriangle(mode, nmodes, aTa);
42
43 int info;
44 char uplo = 'L';
45 int nrhs = (int) rhs->nrows;
46 ptiValue * const neqs = aTa[nmodes]->values;
47
48 /* Cholesky factorization */
49 bool is_spd = true;
50 // lapackf77_spotrf(&uplo, &rank, neqs, &stride, &info);
51 spotrf_(&uplo, &rank, neqs, &stride, &info);
52 if(info) {
53 printf("Gram matrix is not SPD. Trying `gesv`.\n");
54 is_spd = false;
55 }
56
57 /* Continue with Cholesky */
58 if(is_spd) {
59 /* Solve against rhs */
60 // lapackf77_spotrs(&uplo, &rank, &nrhs, neqs, &stride, rhs->values, &stride, &info);
61 spotrs_(&uplo, &rank, &nrhs, neqs, &stride, rhs->values, &stride, &info);
62 if(info) {
63 printf("DPOTRS returned %d\n", info);
64 }
65 }
66 else {
67 int * ipiv = (int*)malloc(rank * sizeof(int));
68
69 /* restore gram matrix */
70 ptiRankMatrixDotMulSeqTriangle(mode, nmodes, aTa);
71
72 sgesv_(&rank, &nrhs, neqs, &stride, ipiv, rhs->values, &stride, &info);
73 // lapackf77_sgesv(&rank, &nrhs, neqs, &stride, ipiv, rhs->values, &stride, &info);
74 // magma_sgesv(rank, nrhs, neqs, stride, ipiv, rhs->values, stride, &info);
75 if(info) {
76 printf("magma_sgesv returned %d\n", info);
77 }
78
79 free(ipiv);
80 }
81
82 return 0;
83}
