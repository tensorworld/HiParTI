24double ptiSparseTensorDensity(ptiSparseTensor const * const tsr)
25{
26 double root = pow((double)tsr->nnz, 1./(double)tsr->nmodes);
27 double density = 1.0;
28 for(ptiIndex m=0; m < tsr->nmodes; ++m) {
29 density *= root / (double)tsr->ndims[m];
30 }
31
32 return density;
33}
