22static __global__ void pti_sMulKernel(
23 ptiValue * __restrict__ X_val,
24 ptiNnzIndex X_nnz,
25 ptiValue a)
26{
27 ptiNnzIndex num_loops_nnz = 1;
28 ptiNnzIndex const nnz_per_loop = gridDim.x * blockDim.x;
29 if(X_nnz > nnz_per_loop) {
30 num_loops_nnz = (X_nnz + nnz_per_loop - 1) / nnz_per_loop;
31 }
32
33 const ptiNnzIndex tidx = threadIdx.x;
34 ptiNnzIndex x;
35
36 for(ptiNnzIndex nl=0; nl<num_loops_nnz; ++nl) {
37 x = blockIdx.x * blockDim.x + tidx + nl * nnz_per_loop;
38 if(x < X_nnz) {
39 X_val[x] = a * X_val[x];
40 }
41 }
42
43}

51int ptiCudaSparseTensorMulScalar(ptiSparseTensor *X, ptiValue a)
52{
53 ptiAssert(a != 0.0);
54 int result;
55
56 ptiTimer timer;
57 ptiNewTimer(&timer, 0);
58 double comp_time;
59
60 /* Device memory allocation */
61 ptiValue *X_val = NULL;
62 result = cudaMalloc((void **) &X_val, X->nnz * sizeof (ptiValue));
63 pti_CheckCudaError(result != 0, "Cuda ptins MulScalar");
64
65 /* Device memory copy */
66 cudaMemcpy(X_val, X->values.data, X->nnz * sizeof (ptiValue), cudaMemcpyHostToDevice);
67
68 ptiStartTimer(timer);
69
70 const ptiNnzIndex max_nblocks = 32768;
71 const ptiNnzIndex max_nthreads_per_block = 256;
72
73 ptiNnzIndex nthreadsx = 1;
74 ptiNnzIndex all_nblocks = 0;
75 ptiNnzIndex nblocks = 0;
76
77 if(X->nnz < max_nthreads_per_block) {
78 nthreadsx = X->nnz;
79 nblocks = 1;
80 } else {
81 nthreadsx = max_nthreads_per_block;
82 all_nblocks = (X->nnz + nthreadsx -1) / nthreadsx;
83 if(all_nblocks < max_nblocks) {
84 nblocks = all_nblocks;
85 } else {
86 nblocks = max_nblocks;
87 }
88 }
89 dim3 dimBlock(nthreadsx);
90 printf("all_nblocks: %lu, nthreadsx: %lu\n", all_nblocks, nthreadsx);
91
92 printf("[Cuda ptins MulScalar] pti_sMulKernel<<<%lu, (%lu)>>>\n", nblocks, nthreadsx);
93 pti_sMulKernel<<<nblocks, dimBlock>>>(X_val, X->nnz, a);
94 result = cudaDeviceSynchronize();
95 pti_CheckCudaError(result != 0, "Cuda ptins MulScalar kernel");
96
97 ptiStopTimer(timer);
98 comp_time = ptiPrintElapsedTime(timer, "Cuda ptins MulScalar");
99
100 cudaMemcpy(X->values.data, X_val, X->nnz * sizeof (ptiValue), cudaMemcpyDeviceToHost);
101 ptiFreeTimer(timer);
102
103 result = cudaFree(X_val);
104 pti_CheckCudaError(result != 0, "Cuda ptins MulScalar");
105
106 printf("[GPU CooMulScalar]: %lf\n", comp_time);
107 printf("\n");
108
109 return 0;
110}
