23int ptiNewKruskalTensor(ptiKruskalTensor *ktsr, ptiIndex nmodes, const ptiIndex ndims[], ptiIndex rank)

44void ptiKruskalTensorInverseShuffleIndices(ptiKruskalTensor * ktsr, ptiIndex ** map_inds) {
45 /* Renumber factor matrices rows */
46 ptiIndex new_i;
47 for(ptiIndex m=0; m < ktsr->nmodes; ++m) {
48 ptiMatrix * mtx = ktsr->factors[m];
49 ptiIndex * mode_map_inds = map_inds[m];
50 ptiValue * tmp_values = malloc(mtx->cap * mtx->stride * sizeof (ptiValue));
51
52 for(ptiIndex i=0; i<mtx->nrows; ++i) {
53 new_i = mode_map_inds[i];
54 for(ptiIndex j=0; j<mtx->ncols; ++j) {
55 tmp_values[i * mtx->stride + j] = mtx->values[new_i * mtx->stride + j];
56 }
57 }
58 free(mtx->values);
59 mtx->values = tmp_values;
60 }
61}

77double KruskalTensorFit(
78 ptiSparseTensor const * const ptien,
79 ptiValue const * const __restrict lambda,
80 ptiMatrix ** mats,
81 ptiMatrix ** ata)
82{
83 ptiIndex const nmodes = ptien->nmodes;
84
85 double ptien_normsq = SparseTensorFrobeniusNormSquared(ptien);
86 // printf("ptien_normsq: %lf\n", ptien_normsq);
87 double const norm_mats = KruskalTensorFrobeniusNormSquared(nmodes, lambda, ata);
88 // printf("norm_mats: %lf\n", norm_mats);
89 double const inner = SparseKruskalTensorInnerProduct(nmodes, lambda, mats);
90 // printf("inner: %lf\n", inner);
91 double residual = ptien_normsq + norm_mats - 2 * inner;
92 // printf("residual: %lf\n", residual);
93 if (residual > 0.0) {
94 residual = sqrt(residual);
95 } else {
96 residual = 0.0;
97 }
98 double fit = 1 - (residual / sqrt(ptien_normsq));
99
100 return fit;
101}

105// Column-major.
106/* Compute a Kruskal tensor's norm is compute on "ata"s. Check Tammy's sparse */
107double KruskalTensorFrobeniusNormSquared(
108 ptiIndex const nmodes,
109 ptiValue const * const __restrict lambda,
110 ptiMatrix ** ata) // ata: column-major
111{
112 ptiIndex const rank = ata[0]->ncols;
113 ptiIndex const stride = ata[0]->stride;
114 ptiValue * const __restrict tmp_atavals = ata[nmodes]->values; // Column-major
115 double norm_mats = 0;
116
117#ifdef HIPARTI_USE_OPENMP
118 #pragma omp parallel for
119#endif
120 for(ptiIndex x=0; x < rank*stride; ++x) {
121 tmp_atavals[x] = 1.;
122 }
123
124 /* Compute Hadamard product for all "ata"s */
125 for(ptiIndex m=0; m < nmodes; ++m) {
126 ptiValue const * const __restrict atavals = ata[m]->values;
127#ifdef HIPARTI_USE_OPENMP
128 #pragma omp parallel for
129#endif
130 for(ptiIndex i=0; i < rank; ++i) {
131 for(ptiIndex j=i; j < rank; ++j) {
132 tmp_atavals[i * stride + j] *= atavals[i * stride + j];
133 }
134 }
135 }
136
137 /* compute lambda^T * aTa[MAX_NMODES] * lambda, only compute a half of them because of its symmetric */
138#ifdef HIPARTI_USE_OPENMP
139 #pragma omp parallel for reduction(+:norm_mats)
140#endif
141 for(ptiIndex i=0; i < rank; ++i) {
142 norm_mats += tmp_atavals[i+(i*stride)] * lambda[i] * lambda[i];
143 for(ptiIndex j=i+1; j < rank; ++j) {
144 norm_mats += tmp_atavals[j+(i*stride)] * lambda[i] * lambda[j] * 2;
145 }
146 }
147
148 return fabs(norm_mats);
149}

153// Row-major, compute via MTTKRP result (mats[nmodes]) and mats[nmodes-1].
154double SparseKruskalTensorInnerProduct(
155 ptiIndex const nmodes,
156 ptiValue const * const __restrict lambda,
157 ptiMatrix ** mats)
158{
159 ptiIndex const rank = mats[0]->ncols;
160 ptiIndex const stride = mats[0]->stride;
161 ptiIndex const last_mode = nmodes - 1;
162 ptiIndex const I = mats[last_mode]->nrows;
163
164 // printf("mats[nmodes-1]:\n");
165 // ptiDumpMatrix(mats[nmodes-1], stdout);
166 // printf("mats[nmodes]:\n");
167 // ptiDumpMatrix(mats[nmodes], stdout);
168
169 ptiValue const * const last_vals = mats[last_mode]->values;
170 ptiValue const * const tmp_vals = mats[nmodes]->values;
171 ptiValue * buffer_accum;
172
173 double inner = 0;
174
175 double * const __restrict accum = (double *) malloc(rank*sizeof(*accum));
176
177#ifdef HIPARTI_USE_OPENMP
178 #pragma omp parallel for
179#endif
180 for(ptiIndex r=0; r < rank; ++r) {
181 accum[r] = 0.0;
182 }
183
184#ifdef HIPARTI_USE_OPENMP
185 #pragma omp parallel
186 {
187 int const nthreads = omp_get_num_threads();
188 #pragma omp master
189 {
190 buffer_accum = (ptiValue *)malloc(nthreads * rank * sizeof(ptiValue));
191 for(ptiIndex j=0; j < nthreads * rank; ++j)
192 buffer_accum[j] = 0.0;
193 }
194 }
195#endif
196
197#ifdef HIPARTI_USE_OPENMP
198 #pragma omp parallel
199 {
200 int const tid = omp_get_thread_num();
201 int const nthreads = omp_get_num_threads();
202 ptiValue * loc_accum = buffer_accum + tid * rank;
203
204 #pragma omp for
205 for(ptiIndex i=0; i < I; ++i) {
206 for(ptiIndex r=0; r < rank; ++r) {
207 loc_accum[r] += last_vals[r+(i*stride)] * tmp_vals[r+(i*stride)];
208 }
209 }
210
211 #pragma omp for
212 for(ptiIndex j=0; j < rank; ++j) {
213 for(int i=0; i < nthreads; ++i) {
214 accum[j] += buffer_accum[i*rank + j];
215 }
216 }
217
218 }
219
220#else
221
222 for(ptiIndex i=0; i < I; ++i) {
223 for(ptiIndex r=0; r < rank; ++r) {
224 accum[r] += last_vals[r+(i*stride)] * tmp_vals[r+(i*stride)];
225 }
226 }
227
228#endif
229
230#ifdef HIPARTI_USE_OPENMP
231 #pragma omp parallel for reduction(+:inner)
232#endif
233 for(ptiIndex r=0; r < rank; ++r) {
234 inner += accum[r] * lambda[r];
235 }
236
237#ifdef HIPARTI_USE_OPENMP
238 free(buffer_accum);
239#endif
240
241 return inner;
242}
