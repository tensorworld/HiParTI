40 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CUDA SpTns * Mtx", "shape mismatch");

59 for(ptiNnzIndex i=1; i<fiberidx.len - 1; ++i) {
60 tmp_flen = (fiberidx.data[i+1] - fiberidx.data[i]) - flen;
61 fvar += tmp_flen * tmp_flen;
62 }
63 tmp_flen = (X->nnz - fiberidx.data[fiberidx.len - 1]) - flen;
64 fvar += tmp_flen * tmp_flen;
65 fvar = sqrt(fvar);
66 printf("nfibs: %zu, flen: %.2f, fvar: %.2f\n", fiberidx.len, flen, fvar);

81 ptiValue *U_val = NULL;

110 if(all_nblocks < max_nblocks) {
