25int ptiSparseMatrixMulVectorCSR(ptiValueVector * y, ptiSparseMatrixCSR *csrmtx, ptiValueVector * x)
26{
27#ifdef HIPARTI_USE_OPENMP
28 #pragma omp parallel for
29#endif
30 for(ptiIndex i = 0; i < csrmtx->nrows; ++i) {
31 for(ptiNnzIndex z = csrmtx->rowptr.data[i]; z < csrmtx->rowptr.data[i+1]; ++z) {
32 ptiIndex col = csrmtx->colind.data[z];
33 y->data[i] += csrmtx->values.data[z] * x->data[col];
34 }
35 }
36 return 0;
37}
