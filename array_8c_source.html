75static void pti_QuickSortPairArray(ptiKeyValuePair * kvarray, ptiIndex const l, ptiIndex const r)
76{
77 ptiIndex i, j, p;
78 if(r-l < 2) {
79 return;
80 }
81 p = (l+r) / 2;
82 for(i = l, j = r-1; ; ++i, --j) {
83 while(pti_PairCompareIndices(kvarray, i, p) < 0) {
84 ++i;
85 }
86 while(pti_PairCompareIndices(kvarray, p, j) < 0) {
87 --j;
88 }
89 if(i >= j) {
90 break;
91 }
92 pti_SwapPairs(kvarray, i, j);
93 if(i == p) {
94 p = j;
95 } else if(j == p) {
96 p = i;
97 }
98 }
99
100 pti_QuickSortPairArray(kvarray, l, i);
101 pti_QuickSortPairArray(kvarray, i, r);
102
103}
