33int ptiSemiSparseTensorSetIndices(
34 ptiSemiSparseTensor *dest,
35 ptiNnzIndexVector *fiberidx,
36 ptiSparseTensor *ref
37) {
38 ptiNnzIndex lastidx = ref->nnz;
39 ptiNnzIndex i;
40 ptiIndex m;
41 int result;
42 assert(dest->nmodes == ref->nmodes);
43 ptiSparseTensorSortIndexAtMode(ref, dest->mode, 0);
44 result = ptiNewNnzIndexVector(fiberidx, 0, 0);
45 pti_CheckError(result, "SspTns SetIndices", NULL);
46 dest->nnz = 0;
47 for(i = 0; i < ref->nnz; ++i) {
48 if(lastidx == ref->nnz || pti_SparseTensorCompareExceptMode(ref, lastidx, ref, i, dest->mode) != 0) {
49 for(m = 0; m < dest->nmodes; ++m) {
50 if(m != dest->mode) {
51 result = ptiAppendIndexVector(&dest->inds[m], ref->inds[m].data[i]);
52 pti_CheckError(result, "SspTns SetIndices", NULL);
53 }
54 }
55 lastidx = i;
56 ++dest->nnz;
57 if(fiberidx != NULL) {
58 result = ptiAppendNnzIndexVector(fiberidx, i);
59 pti_CheckError(result, "SspTns SetIndices", NULL);
60 }
61 }
62 }
63 if(fiberidx != NULL) {
64 result = ptiAppendNnzIndexVector(fiberidx, ref->nnz);
65 pti_CheckError(result, "SspTns SetIndices", NULL);
66 }
67 result = ptiResizeMatrix(&dest->values, dest->nnz);
68 pti_CheckError(result, "SspTns SetIndices", NULL);
69 memset(dest->values.values, 0, dest->nnz * dest->stride * sizeof (ptiValue));
70 return 0;
71}
