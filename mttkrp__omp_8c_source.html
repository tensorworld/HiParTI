54int ptiOmpMTTKRP_Init(ptiSparseTensor const * const X,
55 ptiMatrix * mats[], // mats[nmodes] as temporary space.
56 ptiIndex const mats_order[], // Correspond to the mode order of X.
57 ptiIndex const mode)
58{
59
60 ptiIndex const nmodes = X->nmodes;
61
62 ptiNnzIndex const nnz = X->nnz;
63 ptiIndex const * const ndims = X->ndims;
64 ptiValue const * const vals = X->values.data;
65 ptiIndex const stride = mats[0]->stride;
66 ptiValueVector scratch; // Temporary array
67 ptiNewValueVector(&scratch, nnz * stride, nnz * stride);
68 ptiConstantValueVector(&scratch, 0);
69
70 /* Check the mats. */
71 for(ptiIndex i=0; i<nmodes; ++i) {
72 if(mats[i]->ncols != mats[nmodes]->ncols) {
73 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
74 }
75 if(mats[i]->nrows != ndims[i]) {
76 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
77 }
78 }
79
80 ptiIndex const tmpI = mats[mode]->nrows;
81 ptiIndex const R = mats[mode]->ncols;
82 ptiIndex const * const mode_ind = X->inds[mode].data;
83 ptiMatrix * const M = mats[nmodes];
84 ptiValue * const mvals = M->values;
85 memset(mvals, 0, tmpI*stride*sizeof(ptiValue));
86
87 #pragma omp parallel for
88 for(ptiNnzIndex x=0; x<nnz; ++x) {
89
90 ptiIndex times_mat_index = mats_order[1];
91 ptiMatrix * times_mat = mats[times_mat_index];
92 ptiIndex * times_inds = X->inds[times_mat_index].data;
93 ptiIndex tmp_i = times_inds[x];
94 ptiValue const entry = vals[x];
95 for(ptiIndex r=0; r<R; ++r) {
96 scratch.data[x * stride + r] = entry * times_mat->values[tmp_i * stride + r];
97 }
98
99 for(ptiIndex i=2; i<nmodes; ++i) {
100 times_mat_index = mats_order[i];
101 times_mat = mats[times_mat_index];
102 times_inds = X->inds[times_mat_index].data;
103 tmp_i = times_inds[x];
104
105 for(ptiIndex r=0; r<R; ++r) {
106 scratch.data[x * stride + r] *= times_mat->values[tmp_i * stride + r];
107 }
108 }
109
110 }
111
112 for(ptiNnzIndex x=0; x<nnz; ++x) {
113 ptiIndex const mode_i = mode_ind[x];
114 for(ptiIndex r=0; r<R; ++r) {
115 mvals[mode_i * stride + r] += scratch.data[x * stride + r];
116 }
117 }
118
119 ptiFreeValueVector(&scratch);
120
121 return 0;
122}
