30int ptiNewSparseTensor(ptiSparseTensor *tsr, ptiIndex nmodes, const ptiIndex ndims[]) {
31 ptiIndex i;
32 int result;
33 tsr->nmodes = nmodes;
34 tsr->sortorder = malloc(nmodes * sizeof tsr->sortorder[0]);
35 for(i = 0; i < nmodes; ++i) {
36 tsr->sortorder[i] = i;
37 }
38 tsr->ndims = malloc(nmodes * sizeof *tsr->ndims);
39 pti_CheckOSError(!tsr->ndims, "SpTns New");
40 memcpy(tsr->ndims, ndims, nmodes * sizeof *tsr->ndims);
41 tsr->nnz = 0;
42 tsr->inds = malloc(nmodes * sizeof *tsr->inds);
43 pti_CheckOSError(!tsr->inds, "SpTns New");
44 for(i = 0; i < nmodes; ++i) {
45 result = ptiNewIndexVector(&tsr->inds[i], 0, 0);
46 pti_CheckError(result, "SpTns New", NULL);
47 }
48 result = ptiNewValueVector(&tsr->values, 0, 0);
49 pti_CheckError(result, "SpTns New", NULL);
50 return 0;
51}

97double SparseTensorFrobeniusNormSquared(ptiSparseTensor const * const ptien)
98{
99 double norm = 0;
100 ptiValue const * const restrict vals = ptien->values.data;
101
102#ifdef HIPARTI_USE_OPENMP
103 #pragma omp parallel for reduction(+:norm)
104#endif
105 for(ptiNnzIndex n=0; n < ptien->nnz; ++n) {
106 norm += vals[n] * vals[n];
107 }
108 return norm;
109}

112int pti_DistSparseTensor(ptiSparseTensor * tsr,
113 int const nthreads,
114 ptiNnzIndex * const dist_nnzs,
115 ptiIndex * dist_nrows) {
116
117 ptiNnzIndex global_nnz = tsr->nnz;
118 ptiNnzIndex aver_nnz = global_nnz / nthreads;
119 memset(dist_nnzs, 0, nthreads*sizeof(ptiNnzIndex));
120 memset(dist_nrows, 0, nthreads*sizeof(ptiIndex));
121
122 ptiSparseTensorSortIndex(tsr, 0, 1);
123 ptiIndex * ind0 = tsr->inds[0].data;
124
125 int ti = 0;
126 dist_nnzs[0] = 1;
127 dist_nrows[0] = 1;
128 for(ptiNnzIndex x=1; x<global_nnz; ++x) {
129 if(ind0[x] == ind0[x-1]) {
130 ++ dist_nnzs[ti];
131 } else if (ind0[x] > ind0[x-1]) {
132 if(dist_nnzs[ti] < aver_nnz || ti == nthreads-1) {
133 ++ dist_nnzs[ti];
134 ++ dist_nrows[ti];
135 } else {
136 ++ ti;
137 ++ dist_nnzs[ti];
138 ++ dist_nrows[ti];
139 }
140 } else {
141 pti_CheckError(PTIERR_VALUE_ERROR, "SpTns Dist", "tensor unsorted on mode-0");
142 }
143 }
144
145 return 0;
146
147}

210void ptiSparseTensorInvMap(ptiSparseTensor *tsr, ptiIndex ** map_inds)
211{
212 ptiIndex ** tmp_map_inds = (ptiIndex **)malloc(tsr->nmodes * sizeof(ptiIndex**));
213 pti_CheckOSError(!tmp_map_inds, "ptiSparseTensorInvMap");
214 for(ptiIndex m = 0; m < tsr->nmodes; ++m) {
215 tmp_map_inds[m] = (ptiIndex *)malloc(tsr->ndims[m] * sizeof (ptiIndex));
216 pti_CheckError(!tmp_map_inds[m], "ptiSparseTensorInvMap", NULL);
217 }
218
219 ptiIndex tmp_ind;
220 for(ptiIndex m = 0; m < tsr->nmodes; ++m) {
221 ptiIndex loc = 0;
222 for(ptiNnzIndex i = 0; i < tsr->ndims[m]; ++i) {
223 tmp_ind = map_inds[m][i];
224 tmp_map_inds[m][tmp_ind] = loc;
225 ++ loc;
226 }
227 }
228
229 for(ptiIndex m = 0; m < tsr->nmodes; ++m) {
230 for(ptiNnzIndex i = 0; i < tsr->ndims[m]; ++i) {
231 map_inds[m][i] = tmp_map_inds[m][i];
232 }
233 }
234
235 for(ptiIndex m = 0; m < tsr->nmodes; ++m) {
236 free(tmp_map_inds[m]);
237 }
238 free(tmp_map_inds);
239
240}

250void ptiSparseTensorShuffleModes(ptiSparseTensor *tsr, ptiIndex * mode_order)
251{
253 ptiIndex ** tmp_ind = (ptiIndex **)malloc(tsr->nmodes * sizeof(ptiIndex *));
254 ptiIndex * tmp_ndims = (ptiIndex*)malloc(tsr->nmodes * sizeof(ptiIndex));
255 for(ptiIndex m = 0; m < tsr->nmodes; ++m) {
256 tmp_ind[m] = tsr->inds[m].data;
257 tmp_ndims[m] = tsr->ndims[m];
258 }
259
260 for(ptiIndex m = 0; m < tsr->nmodes; ++m) {
261 ptiIndex pm = mode_order[m];
262
263 tsr->ndims[m] = tmp_ndims[pm];
264 tsr->inds[m].data = tmp_ind[m];
265
266 tsr->sortorder[m] = m;
267 }
268
269 free(tmp_ind);
270 free(tmp_ndims);
271}
