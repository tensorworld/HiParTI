61int ptiMTTKRPHiCOO(
62 ptiSparseTensorHiCOO const * const hitsr,
63 ptiMatrix * mats[], // mats[nmodes] as temporary space.
64 ptiIndex const mats_order[], // Correspond to the mode order of X.
65 ptiIndex const mode)
66{
67 ptiIndex const nmodes = hitsr->nmodes;
68
69 if(nmodes == 3) {
70 ptiAssert(ptiMTTKRPHiCOO_3D_Blocked(hitsr, mats, mats_order, mode) == 0);
71 return 0;
72 }
73
74 ptiIndex const * const ndims = hitsr->ndims;
75 ptiValue const * const restrict vals = hitsr->values.data;
76 ptiIndex const stride = mats[0]->stride;
77 ptiValueVector scratch; // Temporary array
78
79 /* Check the mats. */
80 for(ptiIndex i=0; i<nmodes; ++i) {
81 if(mats[i]->ncols != mats[nmodes]->ncols) {
82 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
83 }
84 if(mats[i]->nrows != ndims[i]) {
85 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
86 }
87 }
88
89 ptiIndex const tmpI = mats[mode]->nrows;
90 ptiIndex const R = mats[mode]->ncols;
91 ptiMatrix * const restrict M = mats[nmodes];
92 ptiValue * const restrict mvals = M->values;
93 memset(mvals, 0, tmpI*stride*sizeof(*mvals));
94 ptiNewValueVector(&scratch, R, R);
95
96 ptiIndex * block_coord = (ptiIndex*)malloc(nmodes * sizeof(*block_coord));
97 ptiIndex * ele_coord = (ptiIndex*)malloc(nmodes * sizeof(*ele_coord));
98
99
100 /* Loop kernels */
101 for(ptiIndex k=0; k<hitsr->kptr.len - 1; ++k) {
102 ptiNnzIndex kptr_begin = hitsr->kptr.data[k];
103 ptiNnzIndex kptr_end = hitsr->kptr.data[k+1];
104
105 /* Loop blocks in a kernel */
106 for(ptiIndex b=kptr_begin; b<kptr_end; ++b) {
107 /* Block indices */
108 for(ptiIndex m=0; m<nmodes; ++m)
109 block_coord[m] = hitsr->binds[m].data[b];
110
111 ptiNnzIndex bptr_begin = hitsr->bptr.data[b];
112 ptiNnzIndex bptr_end = hitsr->bptr.data[b+1];
113 /* Loop entries in a block */
114 for(ptiIndex z=bptr_begin; z<bptr_end; ++z) {
115 /* Element indices */
116 for(ptiIndex m=0; m<nmodes; ++m)
117 ele_coord[m] = (block_coord[m] << hitsr->sb_bits) + hitsr->einds[m].data[z];
118
119 /* Multiply the 1st matrix */
120 ptiIndex times_mat_index = mats_order[1];
121 ptiMatrix * times_mat = mats[times_mat_index];
122 ptiIndex tmp_i = ele_coord[times_mat_index];
123 ptiValue const entry = vals[z];
124 for(ptiIndex r=0; r<R; ++r) {
125 scratch.data[r] = entry * times_mat->values[tmp_i * stride + r];
126 }
127 /* Multiply the rest matrices */
128 for(ptiIndex m=2; m<nmodes; ++m) {
129 times_mat_index = mats_order[m];
130 times_mat = mats[times_mat_index];
131 tmp_i = ele_coord[times_mat_index];
132 for(ptiIndex r=0; r<R; ++r) {
133 scratch.data[r] *= times_mat->values[tmp_i * stride + r];
134 }
135 }
136
137 ptiIndex const mode_i = ele_coord[mode];
138 for(ptiIndex r=0; r<R; ++r) {
139 mvals[mode_i * stride + r] += scratch.data[r];
140 }
141 } // End loop entries
142 } // End loop blocks
143
144 } // End loop kernels
145
146
147 free(block_coord);
148 free(ele_coord);
149 ptiFreeValueVector(&scratch);
150
151 return 0;
152}

155/* Very slow version! Slower than COO in Morton order. */
156int ptiMTTKRPHiCOO_3D(
157 ptiSparseTensorHiCOO const * const hitsr,
158 ptiMatrix * mats[], // mats[nmodes] as temporary space.
159 ptiIndex const mats_order[], // Correspond to the mode order of X.
160 ptiIndex const mode)
161{
162 ptiIndex const nmodes = hitsr->nmodes;
163 ptiIndex const * const ndims = hitsr->ndims;
164 ptiValue const * const restrict vals = hitsr->values.data;
165 ptiIndex const stride = mats[0]->stride;
166
167 /* Check the mats. */
168 ptiAssert(nmodes ==3);
169 for(ptiIndex i=0; i<nmodes; ++i) {
170 if(mats[i]->ncols != mats[nmodes]->ncols) {
171 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
172 }
173 if(mats[i]->nrows != ndims[i]) {
174 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
175 }
176 }
177
178 ptiIndex const tmpI = mats[mode]->nrows;
179 ptiIndex const R = mats[mode]->ncols;
180 ptiMatrix * const restrict M = mats[nmodes];
181 ptiValue * const restrict mvals = M->values;
182 memset(mvals, 0, tmpI*stride*sizeof(*mvals));
183
184 ptiIndex times_mat_index_1 = mats_order[1];
185 ptiMatrix * restrict times_mat_1 = mats[times_mat_index_1];
186 ptiIndex times_mat_index_2 = mats_order[2];
187 ptiMatrix * restrict times_mat_2 = mats[times_mat_index_2];
188
189 /* block_coord is reused, no need to store ele_coord for 3D tensors */
190 ptiBlockIndex * block_coord = (ptiBlockIndex*)malloc(nmodes * sizeof(*block_coord));
191
192 ptiIndex mode_i;
193 ptiIndex tmp_i_1, tmp_i_2;
194 ptiValue entry;
195
196 /* Loop kernels */
197 for(ptiIndex k=0; k<hitsr->kptr.len - 1; ++k) {
198 ptiNnzIndex kptr_begin = hitsr->kptr.data[k];
199 ptiNnzIndex kptr_end = hitsr->kptr.data[k+1];
200
201 /* Loop blocks in a kernel */
202 for(ptiIndex b=kptr_begin; b<kptr_end; ++b) {
203 /* Block indices */
204 for(ptiIndex m=0; m<nmodes; ++m)
205 block_coord[m] = hitsr->binds[m].data[b];
206
207 ptiNnzIndex bptr_begin = hitsr->bptr.data[b];
208 ptiNnzIndex bptr_end = hitsr->bptr.data[b+1];
209 /* Loop entries in a block */
210 for(ptiIndex z=bptr_begin; z<bptr_end; ++z) {
211
212 mode_i = (block_coord[mode] << hitsr->sb_bits) + hitsr->einds[mode].data[z];
213 tmp_i_1 = (block_coord[times_mat_index_1] << hitsr->sb_bits) + hitsr->einds[times_mat_index_1].data[z];
214 tmp_i_2 = (block_coord[times_mat_index_2] << hitsr->sb_bits) + hitsr->einds[times_mat_index_2].data[z];
215 entry = vals[z];
216 for(ptiIndex r=0; r<R; ++r) {
217 mvals[mode_i * stride + r] += entry * times_mat_1->values[tmp_i_1 * stride + r] * times_mat_2->values[tmp_i_2 * stride + r];
218 }
219
220 } // End loop entries
221 } // End loop blocks
222
223 } // End loop kernels
224
225
226 free(block_coord);
227
228 return 0;
229}

232int ptiMTTKRPHiCOO_3D_Blocked(
233 ptiSparseTensorHiCOO const * const hitsr,
234 ptiMatrix * mats[], // mats[nmodes] as temporary space.
235 ptiIndex const mats_order[], // Correspond to the mode order of X.
236 ptiIndex const mode)
237{
238 ptiIndex const nmodes = hitsr->nmodes;
239 ptiIndex const * const ndims = hitsr->ndims;
240 ptiValue const * const restrict vals = hitsr->values.data;
241 ptiIndex const stride = mats[0]->stride;
242
243 /* Check the mats. */
244 ptiAssert(nmodes ==3);
245 for(ptiIndex i=0; i<nmodes; ++i) {
246 if(mats[i]->ncols != mats[nmodes]->ncols) {
247 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
248 }
249 if(mats[i]->nrows != ndims[i]) {
250 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
251 }
252 }
253
254 ptiIndex const tmpI = mats[mode]->nrows;
255 ptiIndex const R = mats[mode]->ncols;
256 ptiMatrix * const restrict M = mats[nmodes];
257 ptiValue * const restrict mvals = M->values;
258 memset(mvals, 0, tmpI*stride*sizeof(*mvals));
259
260 ptiIndex times_mat_index_1 = mats_order[1];
261 ptiMatrix * restrict times_mat_1 = mats[times_mat_index_1];
262 ptiIndex times_mat_index_2 = mats_order[2];
263 ptiMatrix * restrict times_mat_2 = mats[times_mat_index_2];
264
265 ptiElementIndex mode_i;
266 ptiElementIndex tmp_i_1, tmp_i_2;
267 ptiValue entry;
268 ptiValue * restrict blocked_mvals;
269 ptiValue * restrict blocked_times_mat_1;
270 ptiValue * restrict blocked_times_mat_2;
271
272 /* Loop kernels */
273 for(ptiIndex k=0; k<hitsr->kptr.len - 1; ++k) {
274 ptiNnzIndex kptr_begin = hitsr->kptr.data[k];
275 ptiNnzIndex kptr_end = hitsr->kptr.data[k+1];
276
277 /* Loop blocks in a kernel */
278 for(ptiIndex b=kptr_begin; b<kptr_end; ++b) {
279
280 blocked_mvals = mvals + (hitsr->binds[mode].data[b] << hitsr->sb_bits) * stride;
281 blocked_times_mat_1 = times_mat_1->values + (hitsr->binds[times_mat_index_1].data[b] << hitsr->sb_bits) * stride;
282 blocked_times_mat_2 = times_mat_2->values + (hitsr->binds[times_mat_index_2].data[b] << hitsr->sb_bits) * stride;
283
284 ptiNnzIndex bptr_begin = hitsr->bptr.data[b];
285 ptiNnzIndex bptr_end = hitsr->bptr.data[b+1];
286 /* Loop entries in a block */
287 for(ptiIndex z=bptr_begin; z<bptr_end; ++z) {
288
289 mode_i = hitsr->einds[mode].data[z];
290 tmp_i_1 = hitsr->einds[times_mat_index_1].data[z];
291 tmp_i_2 = hitsr->einds[times_mat_index_2].data[z];
292 entry = vals[z];
293
294 ptiValue * const restrict bmvals_row = blocked_mvals + mode_i * stride;
295 ptiValue * const restrict blocked_times_mat_1_row = blocked_times_mat_1 + tmp_i_1 * stride;
296 ptiValue * const restrict blocked_times_mat_2_row = blocked_times_mat_2 + tmp_i_2 * stride;
297
298 for(ptiIndex r=0; r<R; ++r) {
299 bmvals_row[r] += entry *
300 blocked_times_mat_1_row[r]
301 * blocked_times_mat_2_row[r];
302 }
303
304 } // End loop entries
305 } // End loop blocks
306
307 } // End loop kernels
308
309 return 0;
310}

313int ptiMTTKRPHiCOO_MatrixTiling(
314 ptiSparseTensorHiCOO const * const hitsr,
315 ptiRankMatrix * mats[], // mats[nmodes] as temporary space.
316 ptiIndex const mats_order[], // Correspond to the mode order of X.
317 ptiIndex const mode)
318{
319 ptiIndex const nmodes = hitsr->nmodes;
320
321 if(nmodes == 3) {
322 ptiAssert(ptiMTTKRPHiCOO_3D_MatrixTiling(hitsr, mats, mats_order, mode) == 0);
323 return 0;
324 }
325 // else if(nmodes == 4) {
326 // ptiAssert(ptiMTTKRPHiCOO_4D_MatrixTiling(hitsr, mats, mats_order, mode) == 0);
327 // return 0;
328 // }
329
330 ptiIndex const * const ndims = hitsr->ndims;
331 ptiValue const * const restrict vals = hitsr->values.data;
332 ptiElementIndex const stride = mats[0]->stride;
333 ptiValueVector scratch; // Temporary array
334
335 /* Check the mats. */
336 for(ptiIndex i=0; i<nmodes; ++i) {
337 if(mats[i]->ncols != mats[nmodes]->ncols) {
338 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->cols != mats[nmodes]->ncols");
339 }
340 if(mats[i]->nrows != ndims[i]) {
341 pti_CheckError(PTIERR_SHAPE_MISMATCH, "CPU HiCOO SpTns MTTKRP", "mats[i]->nrows != ndims[i]");
342 }
343 }
344
345 ptiIndex const tmpI = mats[mode]->nrows;
346 ptiElementIndex const R = mats[mode]->ncols;
347 ptiRankMatrix * const restrict M = mats[nmodes];
348 ptiValue * const restrict mvals = M->values;
349 memset(mvals, 0, tmpI*stride*sizeof(*mvals));
350 ptiNewValueVector(&scratch, R, R);
351
352 ptiValue ** blocked_times_mat = (ptiValue**)malloc(nmodes * sizeof(*blocked_times_mat));
353
354 /* Loop kernels */
355 for(ptiIndex k=0; k<hitsr->kptr.len - 1; ++k) {
356 ptiNnzIndex kptr_begin = hitsr->kptr.data[k];
357 ptiNnzIndex kptr_end = hitsr->kptr.data[k+1];
358
359 /* Loop blocks in a kernel */
360 for(ptiIndex b=kptr_begin; b<kptr_end; ++b) {
361 /* Block indices */
362 for(ptiIndex m=0; m<nmodes; ++m)
363 blocked_times_mat[m] = mats[m]->values + (hitsr->binds[m].data[b] << hitsr->sb_bits) * stride;
364 ptiValue * blocked_mvals = mvals + (hitsr->binds[mode].data[b] << hitsr->sb_bits) * stride;
365
366 ptiNnzIndex bptr_begin = hitsr->bptr.data[b];
367 ptiNnzIndex bptr_end = hitsr->bptr.data[b+1];
368 /* Loop entries in a block */
369 for(ptiIndex z=bptr_begin; z<bptr_end; ++z) {
370
371 /* Multiply the 1st matrix */
372 ptiIndex times_mat_index = mats_order[1];
373 ptiElementIndex tmp_i = hitsr->einds[times_mat_index].data[z];
374 ptiValue const entry = vals[z];
375 #pragma omp simd
376 for(ptiElementIndex r=0; r<R; ++r) {
377 scratch.data[r] = entry * blocked_times_mat[times_mat_index][(ptiBlockMatrixIndex)tmp_i * stride + r];
378 }
379 /* Multiply the rest matrices */
380 for(ptiIndex m=2; m<nmodes; ++m) {
381 times_mat_index = mats_order[m];
382 tmp_i = hitsr->einds[times_mat_index].data[z];
383 #pragma omp simd
384 for(ptiElementIndex r=0; r<R; ++r) {
385 scratch.data[r] *= blocked_times_mat[times_mat_index][(ptiBlockMatrixIndex)tmp_i * stride + r];
386 }
387 }
388
389 ptiElementIndex const mode_i = hitsr->einds[mode].data[z];
390 #pragma omp simd
391 for(ptiElementIndex r=0; r<R; ++r) {
392 blocked_mvals[(ptiBlockMatrixIndex)mode_i * stride + r] += scratch.data[r];
393 }
394 } // End loop entries
395 } // End loop blocks
396
397 } // End loop kernels
398
399
400 free(blocked_times_mat);
401 ptiFreeValueVector(&scratch);
402
403 return 0;
404}
