28struct ptiTagTimer {

30 struct timespec start_timespec;

34int ptiNewTimer(ptiTimer *timer, int use_cuda) {
35 *timer = (ptiTimer) malloc(sizeof **timer);
36 (*timer)->use_cuda = use_cuda;
37 if(use_cuda) {
38 pti_CheckError(3 + PTIERR_CUDA_ERROR, "Timer New", "CUDA support is disabled in this build");
39 }
40 return 0;
41}

43int ptiStartTimer(ptiTimer timer) {
44 int result;
45 if(timer->use_cuda) {
46 pti_CheckError(3 + PTIERR_CUDA_ERROR, "Timer New", "CUDA support is disabled in this build");
47 } else {
48 result = clock_gettime(CLOCK_MONOTONIC, &timer->start_timespec);
49 pti_CheckOSError(result, "Timer New");
50 }
51 return 0;
52}

54int ptiStopTimer(ptiTimer timer) {
55 int result;
56 if(timer->use_cuda) {
57 pti_CheckError(3 + PTIERR_CUDA_ERROR, "Timer New", "CUDA support is disabled in this build");
58 } else {
59 result = clock_gettime(CLOCK_MONOTONIC, &timer->stop_timespec);
60 pti_CheckOSError(result, "Timer New");
61 }
62 return 0;
63}

65double ptiElapsedTime(const ptiTimer timer) {
66 if(timer->use_cuda) {
67 return NAN;
68 } else {
69 return timer->stop_timespec.tv_sec - timer->start_timespec.tv_sec
70 + (timer->stop_timespec.tv_nsec - timer->start_timespec.tv_nsec) * 1e-9;
71 }
72}

74double ptiPrintElapsedTime(const ptiTimer timer, const char *name) {
75 double elapsed_time = ptiElapsedTime(timer);
76 fprintf(stdout, "[%s]: %.9lf s\n", name, elapsed_time);
77 return elapsed_time;
78}

81double ptiPrintAverageElapsedTime(const ptiTimer timer, const int niters, const char *name) {
82 double elapsed_time = ptiElapsedTime(timer) / niters;
83 fprintf(stdout, "[%s]: %.9lf s\n", name, elapsed_time);
84 return elapsed_time;
