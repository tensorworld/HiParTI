22void ptiSparseMatrixStatusHiCOO(ptiSparseMatrixHiCOO *himtx, FILE *fp)
23{
24 fprintf(fp, "HiCOO Sparse Tensor information ---------\n");
25 fprintf(fp, " DIMS=%"HIPARTI_PRI_INDEX "x%"HIPARTI_PRI_INDEX "\n", himtx->nrows, himtx->ncols);
26 fprintf(fp, " NNZ=%"HIPARTI_PRI_NNZ_INDEX "\n", himtx->nnz);
27 fprintf(fp, " sb=%"HIPARTI_PRI_INDEX "\n", (ptiIndex)pow(2, himtx->sb_bits));
28 fprintf(fp, " nb=%"HIPARTI_PRI_NNZ_INDEX "\n", himtx->bptr.len - 1);
29 fprintf(fp, " nk=%"HIPARTI_PRI_NNZ_INDEX "\n", himtx->kptr.len - 1);
30
31 ptiNnzIndex bytes = himtx->nnz * ( sizeof(ptiValue) + 2 * sizeof(ptiElementIndex) );
32 bytes += himtx->bindI.len * 2 * sizeof(ptiBlockIndex);
33 bytes += himtx->bptr.len * sizeof(ptiNnzIndex);
34 bytes += himtx->kptr.len * sizeof(ptiNnzIndex);
35 /* add kschr */
36 ptiIndex sk = (ptiIndex)pow(2, himtx->sk_bits);
37 ptiIndex kernel_ndim = (himtx->nrows + sk - 1)/sk;
38 for(ptiIndex i=0; i < kernel_ndim; ++i) {
39 bytes += himtx->kschr[i].len * sizeof(ptiIndex);
40 }
41 bytes += kernel_ndim * sizeof(ptiIndexVector *);
42
43 char * bytestr = ptiBytesString(bytes);
44 fprintf(fp, " HiCOO-STORAGE=%s\n", bytestr);
45 free(bytestr);
46
47
48 fprintf(fp, "SCHEDULE INFO [KERNEL]: \n");
49 fprintf(fp, "SCHEDULE : %"HIPARTI_PRI_INDEX" x %"HIPARTI_PRI_INDEX"\n", kernel_ndim, himtx->nkiters);
50
51
52 ptiNnzIndex max_nnzk = 0;
53 ptiNnzIndex min_nnzk = HIPARTI_NNZ_INDEX_MAX;
54 ptiNnzIndex sum_nnzk = 0;
55 ptiNnzIndex avg_nnzk = himtx->nnz / (himtx->kptr.len - 1);
56 double std_nnzk = 0.0;
57 for(ptiIndex k=0; k<himtx->kptr.len - 1; ++k) {
58 ptiNnzIndex nnzk = 0;
59 for(ptiIndex b=himtx->kptr.data[k]; b<himtx->kptr.data[k+1]; ++b) {
60 ptiNnzIndex nnzb = himtx->bptr.data[b+1] - himtx->bptr.data[b];
61 nnzk += nnzb;
62 }
63 sum_nnzk += nnzk;
64 std_nnzk += (nnzk - avg_nnzk) * (nnzk - avg_nnzk);
65 if(min_nnzk > nnzk) min_nnzk = nnzk;
66 if(max_nnzk < nnzk) max_nnzk = nnzk;
67 }
68 std_nnzk = sqrt(std_nnzk / (himtx->kptr.len - 1));
69 assert(sum_nnzk == himtx->nnz);
70 fprintf(fp, "Nnzk: Max=%" HIPARTI_PRI_NNZ_INDEX ", Min=%" HIPARTI_PRI_NNZ_INDEX ", Avg=%" HIPARTI_PRI_NNZ_INDEX ", Std: %.1lf\n", max_nnzk, min_nnzk, avg_nnzk, std_nnzk);
71
72 ptiIndex sb = (ptiIndex)pow(2, himtx->sb_bits);
73 ptiNnzIndex max_nnzb = himtx->bptr.data[1] - himtx->bptr.data[0];
74 ptiNnzIndex min_nnzb = himtx->bptr.data[1] - himtx->bptr.data[0];
75 ptiNnzIndex sum_nnzb = 0;
76 double geo_mean_nnzb = 1;
77 ptiNnzIndex nb = himtx->bptr.len - 1;
78 ptiNnzIndex * nnzb_array = (ptiNnzIndex *)malloc(nb * sizeof(* nnzb_array));
79 for(ptiNnzIndex i=0; i < himtx->bptr.len - 1; ++i) {
80 ptiNnzIndex nnzb = himtx->bptr.data[i+1] - himtx->bptr.data[i];
81 // fprintf(fp, "%lu, ", nnzb);
82 if(max_nnzb < nnzb) {
83 max_nnzb = nnzb;
84 }
85 if(min_nnzb > nnzb) {
86 min_nnzb = nnzb;
87 }
88 sum_nnzb += nnzb;
89 geo_mean_nnzb *= pow( (double)nnzb / sb, 1.0/nb );
90 nnzb_array[i] = nnzb;
91 }
92 assert(sum_nnzb == himtx->nnz);
93 ptiNnzIndex avg_nnzb = (ptiNnzIndex)sum_nnzb / (himtx->bptr.len - 1);
94
95 /* Compute median */
96 ptiQuickSortNnzIndexArray(nnzb_array, 0, nb);
97 long median_loc = (nb + 1) / 2 - 1;
98 assert (median_loc >= 0);
99 ptiNnzIndex median_nnzb = nnzb_array[median_loc];
100 free(nnzb_array);
101
102 fprintf(fp, " block nnz info:\n");
103 fprintf(fp, " Nnzb: Max=%" HIPARTI_PRI_NNZ_INDEX ", Min=%" HIPARTI_PRI_NNZ_INDEX ", Avg=%" HIPARTI_PRI_NNZ_INDEX "\n", max_nnzb, min_nnzb, avg_nnzb);
104 fprintf(fp, " cb: Max=%.3lf, Min=%.3lf, Avg=%.3lf\n", (double)max_nnzb / sb, (double)min_nnzb / sb, (double)avg_nnzb / sb);
105 fprintf(fp, " median cb: %.3lf, geometric mean cb: %.3lf\n", (double)median_nnzb / sb, geo_mean_nnzb);
106 fprintf(fp, " alpha_b: %lf\n", (double)(himtx->bptr.len - 1) / himtx->nnz);
107
108}
