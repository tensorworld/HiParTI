21ptiMutexPool * SptMutexAllocCustom(
22 ptiIndex const num_locks,
23 ptiIndex const pad_size)
24{
25#ifdef HIPARTI_USE_OPENMP
26 ptiMutexPool * pool = (ptiMutexPool*)malloc(sizeof(*pool));
27
28 pool->nlocks = num_locks;
29 pool->padsize = pad_size;
30
31 pool->locks = (omp_lock_t*)malloc(num_locks * pad_size * sizeof(*pool->locks));
32 for(ptiIndex l=0; l < num_locks; ++l) {
33 ptiIndex const lock = ptiMutexTranslateId(l, num_locks, pad_size);
34 omp_init_lock(pool->locks + lock);
35 }
36
37 return pool;
38#else
39 fprintf(stderr, "OpenMP support is not enabled.\n");
40 abort();
41#endif
42}
