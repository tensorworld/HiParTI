30int ptiSemiSparseTensorMulMatrix(
31 ptiSemiSparseTensor *Y,
32 const ptiSemiSparseTensor *X,
33 const ptiMatrix *U,
34 ptiIndex mode
35) {
36 int result;
37 ptiIndex *ind_buf;
38 ptiIndex m;
39 ptiNnzIndex i;
40 if(mode >= X->nmodes) {
41 return -1;
42 }
43 if(X->ndims[mode] != U->nrows) {
44 return -1;
45 }
46 // jli: try to avoid malloc in all operation functions.
47 ind_buf = malloc(X->nmodes * sizeof *ind_buf);
48 if(!ind_buf) {
49 return -1;
50 }
51 for(m = 0; m < X->nmodes; ++m) {
52 ind_buf[m] = X->ndims[m];
53 }
54 ind_buf[mode] = U->ncols;
55 // jli: use pre-processing to allocate Y size outside this function.
56 result = ptiNewSemiSparseTensor(Y, X->nmodes, mode, ind_buf);
57 free(ind_buf);
58 if(result) {
59 return result;
60 }
61 for(m = 0; m < Y->nmodes; ++m) {
62 if(m != mode) {
63 ptiFreeIndexVector(&Y->inds[m]);
64 result = ptiCopyIndexVector(&Y->inds[m], &X->inds[m], 1);
65 if(result != 0) {
66 return result;
67 }
68 }
69 }
70 result = ptiResizeMatrix(&Y->values, X->nnz);
71 if(result != 0) {
72 return result;
73 }
74 Y->nnz = X->nnz;
75 memset(Y->values.values, 0, Y->nnz * Y->stride * sizeof (ptiValue));
76 for(i = 0; i < X->nnz; ++i) {
77 ptiIndex r, k;
78 for(k = 0; k < U->ncols; ++k) {
79 for(r = 0; r < U->nrows; ++r) {
80 Y->values.values[i*Y->stride + k] += X->values.values[i*X->stride + r] * U->values[r*U->stride + k];
81 }
82 }
83 }
84 return 0;
85}
