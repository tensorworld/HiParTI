85int ptiCopyValueVector(ptiValueVector *dest, const ptiValueVector *src, int const nt) {
86 int result = ptiNewValueVector(dest, src->len, src->len);
87 pti_CheckError(result, "ValVec Copy", NULL);
88#ifdef HIPARTI_USE_OPENMP
89 #pragma omp parallel for num_threads(nt)
90 for (ptiNnzIndex i=0; i<src->len; ++i) {
91 dest->data[i] = src->data[i];
92 }
93#else
94 memcpy(dest->data, src->data, src->len * sizeof *src->data);
95#endif
96 return 0;
97}

107int ptiAppendValueVector(ptiValueVector *vec, ptiValue const value) {
108 if(vec->cap <= vec->len) {
109#ifndef MEMCHECK_MODE
110 ptiNnzIndex newcap = vec->cap + vec->cap/2;

112 ptiNnzIndex newcap = vec->len+1;

116 vec->cap = newcap;

118 }
119 vec->data[vec->len] = value;

121 return 0;
