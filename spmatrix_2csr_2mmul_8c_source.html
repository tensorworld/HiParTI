25int ptiSparseMatrixMulMatrixCSR(ptiMatrix * C, ptiSparseMatrixCSR *csrmtx, ptiMatrix * B)
26{
27 for(ptiIndex i = 0; i < csrmtx->nrows; ++i) {
28 for(ptiNnzIndex z = csrmtx->rowptr.data[i]; z < csrmtx->rowptr.data[i+1]; ++z) {
29 ptiIndex col = csrmtx->colind.data[z];
30 ptiValue val = csrmtx->values.data[z];
31 for(ptiNnzIndex c = 0; c < B->ncols; ++c) {
32 C->values[i * C->stride + c] += val * B->values[col * B->stride + c];
33 }
34 }
35 }
36 return 0;
37}
